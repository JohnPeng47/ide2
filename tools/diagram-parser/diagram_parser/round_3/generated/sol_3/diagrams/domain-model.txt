|   |   +-- head { line, startColumn, endColumn }

|   +-- selectionRect { startLine, endLine, startColumn, endColumn }

|       +-- PATCH selected region

|       |   +-- updates MappingDocument nodes + sections
|       |   \-- layoutScene(scene, activeBoxId)
|       \-- GENERATE sub-diagram in dragged region
|           +-- promptMappingPocSelection(operation: generate)
|           +-- creates GridContent + DiagramBox with minimumSize
|           \-- insertGridDiagram(document, target ContainerBox)

+-- evidenceProjectionPipeline
|   +-- MappingSelection.selectedNodes[*].sourceRanges
|   |   \-- selectedEvidenceIds[*]
|   +-- projectSelection(ProjectProjectionRecord, pinned target revision, evidence ids)
|   |   +-- matches sourceEvidenceIds -> targetNodeIds
|   |   +-- projected nodes = matched call-tree targets + descendants
|   |   +-- visible nodes = projected nodes + ancestors
|   |   \-- ProjectionSelectionResult
|   |       +-- ascii
|   |       +-- matchedSourceEvidenceIds[*]
|   |       +-- unmatchedSourceEvidenceIds[*]
|   |       +-- targetNodeIds[*]
|   |       +-- visibleNodeIds[*]
|   |       +-- projectedNodeIds[*]
|   |       \-- rows[*] { nodeId, sourceLine }
|   \-- projectCompactions(CompactionProjection, projected call-tree node ids)
|       +-- callTreeNodeId -> file treeLines[*]
|       +-- visible compaction rows = selected rows + ancestors
|       \-- ProjectedCompactionFile[*]
|           +-- sourceFile + sourceSha256
|           +-- ascii
|           +-- selectedLineCount
|           \-- rows[*] { treeLine, selected, sourceRanges[*] }
|               \-- ProjectedSourcePane opens original mapped source

+-- canvasView: DraggableDiagramCanvas
|   +-- transform { offset { x, y }, scale }
|   +-- gestures: pan | pinch | Ctrl/Cmd-wheel zoom
|   +-- keyboard: arrows pan | +/- zoom | 0 center

+-- explorerView: MappingDiagramExplorer

|   \-- traverses active LineGridScene.root to focus a GridBox
|
+-- submenuView: MappingSubmenu
|   +-- edits active Draft.name
|   +-- creates a Representation
|   \-- saves only a valid, dirty, idle Draft
|
\-- localConversation: LocalChatThread (selection-scoped)

    +-- messages[*] { id, role: user | assistant, text }
    +-- pending + progress
    +-- error
    +-- verbosity
    \-- commands: submit follow-up | retry | collapse
