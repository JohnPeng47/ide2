import { type Shape } from './lib';
import { choose, worker } from './lib';
export { type Shape } from './lib';
export function scopes() {
  const index = 7;
  const reads: (() => number)[] = [];
  for (let index = 0; index < 2; index++) reads.push(() => index);
  for (const index of [3, 4]) reads.push(() => index);
  return () => index;
}
export function invoke(value: number) { return choose(value) + worker['run'](value); }
export const shape: Shape = { value: 1 };
