   10 pub fn DoublyLinkedList(comptime T: type) type {
   11     return struct {
   12         const Self = @This();
   13 
   14         /// The type of the node in the list. This makes it easy to get the
   15         /// node type from the list type.
   16         pub const Node = T;
   17 
   18         first: ?*Node = null,
   19         last: ?*Node = null,
   20 
   21         /// Insert a new node after an existing one.
   22         ///
   23         /// Arguments:
   24         ///     node: Pointer to a node in the list.
   25         ///     new_node: Pointer to the new node to insert.
   26         pub inline fn insertAfter(list: *Self, node: *Node, new_node: *Node) void {
   27             new_node.prev = node;
   28             if (node.next) |next_node| {
   29                 // Intermediate node.
   30                 new_node.next = next_node;
   31                 next_node.prev = new_node;
   32             } else {
   33                 // Last element of the list.
   34                 new_node.next = null;
   35                 list.last = new_node;
   36             }
   37             node.next = new_node;
   38         }
   39 
   40         /// Insert a new node before an existing one.
   41         ///
   42         /// Arguments:
   43         ///     node: Pointer to a node in the list.
   44         ///     new_node: Pointer to the new node to insert.
   45         pub inline fn insertBefore(list: *Self, node: *Node, new_node: *Node) void {
   46             new_node.next = node;
   47             if (node.prev) |prev_node| {
   48                 // Intermediate node.
   49                 new_node.prev = prev_node;
   50                 prev_node.next = new_node;
   51             } else {
   52                 // First element of the list.
   53                 new_node.prev = null;
   54                 list.first = new_node;
   55             }
   56             node.prev = new_node;
   57         }
   58 
   59         /// Insert a new node at the end of the list.
   60         ///
   61         /// Arguments:
   62         ///     new_node: Pointer to the new node to insert.
   63         pub inline fn append(list: *Self, new_node: *Node) void {
   64             if (list.last) |last| {
   65                 // Insert after last.
   66                 list.insertAfter(last, new_node);
   67             } else {
   68                 // Empty list.
   69                 list.prepend(new_node);
   70             }
   71         }
   72 
   73         /// Insert a new node at the beginning of the list.
   74         ///
   75         /// Arguments:
   76         ///     new_node: Pointer to the new node to insert.
   77         pub inline fn prepend(list: *Self, new_node: *Node) void {
   78             if (list.first) |first| {
   79                 // Insert before first.
   80                 list.insertBefore(first, new_node);
   81             } else {
   82                 // Empty list.
   83                 list.first = new_node;
   84                 list.last = new_node;
   85                 new_node.prev = null;
   86                 new_node.next = null;
   87             }
   88         }
   89 
   90         /// Remove a node from the list.
   91         ///
   92         /// Arguments:
   93         ///     node: Pointer to the node to be removed.
   94         pub inline fn remove(list: *Self, node: *Node) void {
   95             if (node.prev) |prev_node| {
   96                 // Intermediate node.
   97                 prev_node.next = node.next;
   98             } else {
   99                 // First element of the list.
  100                 list.first = node.next;
  101             }
  102 
  103             if (node.next) |next_node| {
  104                 // Intermediate node.
  105                 next_node.prev = node.prev;
  106             } else {
  107                 // Last element of the list.
  108                 list.last = node.prev;
  109             }
  110         }
  111 
  112         /// Remove and return the last node in the list.
  113         ///
  114         /// Returns:
  115         ///     A pointer to the last node in the list.
  116         pub inline fn pop(list: *Self) ?*Node {
  117             const last = list.last orelse return null;
  118             list.remove(last);
  119             return last;
  120         }
  121 
  122         /// Remove and return the first node in the list.
  123         ///
  124         /// Returns:
  125         ///     A pointer to the first node in the list.
  126         pub inline fn popFirst(list: *Self) ?*Node {
  127             const first = list.first orelse return null;
  128             list.remove(first);
  129             return first;
  130         }
  131     };
  132 }

   65     /// How the backing memory of the embedded Page was allocated. Pool-owned
   66     /// memory is always a full standard-size item from the memory
   67     /// pool, regardless of the page layout size. Heap-owned memory
   68     /// is allocated directly with the page allocator and is exactly
   69     /// `Page.memory.len` bytes.

   72     /// pages can be smaller than the standard size (e.g. compacted

  168     pub fn pagePreservingState(
  169         self: *const Node,
  170         alloc: Allocator,
  171     ) Allocator.Error!PreservedPage {
  172         return switch (self.data) {
  173             .resident => |*page_| .{ .borrowed = page_ },

  247     const RestoreMode = enum {
  248         /// Decode the compressed representation back into the raw mapping.
  249         preserve,
  250 
  251         /// Discard the compressed representation without decoding it. The
  252         /// caller must not read the page contents before overwriting them.
  253         discard,
  254     };
  255 
  256     /// Restore this node to the resident representation.
  257     ///
  258     /// Preserve mode reconstructs the page contents and is used by `page`.
  259     /// Discard mode skips decoding for callers which will overwrite or destroy
  260     /// the page. Both modes recommit the retained mapping, free the encoded
  261     /// allocation, and leave the node in a valid resident state.
  262     noinline fn restore(self: *Node, comptime mode: RestoreMode) *Page {
  263         const compressed = switch (self.data) {
  264             .resident => |*page_| return page_,
  265             .compressed => |*page_| page_,
  266         };
  267 
  268         // Decommit only discarded the physical pages. Recommit prepares the
  269         // still-reserved mapping for decoding or reuse by the caller.
  270         terminal_mem.recommit(compressed.page.memory);
  271 
  272         const restored = switch (mode) {
  273             .preserve => compressed.restore() catch |err| {
  274                 // Compressed nodes retain the immutable encoding and exact
  275                 // output mapping used to create them. Any decode error is an
  276                 // internal codec bug or memory corruption. Keep this switch
  277                 // exhaustive so new decoder errors require classification.
  278                 switch (err) {
  279                     error.TruncatedInput,
  280                     error.InvalidOffset,
  281                     error.OutputTooSmall,
  282                     error.OutputSizeMismatch,
  283                     => {},
  284                 }
  285 
  286                 log.err("failed to restore compressed page err={}", .{err});
  287                 @panic("failed to restore compressed terminal page");
  288             },
  289 
  290             .discard => compressed.page,
  291         };
  292 
  293         // Remove our compressed data
  294         compressed.deinit();
  295 
  296         // We're now a resident, non-compressed node.
  297         self.data = .{ .resident = restored };
  298         return &self.data.resident;
  299     }

  648 pub fn init(
  649     alloc: Allocator,
  650     opts: Options,
  651 ) Allocator.Error!PageList {
  652     const tw = init_tw;
  653     const cols = opts.cols;
  654     const rows = opts.rows;
  655 
  656     // The screen starts with a single page that is the entire viewport,
  657     // and we'll split it thereafter if it gets too large and add more as
  658     // necessary.
  659     try tw.check(.init_memory_pool);
  660     var pool = try MemoryPool.init(
  661         alloc,
  662         pageAllocator(alloc),
  663         page_preheat,
  664     );
  665     errdefer pool.deinit();
  666 
  667     try tw.check(.init_pages);
  668     var page_serial: u64 = 0;
  669     const page_list, const page_size = try initPages(
  670         &pool,
  671         &page_serial,
  672         cols,
  673         rows,
  674     );
  675     errdefer releasePages(&pool, page_list);
  676 
  677     var limits: Limits = .init(cols, rows);
  678     limits.set(.bytes, opts.max_size);
  679     limits.set(.lines, opts.max_lines);
  680 
  681     // We always track our viewport pin to ensure this is never an allocation
  682     try tw.check(.viewport_pin);
  683     const viewport_pin = try pool.pins.create(pool.alloc);
  684     viewport_pin.* = .{ .node = page_list.first.? };
  685 
  686     try tw.check(.viewport_pin_track);
  687     var tracked_pins = try initTrackedPins(pool.alloc, viewport_pin);
  688     errdefer tracked_pins.deinit(pool.alloc);

  727 fn initPages(
  728     pool: *MemoryPool,
  729     serial: *u64,
  730     cols: size.CellCountInt,
  731     rows: size.CellCountInt,
  732 ) Allocator.Error!struct { List, usize } {
  733     const tw = initPages_tw;
  734 
  735     var page_list: List = .{};
  736     var page_size: usize = 0;
  737 
  738     // Add pages as needed to create our initial viewport.
  739     const cap = initialCapacity(cols);
  740     const layout = Page.layout(cap);
  741     const pooled = layout.total_size <= std_size;
  742     const page_alloc = pool.pages.allocator;
  743 
  744     // Guaranteed by comptime checks in initialCapacity but
  745     // redundant here for safety.
  746     assert(layout.total_size <= size.max_page_size);
  747 
  748     // If we have an error, we need to release the pages we created.
  749     errdefer releasePages(pool, page_list);
  750 
  751     var rem = rows;
  752     while (rem > 0) {
  753         try tw.check(.page_node);
  754         const node = try pool.nodes.create();
  755         errdefer pool.nodes.destroy(node);
  756 
  757         const page_buf = if (pooled) buf: {
  758             try tw.check(.page_buf_std);
  759             const buf = try pool.pages.create();
  760             terminal_mem.recommit(buf);
  761             break :buf buf;
  762         } else buf: {
  763             try tw.check(.page_buf_non_std);
  764             break :buf try page_alloc.alignedAlloc(
  765                 u8,
  766                 .fromByteUnits(std.heap.page_size_min),
  767                 layout.total_size,
  768             );
  769         };
  770         errdefer if (pooled)
  771             pool.pages.destroy(page_buf)
  772         else
  773             page_alloc.free(page_buf);
  774 
  775         // In runtime safety modes we have to memset because the Zig allocator
  776         // interface will always memset to 0xAA for undefined. On freestanding
  777         // (WASM), the WasmAllocator reuses freed slots without zeroing since
  778         // only fresh memory.grow pages are guaranteed zero by the WASM spec.
  779         // On native, the OS page allocator (mmap) returns zeroed pages.
  780         if (comptime std.debug.runtime_safety or builtin.target.os.tag == .freestanding)
  781             @memset(page_buf, 0);
  782 
  783         // Initialize the first set of pages to contain our viewport so that
  784         // the top of the first page is always the active area.
  785         node.* = .{
  786             .data = .{ .resident = .initBuf(.init(page_buf), layout) },
  787             .serial = serial.*,
  788             .owned = if (pooled) .pool else .heap,
  789         };
  790         node.page().size.rows = @min(rem, node.capacity().rows);
  791         rem -= node.rows();
  792 
  793         // Add the page to the list
  794         page_list.append(node);
  795         page_size += page_buf.len;
  796         errdefer comptime unreachable;
  797 
  798         // Increment our serial
  799         serial.* += 1;
  800     }
  801 
  802     assert(page_list.first != null);
  803 
  804     return .{ page_list, page_size };
  805 }
  942 fn releasePages(pool: *MemoryPool, list: List) void {
  943     const page_alloc = pool.pages.allocator;
  944     var it = list.first;
  945     while (it) |node| {
  946         it = node.next;
  947         const page = node.restore(.discard);
  948         switch (node.owned) {
  949             .pool => releasePoolPage(pool, page),
  950             .heap => page_alloc.free(page.memory),
  951         }
  952         pool.nodes.destroy(node);
  953     }
  954 }
  955 
  956 /// Release a pool-owned page during a teardown walk. Unlike
  957 /// destroyNodeExt, this does not zero the page: on native, the item goes
  958 /// straight back to the page allocator, and zeroing it first would write
  959 /// the whole page (and fault a decommitted mapping back in) only for it
  960 /// to be unmapped.
  961 fn releasePoolPage(pool: *MemoryPool, page: *const Page) void {
  962     const item: *align(std.heap.page_size_min) [std_size]u8 =
  963         @ptrCast(@alignCast(page.memory.ptr));
  964 
  965     // The wasm pool's items are shared by every pool in the module and
  966     // never return to the allocator, so they go back to the free list
  967     // zeroed for reuse (mirroring destroyNodeExt).
  968     if (comptime wasm_page_pool) {
  969         _ = terminal_mem.decommit(.zero, item, page.memory.len);
  970         pool.pages.destroy(item);
  971         return;
  972     }
  973 
  974     pool.pages.release(item);
  975 }

 1278 pub fn resize(self: *PageList, opts: Resize) Allocator.Error!void {
 1279     defer self.assertIntegrity();
 1280 
 1281     // Resizing forces all nodes to be decompressed today so we need to
 1282     // reschedule compression.
 1283     // TODO(mitchellh): Deferred reflow on non-viewport/non-active pages.
 1284     self.page_compression.reset();
 1285     self.page_compression.markActivity();
 1286 
 1287     if (comptime std.debug.runtime_safety) {
 1288         // Resize does not work with 0 values, this should be protected
 1289         // upstream
 1290         if (opts.cols) |v| assert(v > 0);
 1291         if (opts.rows) |v| assert(v > 0);
 1292     }
 1293 
 1294     // Resizing (especially with reflow) can cause our row offset to
 1295     // become invalid. Rather than do something fancy like we do other
 1296     // places and try to update it in place, we just invalidate it because
 1297     // its too easy to get the logic wrong in here.
 1298     self.viewport_pin_row_offset = null;
 1299 
 1300     if (!opts.reflow) {
 1301         try self.resizeWithoutReflow(opts);
 1302         // Shrinking the active row count turns former active rows into
 1303         // scrollback even without reflow, which can cross the line limit.
 1304         self.limits.enforce(self, .lines, .reclaim);
 1305         return;
 1306     }
 1307 
 1308     // Recalculate our minimum limits. This allows grow to work properly when
 1309     // increasing beyond the explicit limits to fit the active area.
 1310     const old_limits = self.limits;
 1311     self.limits.resize(
 1312         opts.cols orelse self.cols,
 1313         opts.rows orelse self.rows,
 1314     );
 1315     errdefer self.limits = old_limits;
 1316 
 1317     // On reflow, the main thing that causes reflow is column changes. If
 1318     // only rows change, reflow is impossible. So we change our behavior based
 1319     // on the change of columns.
 1320     const cols = opts.cols orelse self.cols;
 1321     switch (std.math.order(cols, self.cols)) {
 1322         .eq => try self.resizeWithoutReflow(opts),
 1323 
 1324         .gt => {
 1325             // We grow rows after cols so that we can do our unwrapping/reflow
 1326             // before we do a no-reflow grow.
 1327             try self.resizeCols(cols, opts);
 1328             try self.resizeWithoutReflow(opts);
 1329         },
 1330 
 1331         .lt => {
 1332             // We first change our row count so that we have the proper amount
 1333             // we can use when shrinking our cols.
 1334             const old_active_top = self.total_rows - self.rows;
 1335             try self.resizeWithoutReflow(opts: {
 1336                 var copy = opts;
 1337                 copy.cols = self.cols;
 1338                 break :opts copy;
 1339             });
 1340 
 1341             // Changing the row count doesn't move any content, only the top of
 1342             // the active area (e.g. when pulling down scrollback), so the
 1343             // cursor's row in the active area moves by the opposite amount.
 1344             var cols_opts = opts;
 1345             if (cols_opts.cursor) |*c| {
 1346                 const screen_y = old_active_top + c.y;
 1347                 const new_active_top = self.total_rows - self.rows;
 1348                 if (screen_y >= new_active_top) c.y = @intCast(screen_y - new_active_top);
 1349             }
 1350 
 1351             try self.resizeCols(cols, cols_opts);
 1352         },
 1353     }
 1354 
 1355     // Various resize operations can change our total row count such
 1356     // that our viewport pin is now in the active area and has insufficient
 1357     // space. We need to check for this case and fix it up.
 1358     switch (self.viewport) {
 1359         .pin => if (self.pinIsActive(self.viewport_pin.*)) {
 1360             self.viewport = .active;
 1361         },
 1362         .active, .top => {},
 1363     }
 1364 
 1365     // Column reflow can change the physical history row count, and a row
 1366     // resize can move the active boundary. Both may expose whole old pages
 1367     // that are now eligible for line-limit pruning.
 1368     self.limits.enforce(self, .lines, .reclaim);
 1369 }
 1372 fn resizeCols(
 1373     self: *PageList,
 1374     cols: size.CellCountInt,
 1375     opts: Resize,
 1376 ) Allocator.Error!void {
 1377     assert(cols != self.cols);
 1378     const cursor = opts.cursor;
 1379 
 1380     // The active area is always the last `rows` rows, so a reflow that
 1381     // changes the number of rows our text needs slides the active area
 1382     // over the content. If we aren't allowed to pull scrollback then we
 1383     // track the top of the active area so we can restore it afterwards.
 1384     const active_top: ?*Pin = if (!opts.pull_scrollback)
 1385         try self.trackPin(self.getTopLeft(.active))
 1386     else
 1387         null;
 1388     defer if (active_top) |p| self.untrackPin(p);
 1389 
 1390     // If we have a cursor position (x,y), then we try under any col resizing
 1391     // to keep the same number remaining active rows beneath it. This is a
 1392     // very special case if you can imagine clearing the screen (i.e.
 1393     // scrollClear), having an empty active area, and then resizing to less
 1394     // cols then we don't want the active area to "jump" to the bottom and
 1395     // pull down scrollback.

 1397         tracked_pin: *Pin,
 1398         untrack: bool,
 1399         remaining_rows: usize,
 1400         wrapped_rows: usize,

 1403             .x = c.x,

 1405         } }) orelse break :cursor null;

 1436             .untrack = c.pin == null,

 1438             .wrapped_rows = wrapped,
 1439         };

 1447     // after preserved_cursor so that the cursor pin can resolve coordinates in
 1448     // the old active coordinate space.

 1450

 1452     const first_rewritten_node = node: {
 1453         const page = self.pages.first.?.page();

 1455             .{ .cols = cols },
 1456         ) catch |err| err: {
 1457             comptime assert(@TypeOf(err) == error{OutOfMemory});
 1458 
 1459             // We verify all maxed out page layouts work.
 1460             var cap = page.capacity;
 1461             cap.cols = cols;
 1462 
 1463             // We're growing columns so we can only get less rows so use
 1464             // the lesser of our capacity and size so we minimize wasted
 1465             // rows.
 1466             cap.rows = @min(page.size.rows, cap.rows);
 1467             break :err cap;
 1468         };
 1469

 1471         node.page().size.rows = 1;
 1472         break :node node;
 1473     };
 1474 
 1475     // We need to grab our rowIterator now before we rewrite our
 1476     // linked list below.
 1477     var it = self.rowIterator(
 1478         .right_down,
 1479         .{ .screen = .{} },
 1480         null,
 1481     );
 1482     errdefer {
 1483         // If an error occurs, we're in a pretty disastrous broken state,
 1484         // but we should still try to clean up our leaked memory. Free
 1485         // any of the remaining orphaned pages from before. If we reflowed
 1486         // successfully this will be null.
 1487         var node_: ?*Node = if (it.chunk) |chunk| chunk.node else null;
 1488         while (node_) |node| {
 1489             node_ = node.next;
 1490             self.destroyNode(node);
 1491         }
 1492     }
 1493 
 1494     // Reflowed source pages are stashed for reuse as destination
 1495     // pages (see recycle_node). Whether we succeed or fail, a stashed
 1496     // node must not outlive the reflow.
 1497     defer if (self.recycle_node) |node| {
 1498         self.recycle_node = null;
 1499         self.destroyNode(node);
 1500     };
 1501 
 1502     // Set our new page as the only page. This orphans the existing pages
 1503     // in the list, but that's fine since we're gonna delete them anyway.
 1504     self.pages.first = first_rewritten_node;
 1505     self.pages.last = first_rewritten_node;
 1506 
 1507     // Reflow all our rows.
 1508     {
 1509         var reflow_cursor: ReflowCursor = .init(first_rewritten_node);
 1510         while (it.next()) |row| {
 1511             try reflow_cursor.reflowRow(
 1512                 self,
 1513                 row,
 1514                 if (preserved_cursor) |c| c.tracked_pin else null,
 1515             );
 1813             // Move any tracked pins from the source.
 1814             if (row_has_pins) {
 1815                 const pin_keys = list.tracked_pins.keys();
 1816                 for (pin_keys) |p| {
 1817                     if (p.node != row.node or
 1818                         p.y != src_y or
 1819                         p.x != x) continue;
 1820 
 1821                     p.node = self.node;
 1822                     p.x = self.x;
 1823                     p.y = self.y;
 1824                 }
 1825             }
 1826 
 1827             if (self.writeCell(
 1828                 list,
 1829                 &cells[x],
 1830                 src_page,
 1831             )) |result| switch (result) {
 1832                 // Wrote the cell, move to the next.
 1833                 .success => x += 1,
 1834 
 1835                 // Wrote the cell but request to skip the next so skip it.
 1836                 // This is used for things like spacers.
 1837                 .skip_next => {
 1838                     // Remap any tracked pins at the skipped position (x+1)
 1839                     // since we won't process that cell in the loop.
 1840                     if (row_has_pins) for (list.tracked_pins.keys()) |p| {
 1841                         if (p.node != row.node or
 1842                             p.y != src_y or
 1843                             p.x != x + 1) continue;
 1844 
 1845                         p.node = self.node;
 1846                         p.x = self.x;
 1847                         p.y = self.y;
 1848                     };
 3347 pub fn scroll(self: *PageList, behavior: Scroll) void {
 3348     defer self.assertIntegrity();
 3349 
 3350     // Special case no-scrollback mode to never allow scrolling.
 3351     if (self.limits.bytes.explicit == 0) {
 3352         self.viewport = .active;
 3353         return;
 3354     }
 3355 
 3356     // Moving the viewport changes which historical pages are visible. Restart
 3357     // traversal so pages which leave the viewport are reconsidered after the
 3358     // renderer's idle delay. False positives from clamped scrolling are cheap.
 3359     defer {
 3360         self.page_compression.reset();
 3361         self.page_compression.markActivity();
 3362     }
 3363 
 3364     switch (behavior) {
 3365         .active => self.viewport = .active,
 3366         .top => self.viewport = .top,
 3367         .pin => |p| {
 3368             if (self.pinIsActive(p)) {
 3369                 self.viewport = .active;
 3370                 return;
 3371             } else if (self.pinIsTop(p)) {
 3372                 self.viewport = .top;
 3373                 return;
 3374             }
 3375 
 3376             self.viewport_pin.* = p;
 3377             self.viewport = .pin;
 3378             self.viewport_pin_row_offset = null; // invalidate cache
 3379         },
 3380         .row => |n| row: {
 3381             // If we're at the top, pin the top.
 3382             if (n == 0) {
 3383                 self.viewport = .top;
 3384                 break :row;
 3385             }
 3386 
 3387             // If we're below the top of the active area, pin the active area.
 3388             if (n >= self.total_rows - self.rows) {
 3389                 self.viewport = .active;
 3390                 break :row;
 3391             }
 3392 
 3393             // See if there are any other faster paths we can take.
 3394             switch (self.viewport) {
 3395                 .top, .active => {},
 3396                 .pin => if (self.viewport_pin_row_offset) |*v| {
 3397                     // If we have a pin and we already calculated a row offset,
 3398                     // then we can efficiently calculate the delta and move
 3399                     // that much from that pin.
 3400                     const delta: isize = delta: {
 3401                         const n_isize: isize = @intCast(n);
 3402                         const v_isize: isize = @intCast(v.*);
 3403                         break :delta n_isize - v_isize;
 3404                     };
 3405                     self.scroll(.{ .delta_row = delta });
 3406                     return;
 3407                 },
 3408             }
 3409 
 3410             // We have an accurate row offset so store it to prevent
 3411             // calculating this again.
 3412             self.viewport_pin_row_offset = n;
 3413             self.viewport = .pin;
 3414 
 3415             // Slow path, we've just got to traverse the linked list and
 3416             // get to our row. As a slight speedup, let's pick the traversal
 3417             // that's likely faster based on our absolute row and total rows.
 3418             const midpoint = self.total_rows / 2;
 3419             if (n < midpoint) {
 3420                 // Iterate forward from the first node.
 3421                 var node_it = self.pages.first;
 3422                 var rem: usize = n;
 3423                 while (node_it) |node| : (node_it = node.next) {
 3424                     if (rem < node.rows()) {
 3425                         self.viewport_pin.* = .{
 3426                             .node = node,
 3427                             .y = std.math.cast(size.CellCountInt, rem) orelse {
 3428                                 self.viewport = .active;
 3429                                 break :row;
 3430                             },
 3431                         };
 3432                         break :row;
 3433                     }
 3434 
 3435                     rem -= node.rows();
 3436                 }
 3437             } else {
 3438                 // Iterate backwards from the last node.
 3439                 var node_it = self.pages.last;
 3440                 var rem: usize = self.total_rows - n;
 3441                 while (node_it) |node| : (node_it = node.prev) {
 3442                     if (rem <= node.rows()) {
 3443                         self.viewport_pin.* = .{
 3444                             .node = node,
 3445                             .y = std.math.cast(size.CellCountInt, node.rows() - rem) orelse {
 3446                                 self.viewport = .active;
 3447                                 break :row;
 3448                             },
 3449                         };
 3450                         break :row;
 3451                     }
 3452 
 3453                     rem -= node.rows();
 3454                 }
 3455             }
 3456 
 3457             // If we reached here, then we couldn't find the offset.
 3458             // This feels impossible? Just clamp to active, screw it lol.
 3459             self.viewport = .active;
 3460         },
 3461         .delta_prompt => |n| self.scrollPrompt(n),
 3462         .delta_row => |n| delta_row: {
 3463             const amount: usize = @abs(n);
 3464 
 3465             switch (self.viewport) {
 3466                 // If we're at the top and we're scrolling backwards,
 3467                 // we don't have to do anything, because there's nowhere to go.
 3468                 .top => if (n <= 0) break :delta_row,
 3469

 3471                 // have to do anything because it'll result in staying in
 3472                 // the active.
 3473                 .active => if (n >= 0) break :delta_row,
 3474 
 3475                 // If we're already a pin type, then we can fast-path our
 3476                 // delta by simply moving the pin. This has the added benefit
 3477                 // that we can update our row offset cache efficiently, too.
 3478                 .pin => switch (std.math.order(n, 0)) {
 3479                     .eq => break :delta_row,
 3480 
 3481                     .lt => switch (self.viewport_pin.upOverflow(amount)) {
 3482                         .offset => |new_pin| {
 3483                             self.viewport_pin.* = new_pin;

 3485                                 v.* -= amount;
 3486                             }
 3487                             break :delta_row;
 3488                         },
 3489 
 3490                         // If we overflow up we're at the top.
 3491                         .overflow => {
 3492                             self.viewport = .top;
 3493                             break :delta_row;
 3494                         },
 3495                     },
 3496 
 3497                     .gt => switch (self.viewport_pin.downOverflow(amount)) {
 3498                         // If we offset its a valid pin but we still have to
 3499                         // check if we're in the active area.
 3500                         .offset => |new_pin| {
 3501                             if (self.pinIsActive(new_pin)) {
 3502                                 self.viewport = .active;
 3503                             } else {
 3504                                 self.viewport_pin.* = new_pin;
 3505                                 if (self.viewport_pin_row_offset) |*v| {
 3506                                     v.* += amount;
 3507                                 }
 3508                             }
 3509                             break :delta_row;
 3510                         },
 3511 
 3512                         // If we overflow down we're at active.
 3513                         .overflow => {
 3514                             self.viewport = .active;
 3515                             break :delta_row;
 3516                         },
 3517                     },
 3518                 },
 3519             }
 3520 
 3521             // Slow path: we have to calculate the new pin by moving
 3522             // from our viewport.
 3523             const top = self.getTopLeft(.viewport);
 3524             const p: Pin = if (n < 0) switch (top.upOverflow(amount)) {
 3525                 .offset => |v| v,
 3526                 .overflow => |v| v.end,
 3527             } else switch (top.downOverflow(amount)) {
 3528                 .offset => |v| v,
 3529                 .overflow => |v| v.end,
 3530             };
 3531 
 3532             // If we are still within the active area, then we pin the
 3533             // viewport to active. This isn't EXACTLY the same behavior as
 3534             // other scrolling because normally when you scroll the viewport
 3535             // is pinned to _that row_ even if new scrollback is created.
 3536             // But in a terminal when you get to the bottom and back into the
 3537             // active area, you usually expect that the viewport will now
 3538             // follow the active area.
 3539             if (self.pinIsActive(p)) {
 3540                 self.viewport = .active;
 3541                 return;
 3542             }
 3543 
 3544             // If we're at the top, then just set the top. This is a lot
 3545             // more efficient everywhere. We must check this after the
 3546             // active check above because we prefer active if they overlap.
 3547             if (self.pinIsTop(p)) {
 3548                 self.viewport = .top;
 3549                 return;
 3550             }
 3551 
 3552             // Pin is not active so we need to track it.
 3553             self.viewport_pin.* = p;
 3554             self.viewport = .pin;
 3555             self.viewport_pin_row_offset = null; // invalidate cache
 3556         },
 3557     }

 3562 fn scrollPrompt(self: *PageList, delta: isize) void {
 3563     // If we aren't jumping any prompts then we don't need to do anything.
 3564     if (delta == 0) return;
 3565     const delta_start: usize = @abs(delta);
 3566     var delta_rem: usize = delta_start;
 3567 
 3568     // We start at the row before or after our viewport depending on the
 3569     // delta so that we don't land back on our current viewport.
 3570     const start_pin = start: {
 3571         const tl = self.getTopLeft(.viewport);
 3572 
 3573         // If we're moving up we can just move the viewport up because
 3574         // promptIterator handles jumpting to the start of prompts.
 3575         if (delta <= 0) break :start tl.up(1) orelse return;
 3576 
 3577         // If we're moving down and we're presently at some kind of
 3578         // prompt, we need to skip all the continuation lines because
 3579         // promptIterator can't know if we're cutoff or continuing.
 3580         var adjusted: Pin = tl.down(1) orelse return;
 3581         if (tl.rowAndCell().row.semantic_prompt != .none) skip: {
 3582             while (adjusted.rowAndCell().row.semantic_prompt == .prompt_continuation) {
 3583                 adjusted = adjusted.down(1) orelse break :skip;
 3584             }
 3585         }
 3586 
 3587         break :start adjusted;
 3588     };
 3589 
 3590     // Go through prompts delta times
 3591     var it = start_pin.promptIterator(
 3592         if (delta > 0) .right_down else .left_up,
 3593         null,
 3594     );

 4061 pub fn grow(self: *PageList) Allocator.Error!?*List.Node {
 4062     defer self.assertIntegrity();
 4063 
 4064     // Growing can move a complete page behind the active boundary.
 4065     self.page_compression.markActivity();
 4066 
 4067     const last = self.pages.last.?;
 4068     if (last.capacity().rows > last.rows()) {
 4069         // Fast path: we have capacity in the last page. The exposed
 4070         // row requires no clearing work here: rows in unused page
 4071         // capacity are always in the default zero state, either
 4072         // because the page memory was never used (pool buffers are
 4073         // zeroed) or because whatever retired the row reset it (see
 4074         // Page.resetRow).
 4075         const page = last.page();
 4076         page.size.rows += 1;
 4077         page.assertIntegrity();
 4078 
 4079         // Increase our total rows by one
 4080         self.total_rows += 1;
 4081 
 4082         // Growing inside the last page moves the active boundary without
 4083         // allocating; that alone can make the first page wholly historical.
 4084         // At a steady line limit this prunes one page for every page of
 4085         // output, and the next new page takes the pruned item straight
 4086         // back out of the pool, so keep its memory.
 4087         self.limits.enforce(self, .lines, .retain);

 4089     }
 4090 
 4091     // Slower path: we have no space, we need to allocate a new page.
 4092 
 4093     // Get the layout first so our failable work is done early.
 4094     // We'll need this for both paths. `grow_hints` is only updated once
 4095     // the new page exists (see `GrowCapacity.need`).

 4100     const new_size = if (pooled) PagePool.item_size else layout.total_size;
 4101

 4103     // When the first page and the new page both come from the pool, we
 4104     // reuse the first page's pool item instead of allocating a new one.
 4105     //
 4106     // We only take this path if we have more than one page since pruning
 4107     // reuses the popped page. It is possible to have a single page and
 4108     // exceed the max size if that page was adjusted to be larger after
 4109     // initial allocation.

 4111         self.pages.first != self.pages.last and

 4115         assert(first != last);
 4116

 4118         self.total_rows -= first.rows();

 4122         // more row below.

 4134                 // pruned page, then we are now at the top.

 4147         // new first page to the top-left, and mark them as garbage.

 4153             p.x = 0;
 4154             p.garbage = true;

 4180         // the bytes it used.

 4182         assert(old_memory.len <= std_size);

 4195         const page = first.page();
 4196         page.size.rows = 1;
 4197         self.pages.insertAfter(last, first);
 4198         self.total_rows += 1;
 4199         self.pushGrowHint(grow_cap.need, grow_cap.used);
 4200 
 4201         // Reusing the node gives it a fresh generation. Do not begin a new
 4202         // page_serial_epoch here: generations are not monotonic in list order,
 4203         // so older live successors may have lower generations. The epoch only
 4204         // advances when reset invalidates the entire list.
 4205         first.serial = self.page_serial;
 4206         self.page_serial += 1;
 4207 
 4208         // We don't update page_size. A pool-owned page always counts as
 4209         // one pool item, whatever its layout.

 4211         page.assertIntegrity();
 4212 
 4213         // An earlier grow may have left us over the byte limit, because
 4214         // pruning would have cut into the active area (see the trimming
 4215         // costs in growCapacity). Reuse swaps one pool item for another,
 4216         // so it can't fix that on its own. Enforce the limit here.
 4217         self.limits.enforce(self, .bytes, .reclaim);
 4218 
 4219         // Byte-limit recycling may leave history above the independent line
 4220         // limit, so enforce it after the recycled page becomes the new tail.
 4221         self.limits.enforce(self, .lines, .reclaim);

 4223     }
 4224 
 4225     // We need to allocate a new memory buffer.
 4226     const next_node = try self.createPage(.{ .cap = cap });
 4227     // we don't errdefer this because we've added it to the linked
 4228     // list and its fine to have dangling unused pages.
 4229     self.pages.append(next_node);
 4230     self.pushGrowHint(grow_cap.need, grow_cap.used);
 4231     const page = next_node.page();
 4232     page.size.rows = 1;
 4233     page.assertIntegrity();
 4234 
 4235     // Record the increased row count
 4236     self.total_rows += 1;
 4237 
 4238     // The prune above removes at most one old page for this new one,
 4239     // and the two may differ in size. A pooled page is always one pool
 4240     // item, while a heap page is as large as its capacity. If a larger
 4241     // page replaced a smaller one we can be over the byte limit, so
 4242     // remove more history until we are back under it.
 4243     self.limits.enforce(self, .bytes, .reclaim);
 4244 
 4245     // Appending a page can cross the line limit and can make the oldest
 4246     // active-boundary page wholly historical. As in the fast path above,
 4247     // the next new page reuses the pruned item, so keep its memory.
 4248     self.limits.enforce(self, .lines, .retain);
 4249     return next_node;
 4250 }
 4911     // Fix up all our tracked pins to point to the new page.
 4912     const pin_keys = self.tracked_pins.keys();

 4914         if (p.node != node) continue;
 4915         p.node = new_node;
 4916     }
 4917 
 4918     // Insert this page and destroy the old page
 4919     self.pages.insertBefore(node, new_node);
 4920     self.pages.remove(node);
 4921     self.destroyNode(node);

 5349 /// Iterate complete historical pages which do not intersect the viewport.
 5350 ///
 5351 /// All boundaries come from PageList pins and Page metadata. Advancing this
 5352 /// iterator never restores a compressed page or reads its backing memory.
 5353 const CompressionIterator = struct {
 5354     current: *List.Node,
 5355     active: *List.Node,
 5356     viewport_first: *List.Node,
 5357     viewport_last: *List.Node,
 5358 
 5359     fn init(self: *const PageList) CompressionIterator {
 5360         return .{
 5361             .current = self.pages.first.?,
 5362             .active = self.getTopLeft(.active).node,
 5363             .viewport_first = self.getTopLeft(.viewport).node,
 5364             .viewport_last = self.getBottomRight(.viewport).?.node,
 5365         };
 5366     }
 5367 
 5368     fn next(self: *CompressionIterator) ?*List.Node {
 5369         while (self.current != self.active) {
 5370             // The viewport is a contiguous node range. Once traversal reaches
 5371             // its first node, advance through the complete visible range and
 5372             // resume at the next offscreen page.
 5373             if (self.current == self.viewport_first) {
 5374                 while (self.current != self.active and
 5375                     self.current != self.viewport_last)
 5376                 {
 5377                     self.current = self.current.next.?;
 5378                 }
 5379 
 5380                 if (self.current == self.active) return null;
 5381                 self.current = self.current.next.?;
 5382                 continue;
 5383             }
 5384 
 5385             const node = self.current;
 5386             self.current = node.next.?;
 5387             return node;
 5388         }
 5389 
 5390         return null;
 5391     }
 5392 
 5393     fn done(self: *const CompressionIterator) bool {
 5394         return self.current == self.active;
 5395     }
 5396 };

 5461 /// Perform one candidate-bounded incremental cold-history compression step.
 5462 fn compressIncremental(self: *PageList) IncrementalCompressionResult {
 5463     const state = &self.page_compression;
 5464 
 5465     // If we can't reclaim virtual memory, compression is unsupported.
 5466     if (!terminal_mem.canReclaim(.strict)) {
 5467         state.reset();
 5468         return .unsupported;
 5469     }
 5470 
 5471     // Find the node following the exact continuation marker within the cold
 5472     // prefix. A missing marker means the list changed between steps, so begin
 5473     // again at the current first page. This lookup does not touch page memory.
 5474     var it: CompressionIterator = .init(self);
 5475     if (state.last_serial) |last_serial| continuation: {
 5476         while (it.next()) |node| {
 5477             // A newly allocated or replacement node appeared before the
 5478             // marker. Restart immediately so that node cannot be skipped.
 5479             if (node.serial >= state.next_serial) break;
 5480 
 5481             // Not a match? Keep looking
 5482             if (node.serial != last_serial) continue;
 5483 
 5484             // Match! The iterator already points to the next offscreen node.
 5485             break :continuation;
 5486         }
 5487 
 5488         // Not found or otherwise invalid. Reset
 5489         state.last_serial = null;
 5490         it = .init(self);
 5491     }
 5492 
 5493     // Keep track of our next_serial
 5494     state.next_serial = self.page_serial;
 5495 
 5496     // We cap the number of pages we look at to do our best to
 5497     // time-bound the incremental compression.
 5498     var inspected_pages: usize = 0;
 5499     while (inspected_pages < incremental_compression_max_inspected) {
 5500         const node = it.next() orelse break;
 5501 
 5502         state.last_serial = node.serial;
 5503         inspected_pages += 1;
 5504 
 5505         // If this page is already compressed, ignore it.
 5506         if (node.isCompressed()) continue;
 5507 
 5508         // Compression is substantially more expensive even if it fails.
 5509         // So we just try it.
 5510         if (self.compressPage(node)) state.flags.did_compress = true;
 5511         break;
 5512     }
 5513 
 5514     // If we didn't reach our active node, then we still have work to do.
 5515     if (!it.done()) return .pending;
 5516 
 5517     // We reached our active node. So we're done, except that we always
 5518     // do one pass after the first success so we can recompress nodes that
 5519     // were possibly decompressed (e.g. by search, inspector, whatever).
 5520     if (!state.flags.verifying or state.flags.did_compress) {
 5521         const activity_serial: u48 = state.activity_serial;
 5522         state.* = .{
 5523             .flags = .{ .verifying = true },
 5524             .activity_serial = activity_serial,
 5525         };
 5526         return .pending;
 5527     }
 5528 
 5529     // Leave the state fresh while idle so later activity naturally begins at
 5530     // the oldest cold page, including pages restored after this pass.
 5531     state.reset();
 5532     return .complete;
 5533 }

 5535 /// Compress every fully historical resident page which is currently cold.
 5536 fn compressFull(self: *PageList) void {
 5537     var it: CompressionIterator = .init(self);
 5538     while (it.next()) |node| {
 5539 
 5540         // Don't restore an already-compressed page just to recompress it.
 5541         if (node.isCompressed()) continue;
 5542 
 5543         // Failure leaves this node resident and unchanged.
 5544         _ = self.compressPage(node);
 5545     }

 5553 fn compressPage(self: *PageList, node: *List.Node) bool {
 5554     // Recompression requires first restoring the raw page and is a policy
 5555     // decision, so this primitive only accepts resident nodes.
 5556     if (node.isCompressed()) return false;
 5557 
 5558     const page = node.page();
 5559 
 5560     // The scratch size is capped just below the representation's break-even
 5561     // point. Codec limits and pages too small to cover the compressed-state
 5562     // overhead simply make this page ineligible for compression.
 5563     const required = compression.Page.requiredScratch(page.memory.len) catch |err|
 5564         switch (err) {
 5565             error.InputTooLarge,
 5566             error.OutputTooSmall,
 5567             => return false,
 5568         };
 5569     if (required == 0) return false;
 5570 
 5571     // Build the compressed candidate without changing the node. This scope is
 5572     // intentional: its defer releases the borrowed or temporary scratch before
 5573     // we attempt to discard the source mapping below. Only the candidate's
 5574     // exact-sized encoded allocation survives the scope.
 5575     const candidate = candidate: {
 5576         var scratch = CompressionScratch.init(
 5577             &self.pool,
 5578             required,
 5579             page.memory.len,
 5580         ) catch |err| switch (err) {
 5581             error.OutOfMemory => return false,
 5582         };
 5583 
 5584         // The encoder writes at most `required` bytes and reports exactly
 5585         // how many on success. Track that so returning the scratch only
 5586         // clears the prefix it dirtied instead of the whole item.
 5587         var dirty_len: usize = required;
 5588         defer scratch.deinit(&self.pool, dirty_len);
 5589 
 5590         var table: compression.lz4.HashTable = undefined;
 5591         const result = compression.Page.init(
 5592             self.pool.alloc,
 5593             page,
 5594             scratch.bytes()[0..required],
 5595             &table,
 5596         ) catch |err| switch (err) {
 5597             error.OutOfMemory,
 5598             error.InputTooLarge,
 5599             error.OutputTooSmall,
 5600             => return false,
 5601         };
 5602         if (result) |compressed| dirty_len = compressed.encoded.len;
 5603         break :candidate result;
 5604     };
 5605 
 5606     // Null means compression crossed the break-even point. The node and its
 5607     // resident mapping are still untouched in this case.
 5608     var compressed = candidate orelse return false;
 5609 
 5610     // Strict decommit is the final fallible step. It either discards the whole
 5611     // raw mapping or leaves it untouched, so failure can safely free the
 5612     // candidate and preserve the resident node exactly as it was.
 5613     const decommit_allowed: bool = allowed: {
 5614         compressPage_tw.check(.decommit) catch |err| switch (err) {
 5615             error.DecommitFailed => break :allowed false,
 5616         };
 5617         break :allowed true;
 5618     };
 5619     if (!decommit_allowed or !terminal_mem.decommit(
 5620         .strict,
 5621         compressed.page.memory,
 5622         compressed.page.memory.len,
 5623     )) {
 5624         compressed.deinit();
 5625         return false;
 5626     }
 5627 
 5628     // Publish the new state only after both the encoded allocation and the
 5629     // retained-mapping decommit have succeeded.
 5630     node.data = .{ .compressed = compressed };
 5631     return true;
 5632 }

 5660 fn destroyNodeExt(
 5661     pool: *MemoryPool,
 5662     node: *List.Node,
 5663     total_size: ?*usize,
 5664     mode: DestroyMode,
 5665 ) void {
 5666     const page = node.restore(.discard);
 5667 
 5668     // Update our accounting for page size. This must mirror what was
 5669     // added at creation time: a pool-owned page always accounts for a
 5670     // full pool item even if its layout is smaller, while a heap-owned
 5671     // page accounts for its exact memory length.
 5672     if (total_size) |v| v.* -= switch (node.owned) {
 5673         .pool => PagePool.item_size,
 5674         .heap => page.memory.len,
 5675     };
 5676 
 5677     switch (node.owned) {
 5678         .pool => {
 5679             assert(page.memory.len <= std_size);
 5680 
 5681             const item: *align(std.heap.page_size_min) [std_size]u8 =
 5682                 @ptrCast(@alignCast(page.memory.ptr));
 5683 
 5684             // A pooled page gets here only when `grow` could not reuse
 5685             // its item in place: `increaseCapacity` replaced it, a heap
 5686             // page took its place at the byte limit, or the line limit
 5687             // pruned it (see `DestroyMode.retain`). Keep a few
 5688             // zeroed items for reuse and give the rest back to the OS.
 5689             // A decommitted item is still mapped and, on Darwin, stays
 5690             // resident until the kernel needs the memory, so an
 5691             // unbounded free list reads as resident memory the size of
 5692             // every pooled page the list ever held at once.
 5693             reuse: {
 5694                 // The wasm pool has no way to give an item back.
 5695                 if (comptime !wasm_page_pool) {
 5696                     if (pool.pages.freeCount() >= page_preheat) {
 5697                         pool.pages.release(item);
 5698                         break :reuse;
 5699                     }
 5700                 }
 5701 
 5702                 // Reset the memory to zero so it can be reused. Unless
 5703                 // the caller wants it kept, also decommit it where
 5704                 // supported.
 5705                 switch (mode) {
 5706                     .reclaim => _ = terminal_mem.decommit(
 5707                         .zero,
 5708                         item,
 5709                         page.memory.len,
 5710                     ),
 5711                     .retain => @memset(page.memory, 0),
 5712                 }
 5713                 pool.pages.destroy(item);
 5714             }
 5715         },
 5716 
 5717         .heap => {
 5718             const page_alloc = pool.pages.allocator;
 5719             page_alloc.free(page.memory);
 5720         },
 5721     }
 5722 
 5723     pool.nodes.destroy(node);
 5724 }
 5896 pub fn eraseRowBounded(
 5897     self: *PageList,
 5898     pt: point.Point,
 5899     limit: usize,
 5900 ) !void {
 5911     var node: *List.Node = pn.node;
 5912     var page = node.page();
 5913     var rows = page.rows.ptr(page.memory.ptr);
 5914 
 5915     // Erasing history may restore compressed pages. Mark unconditionally
 5916     // because incrementing the activity token is cheaper than locating the
 5917     // active boundary.
 5918     self.page_compression.markActivity();
 5919 
 5920     // If the row limit is less than the remaining rows before the end of the
 5921     // page, then we clear the row, rotate it to the end of the boundary limit
 5922     // and update our pins.
 5923     if (node.rows() - pn.y > limit) {
 5924         // Rotating this bounded region changes its cached row coordinates.
 5925         self.invalidateNodeLayout(node);
 5926         page.resetRow(&rows[pn.y]);
 5927         fastmem.rotateOnce(Row, rows[pn.y..][0 .. limit + 1]);
 5928 
 5929         // Mark the whole page as dirty.
 5930         //
 5931         // Technically we only need to mark from the erased row to the
 5932         // limit but this is a hot function, so we want to minimize work.
 5933         page.dirty = true;
 5950         // Update pins in the shifted region.
 5951         const pin_keys = self.tracked_pins.keys();
 5952         for (pin_keys) |p| {
 5953             if (p.node == node and
 5954                 p.y >= pn.y and
 5955                 p.y <= pn.y + limit)
 5956             {
 5957                 if (p.y == 0) {
 5958                     p.x = 0;
 5959                 } else {
 5960                     p.y -= 1;
 5961                 }
 5962             }
 5963         }
 6263 fn erasePage(self: *PageList, node: *List.Node, mode: DestroyMode) void {
 6264     // Must not be the final page.
 6265     assert(node.next != null or node.prev != null);
 6266 
 6267     // We only support erasing from the front or back, never the middle. The
 6268     // public erase operations maintain this contract by construction.
 6269     assert(node.prev == null or node.next == null);
 6270 
 6271     // Update any tracked pins to move to the previous or next page.
 6272     const pin_keys = self.tracked_pins.keys();
 6273     for (pin_keys) |p| {
 6274         if (p.node != node) continue;
 6275         p.node = node.prev orelse node.next orelse unreachable;
 6276         p.y = 0;
 6277         p.x = 0;
 6278 
 6279         // This doesn't get marked garbage because the tracked pin
 6280         // movement is sensical.
 6281     }
 6282 
 6283     // Remove the page from the linked list
 6284     self.pages.remove(node);
 6285     destroyNodeExt(&self.pool, node, &self.page_size, mode);
 6286 }
 6309 /// Convert the given pin to a tracked pin. A tracked pin will always be
 6310 /// automatically updated as the pagelist is modified. If the point the
 6311 /// pin points to is removed completely, the tracked pin will be updated
 6312 /// to the top-left of the screen.
 6313 pub fn trackPin(self: *PageList, p: Pin) Allocator.Error!*Pin {
 6314     if (build_options.slow_runtime_safety) assert(self.pinIsValid(p));
 6315

 6318     errdefer self.pool.pins.destroy(tracked);
 6319     tracked.* = p;
 6320 
 6321     // Add it to the tracked list
 6322     try self.tracked_pins.putNoClobber(self.pool.alloc, tracked, {});
 6323     errdefer _ = self.tracked_pins.remove(tracked);
 6324 
 6325     return tracked;
 6326 }

 6367 pub fn nodeIsValid(
 6368     self: *const PageList,
 6369     target: *List.Node,
 6370     serial: u64,
 6371 ) bool {
 6372     // Reset invalidates the whole prior epoch, so reject those generations
 6373     // without scanning the live list.
 6374     if (serial < self.page_serial_epoch) return false;
 6375 
 6376     var it = self.pages.first;
 6377     while (it) |node| : (it = node.next) {
 6378         if (node == target) return node.serial == serial;
 6379     }
 6380 
 6381     return false;
 6382 }
 7352 /// Get the top-left of the screen for the given tag.
 7353 pub fn getTopLeft(self: *const PageList, tag: point.Tag) Pin {
 7354     return switch (tag) {
 7355         // The full screen or history is always just the first page.
 7356         .screen, .history => .{ .node = self.pages.first.? },
 7357 
 7358         .viewport => switch (self.viewport) {
 7359             .active => self.getTopLeft(.active),
 7360             .top => self.getTopLeft(.screen),
 7361             .pin => self.viewport_pin.*,
 7362         },
 7363 
 7364         // The active area is calculated backwards from the last page.
 7365         // This makes getting the active top left slower but makes scrolling
 7366         // much faster because we don't need to update the top left. Under
 7367         // heavy load this makes a measurable difference.
 7368         .active => active: {
 7369             var rem = self.rows;
 7370             var it = self.pages.last;
 7371             while (it) |node| : (it = node.prev) {
 7372                 if (rem <= node.rows()) break :active .{
 7373                     .node = node,
 7374                     .y = node.rows() - rem,
 7375                 };
 7376 
 7377                 rem -= node.rows();
 7378             }
 7379 
 7380             unreachable; // assertion: we always have enough rows for active
 7381         },
 7382     };
 7383 }
 7600 /// Runtime-configurable byte and line limits for a PageList.
 7601 const Limits = struct {
 7602     bytes: Limit,
 7603     lines: Limit,
 7604 
 7605     /// The limit keys.
 7606     pub const Key = std.meta.FieldEnum(Limits);
 7607 
 7608     pub const Limit = struct {
 7609         /// Explicit is the specified maximum value for this limit
 7610         /// by the user or maxInt otherwise.
 7611         explicit: usize = std.math.maxInt(usize),
 7612 
 7613         /// Min is the minimum valid maximum for this entry, so if
 7614         /// explicit is lower than this then min wins.
 7615         min: usize,
 7616     };
 7617 
 7618     /// Return unlimited limits with minimums for the given PageList size.
 7619     pub fn init(cols: size.CellCountInt, rows: size.CellCountInt) Limits {
 7620         return .{
 7621             .bytes = .{ .min = minMaxSize(cols, rows) },
 7622             .lines = .{ .min = minMaxLines(cols) },
 7623         };
 7624     }
 7625 
 7626     /// Set an explicit limit. Null means unlimited. This does not enforce the
 7627     /// new value automatically; the caller must still call `enforce`.
 7628     pub fn set(
 7629         self: *Limits,
 7630         comptime key: Key,
 7631         value: ?usize,
 7632     ) void {
 7633         switch (key) {
 7634             .bytes => self.bytes.explicit = value orelse std.math.maxInt(usize),
 7635             .lines => self.lines.explicit = value orelse std.math.maxInt(usize),
 7636         }
 7637     }
 7638 
 7639     /// Recalculate the effective minimums for a new PageList size.
 7640     ///
 7641     /// This must be called whenever either PageList dimension changes so the
 7642     /// effective byte and line limits remain valid for the new size.
 7643     pub fn resize(
 7644         self: *Limits,
 7645         cols: size.CellCountInt,
 7646         rows: size.CellCountInt,
 7647     ) void {
 7648         self.bytes.min = minMaxSize(cols, rows);
 7649         self.lines.min = minMaxLines(cols);
 7650     }
 7651 
 7652     /// Return the effective maximum for a limit.
 7653     pub fn max(self: *const Limits, key: Key) usize {
 7654         return switch (key) {
 7655             .bytes => @max(self.bytes.explicit, self.bytes.min),
 7656             .lines => @max(self.lines.explicit, self.lines.min),
 7657         };
 7658     }
 7659 
 7660     /// Whether the given limit is currently exceeded. Both limits are
 7661     /// heuristics: complete historical pages are the smallest unit that
 7662     /// enforcement removes.
 7663     pub fn exceeded(
 7664         self: *const Limits,
 7665         pagelist: *const PageList,
 7666         limit: Key,
 7667     ) bool {
 7668         return switch (limit) {
 7669             .bytes => pagelist.page_size > self.max(.bytes),
 7670             .lines => pagelist.total_rows > pagelist.rows and
 7671                 pagelist.total_rows - pagelist.rows > self.max(.lines),
 7672         };
 7673     }
 7674 
 7675     /// Prune complete historical pages until the selected limit is satisfied
 7676     /// or the oldest remaining page overlaps the active area.
 7677     pub fn enforce(
 7678         self: *const Limits,
 7679         pagelist: *PageList,
 7680         key: Key,
 7681         mode: DestroyMode,
 7682     ) void {
 7683         if (!self.exceeded(pagelist, key)) return;
 7684 
 7685         // A partially constructed clone can temporarily contain fewer rows than
 7686         // its active area. It has no scrollback to prune.
 7687         if (pagelist.total_rows <= pagelist.rows) return;
 7688 
 7689         // Accumulate the row delta so viewport offsets are fixed up once
 7690         // after all eligible pages have been removed.
 7691         var removed: usize = 0;
 7692         while (self.exceeded(pagelist, key)) {
 7693             const first = pagelist.pages.first.?;
 7694 
 7695             // The page containing the active top may also contain history. Keep
 7696             // that boundary page whole even if its history exceeds the heuristic.
 7697             if (first == pagelist.getTopLeft(.active).node) break;
 7698 
 7699             if (removed == 0) pagelist.page_compression.markActivity();
 7700 
 7701             const first_rows = first.rows();
 7702 
 7703             // Automatic pruning invalidates the content represented by pins in
 7704             // the removed page. erasePage remaps them to the next page below but
 7705             // only the enforcing caller knows that their original content is gone,
 7706             // so mark them as garbage here.
 7707             for (pagelist.tracked_pins.keys()) |p| {
 7708                 if (p.node == first) p.garbage = true;
 7709             }
 7710 
 7711             // erasePage updates the list, pin targets, and byte accounting.
 7712             // Row accounting belongs to the caller because erasePage is also used
 7713             // by paths that already adjusted total_rows.
 7714             pagelist.erasePage(first, mode);
 7715             pagelist.total_rows -= first_rows;
 7716             removed += first_rows;
 7717         }
 7718 
 7719         // Reconcile viewport mode and cached row offsets with the combined prefix
 7720         // removal only after every page and pin points into the final list.
 7721         if (removed > 0) pagelist.fixupViewport(removed);
 7722     }

  441 pub fn reset(self: *Screen) void {

  443     self.pages.reset();

  445     // The above reset preserves tracked pins so we can still use
  446     // our cursor pin, which should be at the top-left already. The
  447     // reset marks every tracked pin as garbage, but we keep using

  933 /// Reloads the cursor pointer information into the screen. This is expensive
  934 /// so it should only be done in cases where the pointers are invalidated
  935 /// in such a way that its difficult to recover otherwise.
  936 pub fn cursorReload(self: *Screen) void {
  937     defer self.assertIntegrity();
  938 
  939     // Our tracked pin is ALWAYS accurate, so we derive the active
  940     // point from the pin. If this returns null it means our pin
  941     // points outside the active area. In that case, we update the
  942     // pin to be the top-left.
  943     const pt: point.Point = self.pages.pointFromPin(
  944         .active,
  945         self.cursor.page_pin.*,
  946     ) orelse reset: {
  947         // Our cached row/cell pointers may be invalid (that is often
  948         // the reason cursorReload is being called), so refresh them
  949         // from the pin first since cursorChangePin below marks the
  950         // old cursor row as dirty.
  951         const old_rac = self.cursor.page_pin.rowAndCell();
  952         self.cursor.page_row = old_rac.row;
  953         self.cursor.page_cell = old_rac.cell;
  954 
  955         // The cursor style and hyperlink IDs are only valid within the
  956         // page that the pin points at, so the pin change must go through
  957         // cursorChangePin, which migrates them when the active top-left
  958         // is on a different page. Writing the pin directly here would
  959         // leave the cursor holding IDs that are dead or alias unrelated
  960         // entries on the new page.
  961         const pin = self.pages.pin(.{ .active = .{} }).?;
  962         self.cursor.x = 0; // Must be set before cursorChangePin
  963         self.cursor.y = 0;
  964         self.cursorChangePin(pin);
  965 
  966         // cursorChangePin can trigger a page capacity adjustment which
  967         // moves the pin again, so we re-read it to derive our point.
  968         break :reset self.pages.pointFromPin(
  969             .active,
  970             self.cursor.page_pin.*,
  971         ).?;
  972     };
  973 
  974     self.cursor.x = @intCast(pt.active.x);
  975     self.cursor.y = @intCast(pt.active.y);
  976     const page_rac = self.cursor.page_pin.rowAndCell();
  977     self.cursor.page_row = page_rac.row;
  978     self.cursor.page_cell = page_rac.cell;
  979 }

 1582 inline fn cursorChangePin(self: *Screen, new: Pin) void {
 1583     // Moving the cursor affects text run splitting (ligatures) so
 1584     // we must mark the old and new page dirty. We do this as long
 1585     // as the pins are not equal
 1586     if (!self.cursor.page_pin.eql(new)) {
 1587         self.cursorMarkDirty();
 1588         new.markDirty();
 1589     }
 1590 
 1591     // If our pin is on the same page, then we can just update the pin.
 1592     // We don't need to migrate any state.
 1593     if (self.cursor.page_pin.node == new.node) {
 1594         self.cursor.page_pin.* = new;
 1595         return;
 1596     }
 1597 
 1598     // If we have an old style then we need to release it from the old page.
 1599     const old_style_: ?style.Style = if (self.cursor.style_id == style.default_id)
 1600         null
 1601     else
 1602         self.cursor.style;
 1603     if (old_style_ != null) {
 1604         // Release the style directly from the old page instead of going through
 1605         // manualStyleUpdate, because the cursor position may have already been
 1606         // updated but the pin has not, which would fail integrity checks.
 1607         const old_page: *Page = self.cursor.page_pin.node.page();
 1608         old_page.styles.release(old_page.memory, self.cursor.style_id);
 1609         self.cursor.style = .{};
 1610         self.cursor.style_id = style.default_id;
 1611     }
 1612 
 1613     // If we have a hyperlink then we need to release it from the old page.
 1614     if (self.cursor.hyperlink != null) {
 1615         const old_page: *Page = self.cursor.page_pin.node.page();
 1616         old_page.hyperlink_set.release(old_page.memory, self.cursor.hyperlink_id);
 1617         // Zero the ID, it is invalid now and style changes below may
 1618         // run integrity checks. We still have self.cursor.hyperlink to
 1619         // rebuild this later.
 1620         self.cursor.hyperlink_id = 0;
 1621     }
 1622 
 1623     // Update our pin to the new page
 1624     self.cursor.page_pin.* = new;
 1625 
 1626     // On the new page, we need to migrate our style
 1627     if (old_style_) |old_style| {
 1628         self.cursor.style = old_style;
 1629         self.manualStyleUpdate() catch |err| {
 1630             // This failure should not happen because manualStyleUpdate
 1631             // handles page splitting, overflow, and more. This should only
 1632             // happen if we're out of RAM. In this case, we'll just degrade
 1633             // gracefully back to the default style.
 1634             log.err("failed to update style on cursor change err={}", .{err});
 1635             self.cursor.style = .{};
 1636             self.cursor.style_id = 0;
 1637         };
 1638     }
 1639 
 1640     // On the new page, we need to migrate our hyperlink
 1641     if (self.cursor.hyperlink) |link| {
 1642         // startHyperlink will try to free old hyperlinks, so set this
 1643         // to null. We free it ourselves later since we're doing some
 1644         // ref-counting shenanigans in this function.
 1645         self.cursor.hyperlink = null;
 1646 
 1647         // Re-add
 1648         self.startHyperlink(link.uri, switch (link.id) {
 1649             .explicit => |v| v,
 1650             .implicit => null,
 1651         }) catch |err| {
 1652             // This shouldn't happen because startHyperlink should handle
 1653             // resizing. This only happens if we're truly out of RAM. Degrade
 1654             // to forgetting the hyperlink.
 1655             log.err("failed to update hyperlink on cursor change err={}", .{err});
 1656         };
 1657 
 1658         // Remove our old link
 1659         link.deinit(self.alloc);
 1660         self.alloc.destroy(link);
 1661     }
 1662 }

 3047                 .continuation, .secondary => .prompt_continuation,
 3048             };
 3049         },
 3050     }

 3053 /// Set the selection to the given selection. If this is a tracked selection
 3054 /// then the screen will take ownership of the selection. If this is untracked
 3055 /// then the screen will convert it to tracked internally. This will automatically
 3056 /// untrack the prior selection (if any).
 3057 ///
 3058 /// Set the selection to null to clear any previous selection.
 3059 ///
 3060 /// This is always recommended over setting `selection` directly. Beyond
 3061 /// managing memory for you, it also performs safety checks that the selection
 3062 /// is always tracked.
 3063 pub fn select(self: *Screen, sel_: ?Selection) Allocator.Error!void {
 3064     const sel = sel_ orelse {
 3065         self.clearSelection();
 3066         return;
 3067     };
 3068 
 3069     // If this selection is untracked then we track it.
 3070     const tracked_sel = if (sel.tracked()) sel else try sel.track(self);
 3071     errdefer if (!sel.tracked()) tracked_sel.deinit(self);
 3072 
 3073     // Untrack prior selection pins that aren't also owned by the replacement.
 3074     // A caller may pass our current tracked selection back to us by value, so
 3075     // releasing both old pins unconditionally would leave the replacement
 3076     // pointing at freed pool entries.
 3077     if (self.selection) |old| {
 3078         const new_bounds = tracked_sel.bounds.tracked;
 3079         switch (old.bounds) {
 3080             .untracked => old.deinit(self),
 3081             .tracked => |old_bounds| {
 3082                 if (old_bounds.start != new_bounds.start and
 3083                     old_bounds.start != new_bounds.end)
 3084                 {
 3085                     self.pages.untrackPin(old_bounds.start);
 3086                 }
 3087                 if (old_bounds.end != new_bounds.start and
 3088                     old_bounds.end != new_bounds.end)
 3089                 {
 3090                     self.pages.untrackPin(old_bounds.end);
 3091                 }
 3092             },
 3093         }
 3094     }
 3095     self.selection = tracked_sel;
 3096     self.dirty.selection = true;
 3097 }
 3098 
 3099 /// Same as select(null) but can't fail.
 3100 pub fn clearSelection(self: *Screen) void {
 3101     if (self.selection) |*sel| {
 3102         sel.deinit(self);
 3103         self.dirty.selection = true;
 3104     }
 3105     self.selection = null;
 3106 }

  624 fn printSliceOf(self: *Terminal, comptime Cp: type, cps: []const Cp) !void {
  625     // Check if we can do the fast path up front. If we can't
  626     // we need to go back to scalar `print`.
  627     const fast = fast: {
  628         // Only the main display is supported.
  629         if (self.status_display != .main) break :fast false;
  630 
  631         // Modes that require per-codepoint handling in print().
  632         // Wraparound is required (its the default) so that our
  633         // row-fill logic below can assume soft-wrap semantics. Insert
  634         // mode shifts cells per print.
  635         if (self.modes.get(.insert)) break :fast false;
  636         if (!self.modes.get(.wraparound)) break :fast false;
  637 
  638         // Single shifts require per-codepoint charset handling.
  639         const screen: *Screen = self.screens.active;
  640         if (screen.charset.single_shift != null) break :fast false;
  641 
  642         break :fast true;
  643     };
  644     if (!fast) {
  645         for (cps) |cp| try self.print(@intCast(cp));
  646         return;
  647     }
  648 
  649     // Specialized on whether a hyperlink is open so that unlinked text
  650     // doesn't pay for the linking. A slice never spans an OSC 8 open or
  651     // close, so this holds for the whole slice.
  652     if (self.screens.active.cursor.hyperlink_id != 0) {
  653         return @call(.never_inline, printSliceBatched, .{ self, true, Cp, cps });
  654     }
  655     return @call(.always_inline, printSliceBatched, .{ self, false, Cp, cps });
  656 }
 1893 pub fn print(self: *Terminal, c: u21) !void {
 1906     // The charset this character is printed with. A single shift applies to
 1907     // exactly one character, so it is used up here, whether the character
 1908     // takes a cell or attaches to the previous one: xterm clears it for a
 1909     // combining character and so do we.
 1910     const charset: charsets.Charset = charset: {
 1911         const state = &self.screens.active.charset;
 1912         const key = state.single_shift orelse state.gl;
 1913         state.single_shift = null;

 1915     };
 1916 
 1917     // Our right margin depends where our cursor is now. A cursor right
 1918     // of the right margin prints up to the screen edge. Every printWrap
 1919     // below moves the cursor to the left margin, inside the margins, so
 1920     // each one resets this to the right margin (as xterm's dotext does).
 1921     var right_limit: usize = if (self.screens.active.cursor.x > self.scrolling_region.right)
 1922         self.cols

 1925

 1928     // purposely ordered in least-likely to most-likely so we can drop out

 1932         self.screens.active.cursor.x > 0)
 1933     grapheme: {
 2117             log.debug("c={X} grapheme attach to left={} primary_cp={X}", .{
 2118                 c,
 2119                 prev.left,

 2121             });
 2122             self.screens.active.cursorMarkDirty();

 2129     // non-single-width characters properly. We have a fast-path for
 2258     // If we're at the column limit, then we need to wrap the next time.

 2451 fn printWrap(self: *Terminal) !void {
 2452     // We only mark that we soft-wrapped if we're at the edge of our
 2453     // full screen. We don't mark the row as wrapped if we're in the
 2454     // middle due to a right margin.

 2456     const mark_wrap = cursor.x == self.cols - 1;
 2457     if (mark_wrap) cursor.page_row.wrap = true;
 2458

 2460     // line. We need to do this before we index() because we may
 2461     // modify memory.
 2462     const old_semantic = cursor.semantic_content;
 2463     const old_semantic_clear = cursor.semantic_content_clear_eol;
 2464 
 2465     // Move to the next line
 2466     try self.index();
 2467     self.screens.active.cursorHorizontalAbsolute(self.scrolling_region.left);
 2468 
 2469     // Our pointer should never move
 2470     assert(cursor == &self.screens.active.cursor);
 2471

 2474     cursor.semantic_content_clear_eol = old_semantic_clear;

 2476         .output, .input => {},
 2477         .prompt => cursor.page_row.semantic_prompt = .prompt_continuation,
 2478     }
 2479 
 2480     if (mark_wrap) {
 2481         const row = self.screens.active.cursor.page_row;
 2482         // Always mark the row as a continuation
 2483         row.wrap_continuation = true;
 2484     }

 2487     self.screens.active.assertIntegrity();

 3019

 3021     screen.cursor.pending_wrap = false;
 3022

 4912 pub fn resize(
 4913     self: *Terminal,
 4914     alloc: Allocator,
 4915     opts: Resize,
 4916 ) ResizeError!void {
 4917     const tw = resize_tw;
 4918 
 4919     // Screen and scrolling-region invariants require non-zero dimensions.
 4920     // Validate before changing any terminal state.
 4921     if (opts.cols == 0 or opts.rows == 0) return error.InvalidValue;
 4922 
 4923     // Pixel geometry and synchronized output are updated on every valid
 4924     // resize attempt, including one that doesn't change the grid dimensions.
 4925     // Save their old values so later allocation failures can roll them back.
 4926     const old_width_px = self.width_px;
 4927     const old_height_px = self.height_px;
 4928     const old_synchronized_output = self.modes.get(.synchronized_output);
 4929     errdefer {
 4930         self.width_px = old_width_px;
 4931         self.height_px = old_height_px;
 4932         self.modes.set(.synchronized_output, old_synchronized_output);
 4933     }
 4934 
 4935     // If our pixel geometry was set, then we set it even if our rows/cols
 4936     // didn't change.
 4937     if (opts.cell_size_px) |cell_size| {
 4938         self.width_px = std.math.mul(
 4939             u32,
 4940             opts.cols,
 4941             cell_size.width,
 4942         ) catch std.math.maxInt(u32);
 4943         self.height_px = std.math.mul(
 4944             u32,
 4945             opts.rows,
 4946             cell_size.height,
 4947         ) catch std.math.maxInt(u32);
 4948     }
 4949 
 4950     self.modes.set(.synchronized_output, false);
 4951 
 4952     // If our cols/rows didn't change, skip grid work but still apply pixels.
 4953     if (self.cols == opts.cols and self.rows == opts.rows) return;
 4954 
 4955     // Build replacement tabstops without touching the current table. Keep
 4956     // ownership here until every fallible resize operation has succeeded.
 4957     var new_tabstops: ?Tabstops = null;
 4958     errdefer if (new_tabstops) |*v| v.deinit(alloc);
 4959     if (self.cols != opts.cols) {
 4960         try tw.check(.tabstops);
 4961         new_tabstops = try .init(
 4962             alloc,
 4963             opts.cols,
 4964             TABSTOP_INTERVAL,
 4965         );
 4966     }
 4967 
 4968     // Resize primary screen, which supports reflow. We do this first
 4969     // because the cleanup situation is a lot better if this succeeds
 4970     // and alt fails than the reverse.
 4971     try tw.check(.primary_screen);
 4972     const primary = self.screens.get(.primary).?;
 4973     try primary.resize(.{
 4974         .cols = opts.cols,
 4975         .rows = opts.rows,
 4976         .reflow = self.modes.get(.wraparound),
 4977         .prompt_redraw = self.flags.shell_redraws_prompt,
 4978         .pull_scrollback = self.flags.resize_pull_scrollback,
 4979     });
 4980 
 4981     // Alternate screen, if it exists, doesn't reflow. The primary resize
 4982     // above can't be losslessly undone, so if the alternate resize fails we
 4983     // replace it with an empty screen at the requested size. If that
 4984     // also fails, we fall back to the primary screen.
 4985     if (self.screens.get(.alternate)) |alt| alt: {
 4986         const err: ResizeError = resize: {
 4987             tw.check(.alternate_screen) catch |err| break :resize err;
 4988             alt.resize(.{
 4989                 .cols = opts.cols,
 4990                 .rows = opts.rows,
 4991                 .reflow = false,
 4992                 .pull_scrollback = self.flags.resize_pull_scrollback,
 4993             }) catch |err| break :resize err;
 4994 
 4995             // Resize succeeded.
 4996             break :alt;
 4997         };
 4998 
 4999         log.warn("alternate screen resize failed, replacing it err={}", .{err});
 5000 
 5001         // If the alternate screen isn't active, then we just free it
 5002         // and move on. It'll be reallocated when it gets reinitialized lazily.
 5003         // In this case, we just lose the prior data if the terminal program
 5004         // expected it to be saved.
 5005         if (self.screens.active_key != .alternate) {
 5006             self.screens.remove(alloc, .alternate);
 5007             break :alt;
 5008         }
 5009 
 5010         // The alt screen is active, so we temporarily switch to primary
 5011         // so we can safely remove the alt and recreate it blank. This loses
 5012         // the data but hopefully keeps us on the alt screen.
 5013         const charset = alt.charset;
 5014         self.screens.switchTo(.primary);
 5015         self.screens.remove(alloc, .alternate);
 5016 
 5017         // Replace the alt screen with an empty version. If this fails
 5018         // we just go back to the primary screen. Not great, but best
 5019         // we can do.
 5020         tw.check(.alternate_screen_init) catch break :alt;
 5021         const replacement = self.screens.getInit(
 5022             self.io(),
 5023             alloc,
 5024             .alternate,
 5025             .{
 5026                 .cols = opts.cols,
 5027                 .rows = opts.rows,
 5028                 .max_scrollback_bytes = 0,
 5029                 .kitty_image_storage_limit = if (comptime build_options.kitty_graphics)
 5030                     primary.kitty_images.total_limit
 5031                 else
 5032                     0,
 5033                 .kitty_image_loading_limits = if (comptime build_options.kitty_graphics)
 5034                     primary.kitty_images.image_limits
 5035                 else {},
 5036             },
 5037         ) catch |init_err| {
 5038             log.warn(
 5039                 "alternate screen replacement failed, falling back to primary err={}",
 5040                 .{init_err},
 5041             );
 5042             break :alt;
 5043         };
 5044 
 5045         replacement.charset = charset;
 5046         self.screens.switchTo(.alternate);
 5047     }
 5048 
 5049     // No more failures are allowed after this point because the screens have
 5050     // committed their new sizes and the remaining Terminal state must follow.
 5051     errdefer comptime unreachable;
 5052 
 5053     // All fallible work is complete. Replace the old tabstop table only now.
 5054     if (new_tabstops) |v| {
 5055         self.tabstops.deinit(alloc);
 5056         self.tabstops = v;
 5057         new_tabstops = null;
 5058     }
 5059 
 5060     // Whenever we resize we just mark it as a screen clear
 5061     self.flags.dirty.clear = true;
 5062 
 5063     // Set our size
 5064     self.cols = opts.cols;
 5065     self.rows = opts.rows;
 5066 
 5067     // Reset the scrolling region
 5068     self.scrolling_region = .{
 5069         .top = 0,
 5070         .bottom = opts.rows - 1,
 5071         .left = 0,
 5072         .right = opts.cols - 1,
 5073     };
 5074 }

 5680 pub fn switchScreen(self: *Terminal, key: ScreenSet.Key) !?*Screen {
 5681     // If we're already on the requested screen we do nothing.
 5682     if (self.screens.active_key == key) return null;
 5683     const old = self.screens.active;
 5684 
 5685     // We always end hyperlink state when switching screens.
 5686     // We need to do this on the original screen.
 5687     old.endHyperlink();
 5688 
 5689     // Switch the screens/
 5690     const new = self.screens.get(key) orelse new: {
 5691         const primary = self.screens.get(.primary).?;
 5692         break :new try self.screens.getInit(
 5693             old.io,
 5694             old.alloc,
 5695             key,
 5696             .{
 5697                 .cols = self.cols,
 5698                 .rows = self.rows,
 5699                 .max_scrollback_bytes = switch (key) {
 5700                     .primary => primary.pages.limits.bytes.explicit,
 5701                     .alternate => 0,
 5702                 },
 5703 
 5704                 // Inherit our Kitty image settings from the primary
 5705                 // screen if we have to initialize.
 5706                 .kitty_image_storage_limit = if (comptime build_options.kitty_graphics)
 5707                     primary.kitty_images.total_limit
 5708                 else
 5709                     0,
 5710                 .kitty_image_loading_limits = if (comptime build_options.kitty_graphics)
 5711                     primary.kitty_images.image_limits
 5712                 else {},
 5713             },
 5714         );
 5715     };
 5716 
 5717     // The new screen should not have any hyperlinks set
 5718     assert(new.cursor.hyperlink_id == 0);
 5719 
 5720     // Bring our charset state with us
 5721     new.charset = old.charset;
 5722 
 5723     // Clear our selection
 5724     new.clearSelection();
 5725 
 5726     if (comptime build_options.kitty_graphics) {
 5727         // Mark kitty images as dirty so they redraw. Without this the
 5728         // images would stay where they were, because the screen's own
 5729         // dirty flags only cover the terminal grid.
 5730         //
 5731         // Each screen has its own counter. A consumer that last read the
 5732         // old screen will compare that value against this screen's, so
 5733         // the two must differ. Moving past both counters guarantees it.
 5734         new.kitty_images.dirty = @max(
 5735             new.kitty_images.dirty,
 5736             old.kitty_images.dirty,
 5737         ) +% 1;
 5738     }
 5739 
 5740     // Mark our terminal as dirty to redraw the grid.
 5741     self.flags.dirty.clear = true;
 5742 
 5743     // Finalize the switch
 5744     self.screens.switchTo(key);
 5745 
 5746     return old;
 5747 }
 5758 pub fn switchScreenMode(
 5759     self: *Terminal,
 5760     mode: SwitchScreenMode,
 5761     enabled: bool,
 5762 ) !void {
 5763     // The behavior in this function is completely based on reading
 5764     // the xterm source, specifically "charproc.c" for
 5765     // `srm_ALTBUF`, `srm_OPT_ALTBUF`, and `srm_OPT_ALTBUF_CURSOR`.
 5766     // We shouldn't touch anything in here without adding a unit
 5767     // test AND verifying the behavior with xterm.
 5768 
 5769     switch (mode) {
 5770         .@"47" => {},
 5771 
 5772         // If we're disabling 1047 and we're on alt screen then
 5773         // we clear the screen.
 5774         .@"1047" => if (!enabled and self.screens.active_key == .alternate) {
 5775             self.eraseDisplay(.complete, false);
 5776         },
 5777 
 5778         // 1049 unconditionally saves the cursor on enabling, even
 5779         // if we're already on the alternate screen.
 5780         .@"1049" => if (enabled) self.saveCursor(),
 5781     }
 5782 
 5783     // Switch screens first to whatever we're going to.
 5784     const to: ScreenSet.Key = if (enabled) .alternate else .primary;
 5785     const old_ = try self.switchScreen(to);
 5786 
 5787     switch (mode) {
 5788         // For these modes, we need to copy the cursor. We only copy
 5789         // the cursor if the screen actually changed, otherwise the
 5790         // cursor is already copied. The cursor is copied regardless
 5791         // of destination screen.
 5792         .@"47", .@"1047" => if (old_) |old| {
 5793             self.screens.active.cursorCopy(old.cursor, .{
 5794                 .hyperlink = false,
 5795             }) catch |err| {
 5796                 log.warn(
 5797                     "cursor copy failed entering alt screen err={}",
 5798                     .{err},
 5799                 );
 5800             };
 5801         },
 5802 
 5803         // Mode 1049 restores cursor on the primary screen when
 5804         // we disable it.
 5805         .@"1049" => if (enabled) {
 5806             assert(self.screens.active_key == .alternate);
 5807             self.eraseDisplay(.complete, false);
 5808 
 5809             // When we enter alt screen with 1049, we always copy the
 5810             // cursor from the primary screen (if we weren't already
 5811             // on it).
 5812             if (old_) |old| {
 5813                 self.screens.active.cursorCopy(old.cursor, .{
 5814                     .hyperlink = false,
 5815                 }) catch |err| {
 5816                     log.warn(
 5817                         "cursor copy failed entering alt screen err={}",
 5818                         .{err},
 5819                     );
 5820                 };
 5821             }
 5822         } else {
 5823             assert(self.screens.active_key == .primary);
 5824             self.restoreCursor();
 5825         },
 5826     }

   29 pub fn BitmapAllocator(comptime chunk_size: comptime_int) type {
   30     return struct {
   31         const Self = @This();
   32 
   33         comptime {
   34             assert(std.math.isPowerOfTwo(chunk_size));
   35         }
   36 
   37         pub const base_align: std.mem.Alignment = .fromByteUnits(@alignOf(u64));
   38         pub const bitmap_bit_size = @bitSizeOf(u64);
   39 
   40         /// The bitmap of available chunks. Each bit represents a chunk. A
   41         /// 0 means the chunk is free and a 1 means it's used, so an
   42         /// all-zero bitmap is a fully free allocator.
   43         bitmap: Offset(u64),
   44         bitmap_count: usize,
   45 
   46         /// Lowest bitmap word index that may contain a free bit; words
   47         /// below it are fully allocated, so alloc scans can start here
   48         /// instead of at zero.
   49         search_start: usize = 0,
   50 
   51         /// The contiguous buffer of chunks.
   52         chunks: Offset(u8),
   65         /// Initialize the allocator map over memory that the caller
   66         /// guarantees is already zero-filled.
   67         ///
   68         /// This writes nothing to the buffer: an all-zero bitmap already
   69         /// marks every chunk as free. Behavior is undefined if the bitmap
   70         /// region is not zero.
   71         pub fn initAssumeZeroed(buf: OffsetBuf, l: Layout) Self {
   72             assert(base_align.check(@intFromPtr(buf.start())));
   73             return .{
   74                 .bitmap = buf.member(u64, l.bitmap_start),
   75                 .bitmap_count = l.bitmap_count,
   76                 .chunks = buf.member(u8, l.chunks_start),
   77             };
   78         }
   93         pub fn alloc(
   94             self: *Self,
   95             comptime T: type,
   96             base: anytype,
   97             n: usize,
   98         ) Allocator.Error![]T {
   99             // note: we don't handle alignment yet, we just require that all
  100             // types are properly aligned. This is a limitation that should be
  101             // fixed but we haven't needed it. Contributor friendly: add tests
  102             // and fix this.
  103             assert(chunk_size % @alignOf(T) == 0);
  104             assert(n > 0);
  105 
  106             const byte_count = std.math.mul(usize, @sizeOf(T), n) catch
  107                 return error.OutOfMemory;
  108             const chunk_count = std.math.divCeil(usize, byte_count, chunk_size) catch
  109                 return error.OutOfMemory;
  110 
  111             // Find the index of the free chunk. This also marks it as used.
  112             // Words below search_start have no free bits, so no free span
  113             // can start in or cross them and it is safe to skip them.
  114             const bitmaps = self.bitmap.ptr(base);
  115             const start = @min(self.search_start, self.bitmap_count);
  116             const rel = findFreeChunks(bitmaps[start..self.bitmap_count], chunk_count) orelse
  117                 return error.OutOfMemory;
  118             const idx = start * bitmap_bit_size + rel;
  119 
  120             // Advance past any words the allocation just filled so the
  121             // next scan starts at the first word with a free bit.
  122             self.search_start = start: {
  123                 var new_start = start;
  124                 while (new_start < self.bitmap_count and
  125                     bitmaps[new_start] == std.math.maxInt(u64))
  126                 {
  127                     new_start += 1;
  128                 }
  129                 break :start new_start;
  130             };
  131 
  132             const chunks = self.chunks.ptr(base);
  133             const ptr: [*]T = @ptrCast(@alignCast(&chunks[idx * chunk_size]));
  134             return ptr[0..n];
  135         }
  136 
  137         pub fn free(self: *Self, base: anytype, slice: anytype) void {
  138             // Convert the slice of whatever type to a slice of bytes. We
  139             // can then use the byte len and chunk size to determine the
  140             // number of chunks that were allocated.
  141             const bytes = std.mem.sliceAsBytes(slice);
  142             const aligned_len = std.mem.alignForward(usize, bytes.len, chunk_size);
  143             const chunk_count = @divExact(aligned_len, chunk_size);
  144 
  145             // From the pointer, we can calculate the exact index.
  146             const chunks = self.chunks.ptr(base);
  147             const chunk_idx = @divExact(@intFromPtr(slice.ptr) - @intFromPtr(chunks), chunk_size);
  148 
  149             // The freed word gains free bits, so scans must not skip it.
  150             self.search_start = @min(
  151                 self.search_start,
  152                 @divFloor(chunk_idx, bitmap_bit_size),
  153             );
  154 
  155             const bitmaps = self.bitmap.ptr(base);
  156 
  157             // Current bitmap index.
  158             var i: usize = @divFloor(chunk_idx, 64);
  159             // Number of chunks we still have to mark as free.
  160             var rem: usize = chunk_count;
  161 
  162             // Mark any bits in the starting bitmap that need to be marked.
  163             {
  164                 // Bit index.
  165                 const bit = chunk_idx % 64;
  166                 // Number of bits we need to mark in this bitmap.
  167                 const bits = @min(rem, 64 - bit);
  168 
  169                 bitmaps[i] &= ~(~@as(u64, 0) >> @intCast(64 - bits) << @intCast(bit));
  170                 rem -= bits;
  171             }
  172 
  173             // Mark any full bitmaps worth of bits that need to be marked.
  174             i += 1;
  175             while (rem > 64) : (i += 1) {
  176                 bitmaps[i] = 0;
  177                 rem -= 64;
  178             }
  179 
  180             // Mark any bits at the start of this last bitmap if it needs it.
  181             if (rem > 0) {
  182                 bitmaps[i] &= ~(~@as(u64, 0) >> @intCast(64 - rem));
  183             }
  184         }

   71 pub fn requiredScratch(raw_len: usize) lz4.CompressError!usize {
   72     // Validate the codec's input limit before doing arithmetic based on the
   73     // raw size. The bound itself is intentionally not returned; see above.
   74     _ = try lz4.compressBound(raw_len);
   75 
   76     const representation_overhead = @sizeOf(Page) - @sizeOf(TerminalPage);
   77     if (raw_len <= representation_overhead) return 0;
   78 
   79     // The savings comparison is strict, so reserve one fewer byte than the
   80     // break-even encoded size.
   81     return raw_len - representation_overhead - 1;
   82 }
   99 pub fn init(
  100     alloc: Allocator,
  101     source: *const TerminalPage,
  102     scratch: []u8,
  103     table: *lz4.HashTable,
  104 ) (Allocator.Error || lz4.CompressError)!?Page {
  105     const required = try requiredScratch(source.memory.len);
  106     if (scratch.len < required) return error.OutputTooSmall;
  107     if (required == 0) return null;
  108 
  109     // Page memory is an opaque snapshot containing structs whose padding has
  110     // no semantic value. Assigning those structs can leave their padding
  111     // undefined in Memcheck even though every byte is addressable and must be
  112     // preserved by compression. Mark the complete byte representation as
  113     // defined at this boundary so that undefined padding does not propagate
  114     // through the codec and obscure real Valgrind failures.
  115     if (std.valgrind.runningOnValgrind() > 0)
  116         std.valgrind.memcheck.makeMemDefined(source.memory);
  117 
  118     const encoded_len = lz4.compress(
  119         source.memory,
  120         scratch[0..required],
  121         table,
  122     ) catch |err| switch (err) {
  123         // The scratch limit is the largest representation worth retaining.
  124         // Running out of room therefore means compression cannot save memory,
  125         // rather than that the caller failed to provide the documented size.
  126         error.OutputTooSmall => return null,
  127         error.InputTooLarge => return err,
  128     };
  129 
  130     // requiredScratch already accounts for the representation structs. The
  131     // retained virtual range contributes no resident bytes after PageList
  132     // decommits it.
  133     assert(@sizeOf(Page) + encoded_len <
  134         @sizeOf(TerminalPage) + source.memory.len);
  135 
  136     return .{
  137         .page = source.*,
  138         .encoded = try alloc.dupe(u8, scratch[0..encoded_len]),
  139         .alloc = alloc,
  140     };
  141 }

  152 /// Restore the embedded terminal page into its retained resident mapping.
  153 ///
  154 /// The caller must recommit `page.memory` before calling this on platforms
  155 /// which require an explicit recommit operation. The compressed value remains
  156 /// valid whether decoding succeeds or fails, allowing the caller to retry or
  157 /// discard it. On success the returned page aliases the same resident mapping;

  159 pub fn restore(self: *const Page) lz4.DecompressError!TerminalPage {
  160     const result = self.page;
  161     _ = try lz4.decompress(self.encoded, result.memory);
  162     return result;

   76     /// A change counter for anything that can affect how images are
   77     /// drawn. It changes when images or placements are added, removed,
   78     /// or modified, and also when placements move on screen without
   79     /// changing, such as during scrolling, resizing, or erasing history.
   80     ///
   81     /// The counter is never reset. Instead, each consumer keeps the last
   82     /// value it saw and compares it to the current value. Because nothing
   83     /// clears the counter, any number of consumers can watch the same
   84     /// storage independently. Within Ghostty, only `RenderState` reads

    5 /// Stack for the key flags. This implements the push/pop behavior
    6 /// of the CSI > u and CSI < u sequences. We implement the stack as
    7 /// fixed size to avoid heap allocation.
    8 pub const FlagStack = struct {
    9     const len = 8;
   10 
   11     flags: [len]Flags = @splat(.disabled),
   12     idx: u3 = 0,
   13 
   14     /// Return the current stack value
   15     pub fn current(self: FlagStack) Flags {
   16         return self.flags[self.idx];
   17     }
   18 
   19     /// Perform the "set" operation as described in the spec for
   20     /// the CSI = u sequence.
   21     pub fn set(
   22         self: *FlagStack,
   23         mode: SetMode,
   24         v: Flags,
   25     ) void {
   26         switch (mode) {
   27             .set => self.flags[self.idx] = v,
   28             .@"or" => self.flags[self.idx] = @bitCast(
   29                 self.flags[self.idx].int() | v.int(),
   30             ),
   31             .not => self.flags[self.idx] = @bitCast(
   32                 self.flags[self.idx].int() & ~v.int(),
   33             ),
   34         }
   35     }
   36 
   37     /// Push a new set of flags onto the stack. If the stack is full
   38     /// then the oldest entry is evicted.
   39     pub fn push(self: *FlagStack, flags: Flags) void {
   40         // Overflow and wrap around if we're full, which evicts
   41         // the oldest entry.
   42         self.idx +%= 1;
   43         self.flags[self.idx] = flags;
   44     }
   45 
   46     /// Pop `n` entries from the stack. This will just wrap around
   47     /// if `n` is greater than the amount in the stack.
   48     pub fn pop(self: *FlagStack, n: usize) void {
   49         // If n is more than our length then we just reset the stack.
   50         // This also avoids a DoS vector where a malicious client
   51         // could send a huge number of pop commands to waste cpu.
   52         if (n >= self.flags.len) {
   53             self.idx = 0;
   54             self.flags = @splat(.disabled);
   55             return;
   56         }
   57 
   58         for (0..n) |_| {
   59             self.flags[self.idx] = .disabled;
   60             self.idx -%= 1;
   61         }
   62     }

  119 /// Release the physical memory behind `memory` while keeping its address
  120 /// range.
  121 ///
  122 /// `memory` must start on a page boundary and be a whole number of pages
  123 /// long, using the smallest page size the target supports. `dirty_len` is
  124 /// the length of the prefix that may hold nonzero bytes. Everything after it
  125 /// must already be zero. Strict mode requires `dirty_len == memory.len`,
  126 /// because a successful decommit throws away everything in the range.
  127 ///
  128 /// Returns true when the operating system released the memory. What false
  129 /// means depends on `mode`:
  130 ///
  131 /// - `.strict`: nothing was released and the contents are unchanged, so the
  132 ///   caller can keep using the memory as it was.
  133 /// - `.zero`: the memory may still be in use, but the dirty prefix has been
  134 ///   cleared. The whole range reads as zeros either way.
  135 ///
  136 /// Zero mode suits memory that is about to be reused and only needs to be
  137 /// clean. Only the bytes that were actually written have to be cleared:
  138 ///
  139 /// ```zig
  140 /// // Only the first `used` bytes of `page` were ever written.
  141 /// _ = mem.decommit(.zero, page, used);
  142 /// // `page` now reads as all zeros.
  143 /// ```
  144 ///
  145 /// In test builds this never calls into the operating system. It clears the
  146 /// dirty prefix and returns true, which imitates a successful release.
  147 pub fn decommit(
  148     comptime mode: DecommitMode,
  149     memory: []align(std.heap.page_size_min) u8,
  150     dirty_len: usize,
  151 ) bool {
  152     assert(memory.len > 0);
  153     assert(@intFromPtr(memory.ptr) % std.heap.page_size_min == 0);
  154     assert(memory.len % std.heap.page_size_min == 0);
  155     assert(dirty_len <= memory.len);
  156     if (comptime mode == .strict) assert(dirty_len == memory.len);
  157 
  158     // Testing allocator ranges may share an OS mapping with unrelated memory,
  159     // so madvise is not safe. Zeroing models the only content guarantee callers
  160     // have after a successful discard.
  161     if (comptime builtin.is_test) {
  162         @memset(memory[0..dirty_len], 0);
  163         return true;
  164     }
  165 
  166     // DONTNEED immediately reclaims private anonymous pages on Linux and
  167     // faults them back as zero-filled pages. We deliberately avoid MADV_FREE:
  168     // it does not reduce RSS until memory pressure and does not guarantee that
  169     // the next read is zero.
  170     if (comptime builtin.target.os.tag == .linux) {
  171         if (std.posix.madvise(
  172             memory.ptr,
  173             memory.len,
  174             std.posix.MADV.DONTNEED,
  175         )) |_| return true else |err| {
  176             log.warn("madvise(DONTNEED) failed err={}", .{err});
  177             if (comptime mode == .strict) return false;
  178             // Zero mode falls through to the memset below.
  179         }
  180     }
  181 
  182     // FREE_REUSABLE removes the range from the Darwin process footprint while
  183     // retaining its mapping. Zero mode clears its dirty prefix first because
  184     // the kernel may preserve the contents. Strict mode avoids that write
  185     // because its caller will replace the entire mapping after recommit.
  186     if (comptime builtin.target.os.tag.isDarwin()) {
  187         if (comptime mode == .zero) @memset(memory[0..dirty_len], 0);
  188 
  189         if (std.posix.madvise(
  190             memory.ptr,
  191             memory.len,
  192             std.posix.MADV.FREE_REUSABLE,
  193         )) |_| return true else |err| {
  194             switch (mode) {
  195                 .strict => {
  196                     log.warn("madvise(FREE_REUSABLE) failed err={}", .{err});
  197                     return false;
  198                 },
  199 
  200                 .zero => {
  201                     // Plain FREE can still reclaim the already-zero mapping
  202                     // under pressure and does not require a reuse pairing.
  203                     std.posix.madvise(
  204                         memory.ptr,
  205                         memory.len,
  206                         std.posix.MADV.FREE,
  207                     ) catch {};
  208                     return false;
  209                 },
  210             }
  211         }
  212     }
  213 
  214     // DiscardVirtualMemory releases the physical pages behind the range but
  215     // leaves it committed, so the commit charge stays with the process and a
  216     // later access finds a zero page or the old contents instead of faulting.
  217     // Zero mode clears its dirty prefix first, as on Darwin: the bytes read
  218     // as zero afterward whether or not the discard took. Strict mode skips
  219     // that write because its caller replaces the entire mapping after
  220     // recommit. The call reports failure through its return value rather
  221     // than the thread's last error.
  222     if (comptime builtin.target.os.tag == .windows) {
  223         if (comptime mode == .zero) @memset(memory[0..dirty_len], 0);
  224 
  225         const rc = windows.exp.kernel32.DiscardVirtualMemory(
  226             memory.ptr,
  227             memory.len,
  228         );
  229         if (rc == windows.ERROR_SUCCESS) return true;
  230 
  231         // Zero mode has already cleared its bytes and strict callers must
  232         // leave the still-resident mapping alone, so there is nothing more
  233         // to do for either mode.
  234         log.warn("DiscardVirtualMemory failed err={d}", .{rc});
  235         return false;
  236     }
  237 
  238     // FreeBSD's madvise cannot release pages right away, so map fresh memory
  239     // over the range instead. See decommitFreeBSD for the details.
  240     if (comptime builtin.target.os.tag == .freebsd) {
  241         if (decommitFreeBSD(memory)) return true;
  242         if (comptime mode == .strict) return false;
  243         // Zero mode falls through to the memset below.
  244     }
  245 
  246     if (comptime mode == .zero) @memset(memory[0..dirty_len], 0);
  247     return false;
  248 }
  278 fn decommitFreeBSD(memory: []align(std.heap.page_size_min) u8) bool {
  279     if (mapFixedFreeBSD(memory)) |err| {
  280         log.warn("mmap(MAP_FIXED) failed err={}", .{err});
  281     } else return true;
  282 
  283     // Find out whether the kernel removed the old mapping before it failed.
  284     // It removes the whole range or none of it, so checking the first page
  285     // is enough. mincore fails with ENOMEM when the address is not mapped.
  286     var vec: [1]u8 = undefined;
  287     if (std.posix.mincore(memory.ptr, std.heap.page_size_min, &vec)) |_| {
  288         return false;
  289     } else |err| switch (err) {
  290         error.OutOfMemory => {},
  291         else => return false,
  292     }
  293 
  294     // The old contents are already gone, so if this second mapping works,
  295     // the result is the same as a successful decommit. The old mapping's
  296     // swap reservation has been returned by now, so this attempt needs no
  297     // more swap than the process already had.
  298     if (mapFixedFreeBSD(memory)) |err| {
  299         // Nothing is mapped at this address, but the caller still owns the
  300         // range and will use it again. Stop here with a clear message
  301         // instead of crashing later on an unexplained memory fault.
  302         std.debug.panic(
  303             "mmap(MAP_FIXED) could not restore a discarded page err={}",
  304             .{err},
  305         );
  306     }
  307     return true;
  308 }

  310 /// Map fresh, zero-filled, private memory over `memory` at the same address.
  311 /// Returns null on success or the error code on failure.
  312 ///
  313 /// This calls libc's `mmap` directly instead of `std.posix.mmap`. When the
  314 /// kernel removes the old mapping but cannot add the new one, FreeBSD
  315 /// reports `EINVAL`. `std.posix.mmap` treats `EINVAL` as impossible
  316 /// (`unreachable`), which would crash a safe build and is undefined
  317 /// behavior in a release build.
  318 fn mapFixedFreeBSD(memory: []align(std.heap.page_size_min) u8) ?std.c.E {
  319     const rc = std.c.mmap(
  320         memory.ptr,
  321         memory.len,
  322         .{ .READ = true, .WRITE = true },
  323         .{ .TYPE = .PRIVATE, .ANONYMOUS = true, .FIXED = true },
  324         -1,
  325         0,
  326     );
  327     if (rc == std.c.MAP_FAILED) return std.c.errno(@as(c_int, -1));
  328     assert(@intFromPtr(rc) == @intFromPtr(memory.ptr));
  329     return null;
  330 }

  332 /// Prepare memory released by `decommit` for reuse.
  333 ///
  334 /// Call this before writing new contents into a range that was decommitted.
  335 /// Only Darwin does any work here. It tells the kernel the pages are in use
  336 /// again (`MADV_FREE_REUSE`), so they count toward the process's memory
  337 /// footprint once more. On every other platform, the range is already ready
  338 /// to use when `decommit` returns, so this does nothing.
  339 ///
  340 /// This cannot fail in a way the caller needs to handle. If the Darwin call
  341 /// fails, only the memory accounting is off. The range is still usable, so
  342 /// the failure is logged and ignored.
  343 pub fn recommit(memory: []align(std.heap.page_size_min) u8) void {
  344     assert(memory.len > 0);
  345     assert(@intFromPtr(memory.ptr) % std.heap.page_size_min == 0);
  346     assert(memory.len % std.heap.page_size_min == 0);
  347 
  348     if (comptime builtin.is_test) return;
  349     if (comptime builtin.target.os.tag.isDarwin()) {
  350         std.posix.madvise(
  351             memory.ptr,
  352             memory.len,
  353             std.posix.MADV.FREE_REUSE,
  354         ) catch |err| {
  355             log.warn("madvise(FREE_REUSE) failed err={}", .{err});
  356         };

  147 /// A page represents a specific section of terminal screen. The primary
  148 /// idea of a page is that it is a fully self-contained unit that can be
  149 /// serialized, copied, etc. as a convenient way to represent a section
  150 /// of the screen.
  151 ///
  152 /// This property is useful for renderers which want to copy just the pages
  153 /// for the visible portion of the screen, or for infinite scrollback where
  154 /// we may want to serialize and store pages that are sufficiently far
  155 /// away from the current viewport.
  156 ///
  157 /// Pages are always backed by a single contiguous block of memory that is
  158 /// aligned on a page boundary. This makes it easy and fast to copy pages
  159 /// around. Within the contiguous block of memory, the contents of a page are
  160 /// thoughtfully laid out to optimize primarily for terminal IO (VT streams)
  161 /// and to minimize memory usage.

  165         // alignment is always divisible by this.
  166         assert(std.heap.page_size_min % @max(
  167             @alignOf(Row),
  168             cells_align,
  169             MetaLayout.alignment,
  170         ) == 0);
  171     }

  245     pub inline fn init(cap: Capacity) !Page {
  246         const l = layout(cap);
  247 
  248         // We allocate page-aligned zeroed memory directly to avoid Zig
  249         // allocator overhead (small but meaningful for this path). Both
  250         // mmap (POSIX) and VirtualAlloc (Windows) guarantee zeroed pages,
  251         // which is a critical property for us.
  252         assert(l.total_size % std.heap.page_size_min == 0);
  253         const backing = try PageAlloc.alloc(l.total_size);
  254         errdefer PageAlloc.free(backing);
  255 
  256         const buf = OffsetBuf.init(backing);
  257         return initBuf(buf, l);
  258     }
  260     /// Initialize a new page using the given backing memory.
  261     /// It is up to the caller to not call deinit on these pages.
  262     ///
  263     /// The backing memory must be zero-filled. A page treats zero as the
  264     /// empty state everywhere: cells are blank when zero, and every
  265     /// metadata member initializes from zeroed memory without writing
  266     /// anything. Only the row headers are written, so the OS pages behind
  267     /// everything else stay untouched until first use.
  268     pub inline fn initBuf(buf: OffsetBuf, l: Layout) Page {
  269         const cap = l.capacity;
  270 
  271         // A page must always have at least one row.
  272         assert(cap.rows > 0);
  273 
  274         const rows = buf.member(Row, l.rows_start);
  275         const cells = buf.member(Cell, l.cells_start);
  276 
  277         // We need to go through and initialize all the rows so that
  278         // they point to a valid offset into the cells, since the rows
  279         // zero-initialized aren't valid.
  280         const cells_len = @as(usize, cap.cols) * @as(usize, cap.rows);
  281         const cells_ptr = cells.ptr(buf)[0..cells_len];
  282         for (rows.ptr(buf)[0..cap.rows], 0..) |*row, y| {
  283             const start = y * cap.cols;
  284             row.* = .{
  285                 .cells = getOffset(Cell, buf, &cells_ptr[start]),
  286             };
  287         }
  288 
  289         return .{
  290             .memory = @alignCast(buf.start()[0..l.total_size]),
  291             .rows = rows,
  292             .cells = cells,
  293             .styles = StyleSet.initAssumeZeroed(
  294                 buf.add(l.styles_start),
  295                 l.styles_layout,
  296                 .{},
  297             ),
  298             .string_alloc = .initAssumeZeroed(
  299                 buf.add(l.string_alloc_start),
  300                 l.string_alloc_layout,
  301             ),
  302             .grapheme_alloc = .initAssumeZeroed(
  303                 buf.add(l.grapheme_alloc_start),
  304                 l.grapheme_alloc_layout,
  305             ),
  306             .grapheme_map = .initAssumeZeroed(
  307                 buf.add(l.grapheme_map_start),
  308                 l.grapheme_map_layout,
  309             ),
  310             .hyperlink_map = .initAssumeZeroed(
  311                 buf.add(l.hyperlink_map_start),
  312                 l.hyperlink_map_layout,
  313             ),
  314             .hyperlink_set = .initAssumeZeroed(
  315                 buf.add(l.hyperlink_set_start),
  316                 l.hyperlink_set_layout,
  317                 .{},
  318             ),
  319             .size = .{ .cols = cap.cols, .rows = cap.rows },
  320             .capacity = cap,
  321             .dirty = false,
  322         };
  323     }
  325     /// Deinitialize the page, freeing any backing memory. Do NOT call
  326     /// this if you allocated the backing memory yourself (i.e. you used
  327     /// initBuf).
  328     pub inline fn deinit(self: *Page) void {
  329         PageAlloc.free(self.memory);
  330         self.* = undefined;
  331     }
  693     pub inline fn cloneBuf(self: *const Page, buf: []align(std.heap.page_size_min) u8) Page {
  694         assert(buf.len >= self.memory.len);
  695 
  696         // The entire concept behind a page is that everything is stored
  697         // as offsets so we can do a simple linear copy of the backing
  698         // memory and copy all the offsets and everything will work.
  699         var result = self.*;
  700         result.memory = buf[0..self.memory.len];
  701 
  702         // This is a memcpy. We may want to investigate if there are
  703         // faster ways to do this (i.e. copy-on-write tricks) but I suspect
  704         // they'll be slower. I haven't experimented though.
  705         // std.log.warn("copy bytes={}", .{self.memory.len});
  706         fastmem.copy(u8, result.memory, self.memory);
  707 
  708         return result;
  709     }

 1410     pub fn insertHyperlink(
 1411         self: *Page,
 1412         link: hyperlink.Hyperlink,
 1413     ) InsertHyperlinkError!hyperlink.Id {
 1414         // Insert our URI into the page strings table.
 1415         const page_uri: Offset(u8).Slice = uri: {
 1416             const buf = self.string_alloc.alloc(
 1417                 u8,
 1418                 self.memory,
 1419                 link.uri.len,
 1420             ) catch |err| switch (err) {
 1421                 error.OutOfMemory => return error.StringsOutOfMemory,
 1422             };
 1423             errdefer self.string_alloc.free(self.memory, buf);
 1424             @memcpy(buf, link.uri);
 1425 
 1426             break :uri .{
 1427                 .offset = size.getOffset(u8, self.memory, &buf[0]),
 1428                 .len = link.uri.len,
 1429             };
 1430         };
 1431         errdefer self.string_alloc.free(
 1432             self.memory,
 1433             page_uri.slice(self.memory),
 1434         );
 1435 
 1436         // Allocate an ID for our page memory if we have to.
 1437         const page_id: hyperlink.PageEntry.Id = switch (link.id) {
 1438             .explicit => |id| explicit: {
 1439                 const buf = self.string_alloc.alloc(
 1440                     u8,
 1441                     self.memory,
 1442                     id.len,
 1443                 ) catch |err| switch (err) {
 1444                     error.OutOfMemory => return error.StringsOutOfMemory,
 1445                 };
 1446                 errdefer self.string_alloc.free(self.memory, buf);
 1447                 @memcpy(buf, id);
 1448 
 1449                 break :explicit .{
 1450                     .explicit = .{
 1451                         .offset = size.getOffset(u8, self.memory, &buf[0]),
 1452                         .len = id.len,
 1453                     },
 1454                 };
 1455             },
 1456 
 1457             .implicit => |id| .{ .implicit = id },
 1458         };
 1459         errdefer switch (page_id) {
 1460             .implicit => {},
 1461             .explicit => |slice| self.string_alloc.free(
 1462                 self.memory,
 1463                 slice.slice(self.memory),
 1464             ),
 1465         };
 1466 
 1467         // Build our entry
 1468         const entry: hyperlink.PageEntry = .{
 1469             .id = page_id,
 1470             .uri = page_uri,
 1471         };
 1472 
 1473         // Put our hyperlink into the hyperlink set to get an ID
 1474         const id = self.hyperlink_set.addContext(
 1475             self.memory,
 1476             entry,
 1477             .{ .page = self },
 1478         ) catch |err| switch (err) {
 1479             error.OutOfMemory => return error.SetOutOfMemory,
 1480             error.NeedsRehash => return error.SetNeedsRehash,
 1481         };
 1482         errdefer self.hyperlink_set.release(self.memory, id);

 1487     /// Set the hyperlink for the given cell. If the cell already has a
 1488     /// hyperlink, then this will handle memory management and refcount
 1489     /// update for the prior hyperlink.
 1490     ///
 1491     /// DOES NOT increment the reference count for the new hyperlink!
 1492     ///
 1493     /// Caller is responsible for updating the refcount in the hyperlink
 1494     /// set as necessary by calling `use` if the id was not acquired with
 1495     /// `add`.
 1496     pub inline fn setHyperlink(self: *Page, row: *Row, cell: *Cell, id: hyperlink.Id) error{HyperlinkMapOutOfMemory}!void {
 1497         defer self.assertIntegrity();
 1498 
 1499         const cell_offset = getOffset(Cell, self.memory, cell);
 1500         var map = self.hyperlink_map.map(self.memory);
 1501         const gop = map.getOrPut(cell_offset) catch |e| {
 1502             comptime assert(@TypeOf(e) == error{OutOfMemory});
 1503             // The hyperlink map capacity needs to be increased.
 1504             return error.HyperlinkMapOutOfMemory;
 1505         };
 1506 
 1507         if (gop.found_existing) {
 1508             // Always release the old hyperlink, because even if it's actually
 1509             // the same as the one we're setting, we'd end up double-counting
 1510             // if we left the reference count be, because the caller does not
 1511             // know whether it's the same and will have increased the count
 1512             // outside of this function.
 1513             self.hyperlink_set.release(self.memory, gop.value_ptr.*);
 1514 
 1515             // If the hyperlink matches then we don't need to do anything.
 1516             if (gop.value_ptr.* == id) {
 1517                 // It is possible for cell hyperlink to be false but row
 1518                 // must never be false. The cell hyperlink can be false because
 1519                 // in Terminal.print we clear the hyperlink for the cursor cell
 1520                 // before writing the cell again, so if someone prints over
 1521                 // a cell with a matching hyperlink this state can happen.
 1522                 // This is tested in Terminal.zig.
 1523                 assert(row.hyperlink);
 1524                 cell.hyperlink = true;
 1525                 return;
 1526             }
 1527         }
 1528 
 1529         // Set the hyperlink on the cell and in the map.
 1530         gop.value_ptr.* = id;
 1531         cell.hyperlink = true;
 1532         row.hyperlink = true;
 1533     }

 1642     /// Append a codepoint to the given cell as a grapheme. Once the cell has
 1643     /// grapheme_max_len suffix codepoints, additional codepoints are ignored.
 1644     pub fn appendGrapheme(self: *Page, row: *Row, cell: *Cell, cp: u21) Allocator.Error!void {
 1645         defer self.assertIntegrity();
 1646 
 1647         if (build_options.slow_runtime_safety) assert(cell.codepoint() != 0);
 1648 
 1649         const cell_offset = getOffset(Cell, self.memory, cell);
 1650         var map = self.grapheme_map.map(self.memory);
 1651 
 1652         // If this cell has no graphemes, we can go faster by knowing we
 1653         // need to allocate a new grapheme slice and update the map.
 1654         if (cell.content_tag != .codepoint_grapheme) {
 1655             const cps = try self.grapheme_alloc.alloc(u21, self.memory, 1);
 1656             errdefer self.grapheme_alloc.free(self.memory, cps);
 1657             cps[0] = cp;
 1658 
 1659             try map.putNoClobber(cell_offset, .{
 1660                 .offset = getOffset(u21, self.memory, @ptrCast(cps.ptr)),
 1661                 .len = 1,
 1662             });
 1663             errdefer map.remove(cell_offset);
 1664 
 1665             cell.content_tag = .codepoint_grapheme;
 1666             row.grapheme = true;
 1667 
 1668             return;
 1669         }
 1670 
 1671         // The cell already has graphemes. We need to append to the existing
 1672         // grapheme slice and update the map.
 1673         assert(row.grapheme);
 1674 
 1675         const slice = map.getPtr(cell_offset).?;
 1676 
 1677         // Terminal input is untrusted. In addition to bounding memory, this
 1678         // prevents repeated chunk growth and copying from becoming quadratic.
 1679         if (slice.len >= grapheme_max_len) return;
 1680 
 1681         // If our slice len doesn't divide evenly by the grapheme chunk
 1682         // length then we can utilize the additional chunk space.
 1683         if (slice.len % grapheme_chunk_len != 0) {
 1684             const cps = slice.offset.ptr(self.memory);
 1685             cps[slice.len] = cp;
 1686             slice.len += 1;
 1687             return;
 1688         }
 1689 
 1690         // We are out of chunk space. There is no fast path here. We need
 1691         // to allocate a larger chunk. This is a very slow path. We expect
 1692         // most graphemes to fit within our chunk size.
 1693         const cps = try self.grapheme_alloc.alloc(u21, self.memory, slice.len + 1);
 1694         errdefer self.grapheme_alloc.free(self.memory, cps);
 1695         const old_cps = slice.slice(self.memory);
 1696         fastmem.copy(u21, cps[0..old_cps.len], old_cps);
 1697         cps[slice.len] = cp;
 1698         slice.* = .{
 1699             .offset = getOffset(u21, self.memory, @ptrCast(cps.ptr)),
 1700             .len = slice.len + 1,
 1701         };
 1702 
 1703         // Free our old chunk
 1704         self.grapheme_alloc.free(self.memory, old_cps);
 1705     }

 1707     /// Returns the codepoints for the given cell. These are the codepoints
 1708     /// in addition to the first codepoint. The first codepoint is NOT
 1709     /// included since it is on the cell itself.
 1710     pub inline fn lookupGrapheme(self: *const Page, cell: *const Cell) ?[]u21 {
 1711         const cell_offset = getOffset(Cell, self.memory, cell);
 1712         const map = self.grapheme_map.map(self.memory);
 1713         const slice = map.get(cell_offset) orelse return null;
 1714         return slice.slice(self.memory);
 1715     }

 1717     /// Move the graphemes from one cell to another. This can't fail
 1718     /// because we avoid any allocations since we're just moving data.
 1719     ///
 1720     /// WARNING: This will NOT change the content_tag on the cells because
 1721     /// there are scenarios where we want to move graphemes without changing
 1722     /// the content tag. Callers beware but assertIntegrity should catch this.
 1723     pub inline fn moveGrapheme(self: *Page, src: *Cell, dst: *Cell) void {
 1724         if (build_options.slow_runtime_safety) {
 1725             assert(src.hasGrapheme());
 1726             assert(!dst.hasGrapheme());
 1727         }
 1728 
 1729         const src_offset = getOffset(Cell, self.memory, src);
 1730         const dst_offset = getOffset(Cell, self.memory, dst);
 1731         var map = self.grapheme_map.map(self.memory);
 1732         const entry = map.getEntry(src_offset).?;
 1733         const value = entry.value_ptr.*;
 1734         map.removeByPtr(entry.key_ptr);
 1735         map.putAssumeCapacityNoClobber(dst_offset, value);
 1736     }

 1738     /// Clear the graphemes for a given cell.
 1739     ///
 1740     /// In order to update the grapheme flag on the row, call
 1741     /// `updateRowGraphemeFlag` after you finish clearing any
 1742     /// graphemes in the row.
 1743     pub fn clearGrapheme(self: *Page, cell: *Cell) void {
 1744         defer self.assertIntegrity();
 1745         if (build_options.slow_runtime_safety) assert(cell.hasGrapheme());
 1746 
 1747         // Get our entry in the map, which must exist
 1748         const cell_offset = getOffset(Cell, self.memory, cell);
 1749         var map = self.grapheme_map.map(self.memory);
 1750         const entry = map.getEntry(cell_offset).?;

 1752         // Free our grapheme data
 1753         const cps = entry.value_ptr.slice(self.memory);
 1754         self.grapheme_alloc.free(self.memory, cps);
 1755 
 1756         // Remove the entry
 1757         map.removeByPtr(entry.key_ptr);
 1758

 1760         cell.content_tag = .codepoint;

 1894     /// The memory layout for a page given a desired minimum cols
 1895     /// and rows size.
 1896     ///
 1897     /// The backing memory is laid out as the row headers, the cell
 1898     /// array, and then the metadata block (see `MetaLayout`):
 1899     ///
 1900     ///   [rows][cells][styles, graphemes, strings, hyperlinks]
 1901     ///
 1902     /// Row headers always start at offset zero. The cell array is aligned
 1903     /// to a cache line (see `cells_align`). Initializing a page writes only
 1904     /// the row headers: the cells and every metadata member treat zero as
 1905     /// their empty state, so the OS pages behind everything past the row
 1906     /// headers stay untouched until something is stored in them.
 1907     pub inline fn layout(cap: Capacity) Layout {
 1908         const meta: MetaLayout = .init(cap);
 1909 
 1910         const rows_count: usize = @intCast(cap.rows);
 1911         const rows_start = 0;
 1912         const rows_end: usize = rows_start + (rows_count * @sizeOf(Row));
 1913 
 1914         const cells_count: usize = @as(usize, cap.cols) * @as(usize, cap.rows);
 1915         const cells_start = alignForward(usize, rows_end, cells_align);
 1916         const cells_end = cells_start + (cells_count * @sizeOf(Cell));
 1917 
 1918         const meta_start = alignForward(usize, cells_end, MetaLayout.alignment);
 1919         const meta_end = meta_start + meta.total_size;
 1920 
 1921         const total_size = alignForward(usize, meta_end, std.heap.page_size_min);
 1922 
 1923         return .{
 1924             .total_size = total_size,
 1925             .rows_start = rows_start,
 1926             .rows_size = rows_end - rows_start,
 1927             .cells_start = cells_start,
 1928             .cells_size = cells_end - cells_start,
 1929             .styles_start = meta_start + meta.styles_start,
 1930             .styles_layout = meta.styles_layout,
 1931             .grapheme_alloc_start = meta_start + meta.grapheme_alloc_start,
 1932             .grapheme_alloc_layout = meta.grapheme_alloc_layout,
 1933             .grapheme_map_start = meta_start + meta.grapheme_map_start,
 1934             .grapheme_map_layout = meta.grapheme_map_layout,
 1935             .string_alloc_start = meta_start + meta.string_alloc_start,
 1936             .string_alloc_layout = meta.string_alloc_layout,
 1937             .hyperlink_map_start = meta_start + meta.hyperlink_map_start,

 1939             .hyperlink_set_start = meta_start + meta.hyperlink_set_start,
 1940             .hyperlink_set_layout = meta.hyperlink_set_layout,
 1941             .capacity = cap,
 1942         };

 1975         /// Compute the layout of the block for the given capacity.
 1976         pub fn init(cap: Capacity) MetaLayout {
 1977             const styles_layout: StyleSet.Layout = .init(cap.styles);
 1978             const styles_start = 0;
 1979             const styles_end = styles_start + styles_layout.total_size;
 1980 
 1981             const grapheme_alloc_layout = GraphemeAlloc.layout(cap.grapheme_bytes);
 1982             const grapheme_alloc_start = alignForward(usize, styles_end, GraphemeAlloc.base_align.toByteUnits());
 1983             const grapheme_alloc_end = grapheme_alloc_start + grapheme_alloc_layout.total_size;
 1984 
 1985             const grapheme_count: usize = count: {
 1986                 if (cap.grapheme_bytes == 0) break :count 0;
 1987                 // Use divCeil to match GraphemeAlloc.layout() which uses alignForward,
 1988                 // ensuring grapheme_map has capacity when grapheme_alloc has chunks.
 1989                 const base = std.math.divCeil(usize, cap.grapheme_bytes, grapheme_chunk) catch unreachable;
 1990                 break :count std.math.ceilPowerOfTwo(usize, base) catch unreachable;
 1991             };
 1992             const grapheme_map_layout = GraphemeMap.layout(@intCast(grapheme_count));
 1993             const grapheme_map_start = alignForward(usize, grapheme_alloc_end, GraphemeMap.base_align.toByteUnits());

 2009 
 2010             const hyperlink_map_layout = hyperlink.Map.layout(cap.hyperlink_cells);
 2011             const hyperlink_map_start = alignForward(usize, hyperlink_set_end, hyperlink.Map.base_align.toByteUnits());
 2012             const hyperlink_map_end = hyperlink_map_start + hyperlink_map_layout.total_size;
 2013 
 2014             return .{
 2015                 .total_size = hyperlink_map_end,
 2016                 .styles_start = styles_start,
 2017                 .styles_layout = styles_layout,
 2018                 .grapheme_alloc_start = grapheme_alloc_start,
 2019                 .grapheme_alloc_layout = grapheme_alloc_layout,
 2020                 .grapheme_map_start = grapheme_map_start,
 2021                 .grapheme_map_layout = grapheme_map_layout,
 2022                 .string_alloc_start = string_start,
 2023                 .string_alloc_layout = string_layout,
 2024                 .hyperlink_set_start = hyperlink_set_start,
 2025                 .hyperlink_set_layout = hyperlink_set_layout,
 2026                 .hyperlink_map_start = hyperlink_map_start,
 2027                 .hyperlink_map_layout = hyperlink_map_layout,

  302         pub fn addContext(self: *Self, base: anytype, value: T, ctx: Context) AddError!Id {
  303             const items = self.items.ptr(base);
  304 
  305             // Trim dead items from the end of the list.
  306             while (self.next_id > 1 and items[self.next_id - 1].meta.ref == 0) {
  307                 self.next_id -= 1;
  308                 self.deleteItem(base, self.next_id, ctx);
  309             }
  310 
  311             // If the item already exists, return it.
  312             if (self.lookupContext(base, value, ctx)) |id| {
  313                 // Notify the context that the value is "deleted" because
  314                 // we're reusing the existing value in the set. This allows
  315                 // callers to clean up any resources associated with the value.
  316                 if (comptime @hasDecl(Context, "deleted")) ctx.deleted(value);
  317 
  318                 items[id].meta.ref += 1;
  319                 return id;
  320             }
  321 
  322             // While it should be statistically impossible to exceed the
  323             // bounds of `psl_stats`, the hash function is not perfect and
  324             // in such a case we want to remain stable. If we're about to
  325             // insert an item and there's something with a PSL of `len - 1`,
  326             // we may end up with a PSL of `len` which would exceed the bounds.
  327             // In such a case, we claim to be out of memory.
  328             if (self.psl_stats[self.psl_stats.len - 1] > 0) {
  329                 @branchHint(.cold);
  330                 return AddError.OutOfMemory;
  331             }
  332 
  333             // If the item doesn't exist, we need an available ID.
  334             if (self.next_id >= self.layout.cap) {
  335                 // Every ID is taken. Reuse a dead one if at least 10% are
  336                 // dead. Otherwise ask the caller to grow, since we would
  337                 // run out again almost immediately. 90% is arbitrary.
  338                 const reuse_threshold = 0.9;
  339                 if (self.living < @as(Id, @intFromFloat(@as(f64, @floatFromInt(self.layout.cap)) * reuse_threshold))) {
  340                     // The trim at the top leaves the last ID live, so a
  341                     // dead ID must exist below it.
  342                     const dead_id = self.reclaimDeadId(base, ctx) orelse {
  343                         @branchHint(.cold);
  344 
  345                         // Only possible if `living` is wrong. Fail loudly
  346                         // in safe builds. Release builds recover by
  347                         // asking the caller to rehash.
  348                         if (comptime std.debug.runtime_safety) assert(false);
  349                         return AddError.NeedsRehash;
  350                     };
  351 
  352                     const id = self.insert(base, value, dead_id, ctx);
  353                     items[id].meta.ref += 1;
  354                     assert(items[id].meta.ref == 1);
  355                     self.living += 1;
  356 
  357                     // `insert` may reuse a smaller dead ID it reaped on its
  358                     // way, leaving `dead_id` dead and out of the table.
  359                     // Reclaim it next.
  360                     if (id != dead_id) self.free_cursor = dead_id;
  361 
  362                     return id;
  363                 }
  364 
  365                 // If we don't have at least 10% dead items then
  366                 // we claim we're out of memory.
  367                 return AddError.OutOfMemory;
  368             }
  369 
  370             const id = self.insert(base, value, self.next_id, ctx);
  371             items[id].meta.ref += 1;
  372             assert(items[id].meta.ref == 1);
  373             self.living += 1;
  374 
  375             // Its possible insert returns a different ID by reusing a
  376             // dead item so we only need to update next id if we used it.
  377             if (id == self.next_id) self.next_id += 1;
  378 
  379             return id;
  380         }

   40 /// The offset from the base address of the page to the start of some data.
   41 /// This is typed for ease of use.
   42 ///
   43 /// This is a packed struct so we can attach methods to an int.
   44 pub fn Offset(comptime T: type) type {
   45     return packed struct(OffsetInt) {
   46         const Self = @This();
   47 
   48         offset: OffsetInt = 0,
   49 
   50         /// A slice of type T that stores via a base offset and len.
   51         pub const Slice = struct {
   52             offset: Self = .{},
   53             len: usize = 0,
   54 
   55             /// Returns a slice for the data, properly typed.
   56             pub inline fn slice(self: Slice, base: anytype) []T {
   57                 return self.offset.ptr(base)[0..self.len];
   58             }
   59         };
   60 
   61         /// Returns a pointer to the start of the data, properly typed.
   62         pub inline fn ptr(self: Self, base: anytype) [*]T {
   63             // The offset must be properly aligned for the type since
   64             // our return type is naturally aligned. We COULD modify this
   65             // to return arbitrary alignment, but its not something we need.
   66             const addr = intFromBase(base) + self.offset;
   67             assert(addr % @alignOf(T) == 0);
   68             return @ptrFromInt(addr);
   69         }
   70     };
   71 }
   73 /// Represents a buffer that is offset from some base pointer.
   74 /// Offset-based structures should use this as their initialization
   75 /// parameter so that they can know what segment of memory they own
   76 /// while at the same time initializing their offset fields to be
   77 /// against the true base.
   78 ///
   79 /// The term "true base" is used to describe the base address of
   80 /// the allocation, which i.e. can include memory that you do NOT
   81 /// own and is used by some other structures. All offsets are against
   82 /// this "true base" so that to determine addresses structures don't
   83 /// need to add up all the intermediary offsets.
   84 pub const OffsetBuf = struct {
   85     /// The true base pointer to the backing memory. This is
   86     /// "byte zero" of the allocation. This plus the offset make
   87     /// it easy to pass in the base pointer in all usage to this
   88     /// structure and the offsets are correct.
   89     base: [*]u8,
   90 
   91     /// Offset from base where the beginning of /this/ data
   92     /// structure is located. We use this so that we can slowly
   93     /// build up a chain of offset-based structures but always
   94     /// have the base pointer sent into functions be the true base.
   95     offset: usize = 0,
   96 
   97     /// Initialize a zero-offset buffer from a base.
   98     pub fn init(base: anytype) OffsetBuf {
   99         return initOffset(base, 0);
  100     }
  101 
  102     /// Initialize from some base pointer and offset.
  103     pub fn initOffset(base: anytype, offset: usize) OffsetBuf {
  104         return .{
  105             .base = @ptrFromInt(intFromBase(base)),
  106             .offset = offset,
  107         };
  108     }
  109 
  110     /// The base address for the start of the data for the user
  111     /// of this OffsetBuf. This is where your data structure should
  112     /// begin; anything before this is NOT your memory.
  113     pub fn start(self: OffsetBuf) [*]u8 {
  114         const ptr = self.base + self.offset;
  115         return @ptrCast(ptr);
  116     }
  117 
  118     /// Returns an Offset calculation for some child member of
  119     /// your struct. The offset is against the true base pointer
  120     /// so that future callers can pass that in as the base.
  121     pub fn member(
  122         self: OffsetBuf,
  123         comptime T: type,
  124         len: usize,
  125     ) Offset(T) {
  126         return .{ .offset = @intCast(self.offset + len) };
  127     }
  149 pub inline fn getOffset(
  150     comptime T: type,
  151     base: anytype,
  152     ptr: *const T,
  153 ) Offset(T) {
  154     const base_int = intFromBase(base);
  155     const ptr_int = @intFromPtr(ptr);
  156     const offset = ptr_int - base_int;
  157     return .{ .offset = @intCast(offset) };
