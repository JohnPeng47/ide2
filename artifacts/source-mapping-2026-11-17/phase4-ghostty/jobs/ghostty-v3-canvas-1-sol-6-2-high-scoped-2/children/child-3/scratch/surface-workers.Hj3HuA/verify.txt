  234         }

  240         var seen_eacces = false;

  242         while (it.next()) |search_path| {

  152         // Take the latest present from the renderer's queue, if any, and
  153         // build a texture from it. `take` closes any present that was

  180         if (widget.getWidth() == 0 or widget.getHeight() == 0) return;

  182         // Map one texture texel to one device pixel. Widget CSS size would
  183         // stretch a fractionally scaled buffer. Snap the surface-relative
  184         // origin so parent chrome cannot place the texture between pixels.
  185         const layout = scale_util.widgetLayout(widget);
  186         const origin = layout.snapOrigin();

  188         const css_w: f32 = @as(f32, @floatFromInt(texture.getWidth())) / scale;
  189         const css_h: f32 = @as(f32, @floatFromInt(texture.getHeight())) / scale;

  191         // GTK, Metal, Vulkan, etc. all have their origin points in the
  192         // top left, but not OpenGL!
  193         const flip = !rendererpkg.Renderer.custom_shader_y_is_down;
  194         if (flip) snap.save();
  195         defer if (flip) snap.restore();
  196         if (flip) {
  197             snap.translate(&.{ .f_x = 0, .f_y = css_h });
  198             snap.scale(1, -1);

  202             .f_origin = .{ .f_x = @floatCast(origin.x), .f_y = @floatCast(origin.y) },

   36 /// The number of threads currently waiting to acquire `mutex` via
   37 /// `lockDemand`. This is not protected by the mutex; it is read by
   38 /// hot lock/unlock loops (the IO parse thread) in `yieldToDemand` to
   39 /// decide whether to hand the mutex off before relocking it.
   40 demand: std.atomic.Value(u32) = .init(0),
   41 
   42 /// Handoff generation counter. Incremented (with a futex wake) by
   43 /// `unlockDemand` after a demanding waiter releases the mutex, so that
   44 /// `yieldToDemand` knows the waiter had its turn.
   45 handoff_gen: std.atomic.Value(u32) = .init(0),
   46 
   47 /// How long `yieldToDemand` sleeps waiting for a demanding waiter to
   48 /// take its turn before giving up. This bounds how long the IO parse
   49 /// thread can stall if a wake is lost or the waiter is descheduled; a
   50 /// demanding critical section (the renderer's frame snapshot) is
   51 /// microseconds, so one millisecond is generous.
   52 const handoff_timeout_ns = 1 * std.time.ns_per_ms;
   53 
   54 /// Acquire `mutex` while signaling demand for it. Use this instead of
   55 /// locking the mutex directly on threads that must not be starved by
   56 /// a hot lock/unlock loop (the renderer's frame snapshot). Must be
   57 /// released with `unlockDemand`; releasing with `mutex.unlock` keeps
   58 /// the data safe but makes parked `yieldToDemand` callers wait out
   59 /// their full timeout.
   60 ///
   61 /// Both `std.Thread.Mutex` and os_unfair_lock are unfair: a running
   62 /// thread that unlocks and immediately relocks beats a sleeping
   63 /// waiter every time, because the waiter must first be woken and
   64 /// scheduled. Under sustained pty output the IO parse thread is
   65 /// exactly such a loop, so without this signal the renderer can
   66 /// starve for as long as the output lasts.
   67 pub fn lockDemand(self: *State, io: std.Io) void {
   68     _ = self.demand.fetchAdd(1, .monotonic);
   69     self.mutex.lockUncancelable(io);
   70     const prev = self.demand.fetchSub(1, .monotonic);
   71     assert(prev > 0);
   72 }
   73 
   74 /// Release `mutex` acquired via `lockDemand` and notify hot loops
   75 /// parked in `yieldToDemand` that the demanding waiter had its turn.
   76 pub fn unlockDemand(self: *State, io: std.Io) void {
   77     self.mutex.unlock(io);
   78     _ = self.handoff_gen.fetchAdd(1, .monotonic);
   79     io.futexWake(u32, &self.handoff_gen.raw, 1);
   80 }
   81 
   82 /// Called by hot lock/unlock loops between critical sections, with
   83 /// `mutex` NOT held: if a `lockDemand` waiter exists, sleep until it
   84 /// has acquired and released the mutex (or the timeout passes). This
   85 /// is the handoff that unfair mutexes never do on their own.
   86 ///
   87 /// The orderings here are all monotonic because these atomics are a
   88 /// scheduling heuristic, not a synchronization boundary: the mutex
   89 /// itself orders the protected data, and the timeout bounds any
   90 /// staleness.
   91 pub fn yieldToDemand(self: *State, io: std.Io) void {
   92     if (self.demand.load(.monotonic) == 0) return;
   93 
   94     // Snapshot the generation before rechecking demand: if the waiter
   95     // acquires and releases between our check and the wait below, the
   96     // generation no longer matches and timedWait returns immediately.
   97     const gen = self.handoff_gen.load(.monotonic);
   98     if (self.demand.load(.monotonic) == 0) return;
   99     io.futexWaitTimeout(
  100         u32,
  101         &self.handoff_gen.raw,
  102         gen,
  103         .{ .duration = .{ .raw = .fromNanoseconds(handoff_timeout_ns), .clock = .awake } },
  104     ) catch {};
  105 }

  200 fn threadMain_(self: *Thread) !void {

  202 
  203     // Right now, on Darwin, `std.Thread.setName` can only name the current
  204     // thread, and we have no way to get the current thread from within it,
  205     // so instead we use this code to name the thread instead.
  206     if (builtin.target.os.tag.isDarwin()) {
  207         internal_os.macos.pthread_setname_np(&"renderer".*);
  208     }
  209 
  210     // Setup our crash metadata
  211     crash.sentry.thread_state = .{
  212         .type = .renderer,
  213         .surface = self.renderer.surface_mailbox.surface,
  214     };
  215     defer crash.sentry.thread_state = null;
  216 
  217     // Setup our thread QoS
  218     self.setQosClass();
  219 
  220     // Run our loop start/end callbacks if the renderer cares.
  221     const has_loop = @hasDecl(rendererpkg.Renderer, "loopEnter");
  222     if (has_loop) try self.renderer.loopEnter(self);
  223     defer if (has_loop) self.renderer.loopExit();
  224 
  225     // Run our thread start/end callbacks. This is important because some
  226     // renderers have to do per-thread setup. For example, OpenGL has to set
  227     // some thread-local state since that is how it works.
  228     try self.renderer.threadEnter(self.surface);
  229     defer self.renderer.threadExit();

  231     // Start the async handlers
  232     self.wakeup.wait(&self.loop, &self.wakeup_c, Thread, self, wakeupCallback);
  233     self.stop.wait(&self.loop, &self.stop_c, Thread, self, stopCallback);
  234     self.draw_now.wait(&self.loop, &self.draw_now_c, Thread, self, drawNowCallback);
  235 
  236     // Send an initial wakeup message so that we render right away.
  237     try self.wakeup.notify();

  241         &self.loop,
  242         &self.cursor_c,
  243         cursorBlinkInterval(),

  245         self,
  246         cursorTimerCallback,
  247     );

  250     // animation wakes (e.g. custom shaders loaded at startup).
  251     self.armAnimationTimer();

  290 fn drainMailbox(self: *Thread) !void {
  291     // There's probably a more elegant way to do this...
  292     //
  293     // This is effectively an @autoreleasepool{} block, which we need in
  294     // order to ensure that autoreleased objects are properly released.
  295     const pool = if (builtin.target.os.tag.isDarwin())
  296         @import("objc").AutoreleasePool.init()
  297     else
  298         void;
  299     defer if (builtin.target.os.tag.isDarwin()) pool.deinit();

  301     while (self.mailbox.pop(global.io())) |message| {
  302         log.debug("mailbox message={}", .{message});
  303         switch (message) {
  304             .crash => @panic("crash request, crashing intentionally"),
  305 
  306             .visible => |v| visible: {
  307                 // If our state didn't change we do nothing.
  308                 if (self.flags.visible == v) break :visible;
  309 
  310                 // Set our visible state
  311                 self.flags.visible = v;
  312 
  313                 // Visibility affects our QoS class
  314                 self.setQosClass();
  315 
  316                 // If we became visible then we immediately rebuild cells
  317                 // (renderCallback skips updateFrame while invisible) and
  318                 // draw. Going through renderCallback also reschedules
  319                 // any Kitty graphics animation wakeup that lapsed
  320                 // while we were invisible.
  321                 if (v) _ = renderCallback(self, undefined, undefined, {});
  322 
  323                 // Notify the renderer so it can update any state.
  324                 self.renderer.setVisible(v);
  325 
  326                 // Note that we're explicitly today not stopping any
  327                 // cursor timers, draw timers, etc. These things have very
  328                 // little resource cost and properly maintaining their active
  329                 // state across different transitions is going to be bug-prone,
  330                 // so its easier to just let them keep firing and have them
  331                 // check the visible state themselves to control their behavior.
  332             },
  333 
  334             .focus => |v| focus: {
  335                 // If our state didn't change we do nothing.
  336                 if (self.flags.focused == v) break :focus;

  537 fn renderCallback(
  538     self_: ?*Thread,
  539     _: *xev.Loop,
  540     _: *xev.Completion,
  541     r: xev.Timer.RunError!void,
  542 ) xev.CallbackAction {
  543     _ = r catch |err| switch (err) {
  544         // Sent when a scheduled animation wakeup is superseded by a
  545         // newer one (Timer.reset cancels the pending run). Nothing to
  546         // do; the replacement timer carries on.
  547         error.Canceled => return .disarm,
  548         else => unreachable,
  549     };
  550     const t: *Thread = self_ orelse {
  551         // This shouldn't happen so we log it.
  552         log.warn("render callback fired without data set", .{});
  553         return .disarm;
  554     };
  555 
  556     // If the display is now unrealized, release GPU resources now
  557     // we're on the render thread, and do not try to update and draw
  558     // this frame.
  559     if (!t.renderer.display_realized) {
  560         t.renderer.draw_mutex.lockUncancelable(global.io());
  561         defer t.renderer.draw_mutex.unlock(global.io());
  562 
  563         t.renderer.releaseGpuResources();
  564         return .disarm;
  565     }
  566 
  567     // If we're not visible there's no point spending CPU rebuilding cells —
  568     // we'll catch up when the .visible mailbox message flips us back on.
  569     // Kitty graphics animations pause with us and resume on visibility.
  570     if (!t.flags.visible) return .disarm;
  571 
  572     // Update our frame data
  573     t.renderer.updateFrame(
  574         t.state,
  575         t.flags.cursor_blink_visible,
  576     ) catch |err|
  577         log.warn("error rendering err={}", .{err});
  578 
  579     // Draw
  580     t.drawFrame(false);
  581 
  582     // Schedule the next animation wake, if the renderer needs one.
  583     t.armAnimationTimer();
  584 
  585     return .disarm;
  586 }

  588 /// Schedule the animation timer for the renderer's next animation
  589 /// wake, if it needs one.
  590 ///
  591 /// This is called after every frame update or animation draw and
  592 /// whenever the wake inputs change (focus, config, visibility
  593 /// regain). Resetting a pending timer is always safe: every call
  594 /// recomputes the wake, so the deadline only ever moves toward the
  595 /// actual next wake.
  596 fn armAnimationTimer(self: *Thread) void {
  597     const wake = self.renderer.animationWake() orelse return;
  598     self.animation_wake = wake.kind;
  599     self.render_h.reset(
  600         &self.loop,
  601         &self.render_c,
  602         &self.render_c_cancel,
  603         wake.delay_ms,
  604         Thread,
  605         self,
  606         animationTimerCallback,
  607     );
  608 }

  618         // newer one (Timer.reset cancels the pending run). Nothing to

  646         .draw => {

  303         /// render a frame, so that we can start building the next frame while
  304         /// the previous frame is still being processed on the GPU.
  305         const SwapChain = struct {
  306             // The count of buffers we use for double/triple buffering.
  307             // If this is one then we don't do any double+ buffering at all.
  308             // This is comptime because there isn't a good reason to change
  309             // this at runtime and there is a lot of complexity to support it.
  310             const buf_count = GraphicsAPI.swap_chain_count;
  311 
  312             /// `buf_count` structs that can hold the
  313             /// data needed by the GPU to draw a frame.
  314             frames: [buf_count]FrameState,
  315             /// Index of the most recently used frame state struct.
  316             frame_index: std.math.IntFittingRange(0, buf_count) = 0,
  317             /// Semaphore that we wait on to make sure we have an available
  318             /// frame state struct so we can start working on a new frame.
  319             frame_sema: std.Io.Semaphore = .{ .permits = buf_count },

  321             pub fn init(api: GraphicsAPI, custom_shaders: bool) !SwapChain {

  327                 }

 1107         /// The cadence of continuous (draw-only) animation wakes,
 1108         /// i.e. 120fps, and the floor for any animation wake delay.
 1109         pub const draw_interval_ms: u64 = 8;
 1110 
 1111         /// A point in the future when the renderer needs to be driven
 1112         /// again to keep animating, and what kind of drive it needs.
 1113         pub const AnimationWake = struct {
 1114             /// Delay in milliseconds until the wake is due.
 1115             delay_ms: u64,
 1116             kind: Kind,
 1117 
 1118             pub const Kind = enum {
 1119                 /// A redraw alone suffices, no updateFrame. Much cheaper
 1120                 /// than `update`.
 1121                 draw,
 1122 
 1123                 /// Frame data must be updated first: updateFrame, then draw.
 1124                 update,
 1125             };
 1126         };
 1127 
 1128         /// The soonest animation wake this renderer needs, if any:
 1129         /// custom shader animation wants continuous draw-only wakes
 1130         /// at draw_interval_ms while active, and a running Kitty
 1131         /// graphics animation wants an update wake when its next
 1132         /// frame is due. The renderer thread drives its animation
 1133         /// timer off this, re-querying after every wake.
 1134         ///
 1135         /// Must be called on the render thread.
 1136         pub fn animationWake(self: *const Self) ?AnimationWake {
 1137             // Custom shaders animate by redrawing on a fixed cadence,
 1138             // gated by configuration and focus.
 1139             const shader_delay: ?u64 = shader: {
 1140                 if (!self.has_custom_shaders) break :shader null;
 1141                 break :shader switch (self.config.custom_shader_animation) {
 1142                     .false => null,
 1143                     .always => draw_interval_ms,
 1144                     .true => if (self.focused) draw_interval_ms else null,
 1145                 };
 1146             };
 1147 
 1148             // Kitty animations tick during updateFrame; between
 1149             // updates the deadline is absolute on the animation
 1150             // clock, so a stream of draw wakes recomputing this
 1151             // cannot starve it into the future.
 1152             const kitty_delay: ?u64 = kitty: {
 1153                 const next = self.kitty_animation_next_ms orelse break :kitty null;
 1154                 const base = self.kitty_animation_clock orelse break :kitty null;
 1155                 const now: std.Io.Timestamp = .now(global.io(), .awake);
 1156                 const now_ms: u64 = @intCast(@divTrunc(
 1157                     base.durationTo(now).nanoseconds,
 1158                     std.time.ns_per_ms,
 1159                 ));
 1160                 // Never wake faster than the draw interval; an
 1161                 // overdue frame is picked up on the next wake.
 1162                 break :kitty @max(next -| now_ms, draw_interval_ms);
 1163             };
 1164 
 1165             // An update wake includes a draw, so it wins ties.
 1166             if (kitty_delay) |k| {
 1167                 if (shader_delay == null or k <= shader_delay.?) {
 1168                     return .{ .delay_ms = k, .kind = .update };
 1169                 }
 1170             }
 1171 
 1172             if (shader_delay) |s| return .{ .delay_ms = s, .kind = .draw };
 1173 
 1174             return null;
 1175         }

 1371         pub fn updateFrame(
 1372             self: *Self,
 1373             state: *renderer.State,
 1374             cursor_blink_visible: bool,
 1375         ) Allocator.Error!void {
 1376             // CoreText shaping accumulates objects for deferred release over
 1377             // the course of a frame. Always flush those objects, including
 1378             // when rebuilding the frame fails due to memory pressure.
 1379             defer self.font_shaper.endFrame();
 1380 
 1381             // We fully deinit and reset the terminal state every so often
 1382             // so that a particularly large terminal state doesn't cause
 1383             // the renderer to hold on to retained memory.
 1384             //
 1385             // Frame count is ~12 minutes at 120Hz.
 1386             const max_terminal_state_frame_count = 100_000;
 1387             if (self.terminal_state_frame_count >= max_terminal_state_frame_count) {
 1388                 self.terminal_state.deinit(self.alloc);
 1389                 self.terminal_state = .empty;
 1390                 self.terminal_state_frame_count = 0;
 1391             }
 1392             self.terminal_state_frame_count += 1;
 1393 
 1394             // Create an arena for all our temporary allocations while rebuilding
 1395             var arena = ArenaAllocator.init(self.alloc);
 1396             defer arena.deinit();
 1397             const arena_alloc = arena.allocator();
 1398 
 1399             // Data we extract out of the critical area.
 1400             const Critical = struct {
 1401                 links: terminal.RenderState.CellSet,
 1402                 mouse: renderer.State.Mouse,
 1403                 preedit: ?renderer.State.Preedit,
 1404                 scrollbar: terminal.Scrollbar,
 1405                 overlay_features: []const Overlay.Feature,
 1406             };
 1407 
 1408             // Update all our data as tightly as possible within the mutex.
 1409             var critical: Critical = critical: {
 1410                 // NOTE: This code needs be updated to 0.16.0 before you
 1411                 // un-comment it ;)
 1412                 //
 1413                 // const start = try std.time.Instant.now();
 1414                 // const start_micro = std.time.microTimestamp();
 1415                 // defer {
 1416                 //     const end = std.time.Instant.now() catch unreachable;
 1417                 //     std.log.err("[updateFrame critical time] start={}\tduration={} us", .{ start_micro, end.since(start) / std.time.ns_per_us });
 1418                 // }
 1419 
 1420                 state.lockDemand(global.io());
 1421                 defer state.unlockDemand(global.io());
 1422 
 1423                 // If we're in a synchronized output state, we pause all rendering.
 1424                 if (state.terminal.modes.get(.synchronized_output)) {
 1425                     log.debug("synchronized output started, skipping render", .{});
 1426                     return;
 1427                 }
 1428 
 1429                 // If scroll-to-bottom on output is enabled, check if the final line
 1430                 // changed by comparing the bottom-right pin. If the node pointer or
 1431                 // y offset changed, new content was added to the screen.
 1432                 // Update this BEFORE we update our render state so we can
 1433                 // draw the new scrolled data immediately.
 1434                 if (self.config.scroll_to_bottom_on_output) scroll: {
 1435                     const br = state.terminal.screens.active.pages.getBottomRight(.screen) orelse break :scroll;
 1436 
 1437                     // If the pin hasn't changed, then don't scroll.
 1438                     if (self.last_bottom_node == @intFromPtr(br.node) and
 1439                         self.last_bottom_y == br.y) break :scroll;
 1440 
 1441                     // Update tracked pin state for next frame
 1442                     self.last_bottom_node = @intFromPtr(br.node);
 1443                     self.last_bottom_y = br.y;
 1444 
 1445                     // Scroll
 1446                     state.terminal.scrollViewport(.bottom);
 1447                 }
 1448 
 1449                 // Advance any running Kitty graphics animations to the
 1450                 // frame due now, and remember when the next frame is
 1451                 // due (as an absolute deadline, see animationWake) so
 1452                 // the renderer thread can schedule a wakeup for it.
 1453                 //
 1454                 // This runs before the render state update. Advancing

 1712         /// Draw the frame to the screen.
 1713         ///
 1714         /// If `sync` is true, this will synchronously block until
 1715         /// the frame is finished drawing and has been presented.
 1716         pub fn drawFrame(
 1717             self: *Self,
 1718             sync: bool,
 1719         ) !void {
 1720             // Everything that touches draw state happens under the draw
 1721             // mutex. The display link is synced after the mutex is
 1722             // released. It doesn't need the lock, and this keeps the
 1723             // critical section to just drawing.
 1724             const sync_display_link = locked: {
 1725                 self.draw_mutex.lockUncancelable(global.io());
 1726                 defer self.draw_mutex.unlock(global.io());
 1727                 break :locked try self.drawFrameLocked(sync);
 1728             };
 1729 
 1730             if (sync_display_link) self.syncDisplayLink(null, null);
 1731         }

 1733         /// The body of `drawFrame`. Must be called with `draw_mutex` held.
 1734         ///
 1735         /// Returns true if the display link should be resynced once the
 1736         /// draw mutex is released. This is only ever true on the no-redraw
 1737         /// path, which a sync draw never takes. Sync draws come from the
 1738         /// main thread, so this keeps `syncDisplayLink` on the render
 1739         /// thread as its contract requires.
 1740         fn drawFrameLocked(
 1741             self: *Self,
 1742             sync: bool,
 1743         ) !bool {
 1744             // After the graphics API is complete (so we defer) we want to
 1745             // update our scrollbar state.
 1746             defer if (self.scrollbar_dirty) {
 1747                 // Fail instantly if the surface mailbox if full, we'll just
 1748                 // get it on the next frame.
 1749                 if (self.surface_mailbox.push(.{
 1750                     .scrollbar = self.scrollbar,
 1751                 }, .instant) > 0) self.scrollbar_dirty = false;
 1752             };
 1753 
 1754             // Let our graphics API do any bookkeeping, etc.
 1755             // that it needs to do before / after `drawFrame`.
 1756             self.api.drawFrameStart();
 1757             defer self.api.drawFrameEnd();
 1758 
 1759             // Retrieve the most up-to-date surface size from the Graphics API
 1760             const surface_size = try self.api.surfaceSize();
 1761 
 1762             // If either of our surface dimensions is zero
 1763             // then drawing is absurd, so we just return.
 1764             if (surface_size.width == 0 or surface_size.height == 0) return false;
 1765 
 1766             // If we have no graphics context we can't draw. This is
 1767             // only the case while unrealized (GTK) or after the render
 1768             // thread exits; displayRealized rebuilds the swap chain.
 1769             if (!self.display_realized) return false;
 1770 
 1771             // Get our swap chain, rebuilding it if it was released
 1772             // while we were hidden. Rebuilding is deferred to draw
 1773             // time because resource creation must happen somewhere
 1774             // our graphics API allows it (OpenGL requires a current
 1775             // context, which drawFrame guarantees).
 1776             const swap_chain: *SwapChain, const swap_chain_rebuilt: bool =
 1777                 if (self.swap_chain) |*sc| .{ sc, false } else rebuild: {
 1778                     self.swap_chain = try SwapChain.init(
 1779                         self.api,
 1780                         self.has_custom_shaders,
 1781                     );
 1782                     break :rebuild .{ &self.swap_chain.?, true };
 1783                 };
 1784 
 1785             const size_changed =
 1786                 self.size.screen.width != surface_size.width or
 1787                 self.size.screen.height != surface_size.height;
 1788 
 1789             // Conditions under which we need to draw the frame, otherwise we
 1790             // don't need to since the previous frame should be identical.
 1791             //
 1792             // While any animation is in progress (a pending animation wake)
 1793             // every draw must actually render.
 1794             const needs_redraw =
 1795                 size_changed or
 1796                 swap_chain_rebuilt or
 1797                 self.cells_rebuilt or
 1798                 self.images_rebuilt or
 1799                 self.animationWake() != null or
 1800                 sync;
 1801 
 1802             if (!needs_redraw) {
 1803                 // Ask our caller to resync the display link once the draw
 1804                 // mutex is released, because we can probably pause the
 1805                 // display link at this point.
 1806                 return true;
 1807             }
 1808             self.cells_rebuilt = false;
 1809             self.images_rebuilt = false;
 1810 
 1811             // Wait for a frame to be available.
 1812             const frame = swap_chain.nextFrame();
 1813             errdefer swap_chain.releaseFrame();
 1814             // log.debug("drawing frame index={}", .{swap_chain.frame_index});
 1815 
 1816             // If we need to reinitialize our shaders, do so.
 1817             if (self.reinitialize_shaders) {
 1818                 self.reinitialize_shaders = false;
 1819                 self.shaders.deinit(self.alloc);
 1820                 try self.initShaders();
 1821             }
 1822 
 1823             // Our shaders should not be defunct at this point.
 1824             assert(!self.shaders.defunct);
 1825 
 1826             // If we have custom shaders, make sure we have the
 1827             // custom shader state in our frame state, otherwise
 1828             // if we have a state but don't need it we remove it.
 1829             if (self.has_custom_shaders) {
 1830                 if (frame.custom_shader_state == null) {
 1831                     frame.custom_shader_state = try .init(self.api);
 1832                     try frame.custom_shader_state.?.resize(
 1833                         self.api,
 1834                         surface_size.width,
 1835                         surface_size.height,
 1836                     );
 1837                 }
 1838             } else if (frame.custom_shader_state) |*state| {
 1839                 state.deinit();
 1840                 frame.custom_shader_state = null;
 1841             }
 1842 
 1843             // If our stored size doesn't match the
 1844             // surface size we need to update it.
 1845             if (size_changed) {
 1846                 self.size.screen = .{
 1847                     .width = surface_size.width,
 1848                     .height = surface_size.height,
 1849                 };
 1850                 self.updateScreenSizeUniforms();
 1851             }
 1852 
 1853             // If this frame's target isn't the correct size, or the target
 1854             // config has changed (such as when the blending mode changes),
 1855             // remove it and replace it with a new one with the right values.
 1856             if (frame.target.width != self.size.screen.width or
 1857                 frame.target.height != self.size.screen.height or
 1858                 frame.target_config_modified != self.target_config_modified)
 1859             {

 1872             try self.uploadBackgroundImage();

 1874             // Update per-frame custom shader uniforms.
 1875             try self.updateCustomShaderUniformsForFrame();

 1877             // Setup our frame data
 1878             try frame.uniforms.sync(&.{self.uniforms});
 1879             try frame.cells_bg.sync(self.cells.bg_cells);

 1890             texture: {

 1893                 self.font_grid.lock.lockSharedUncancelable(global.io());
 1894                 defer self.font_grid.lock.unlockShared(global.io());

 1922                 // The bg image shader also draws the main bg color.

 1925                 // draw the background color by itself in its own step.

 1928                 //       would require us to do color space conversion on the

 1982                     },

 2049         ) void {

 2066             self.swap_chain.?.releaseFrame();

   51 /// Complete this frame and present the target.
   52 ///
   53 /// If `sync` is true, this will block until the frame is presented.
   54 ///
   55 /// NOTE: For OpenGL, `sync` is ignored and we never block, instead
   56 /// pushing the newly presented and exported frame to the frame queue.
   57 pub fn complete(self: *const Self, sync: bool) void {
   58     _ = sync;
   59 
   60     // If there are any GL errors, consider the frame unhealthy.
   61     const health: Health = if (gl.errors.getError()) .healthy else |_| .unhealthy;
   62 
   63     // If the frame is healthy, draw it to an ExportedFrame
   64     // Then sync before exporting and pushing to the present queue.
   65     // The apprt pulls from this queue in its snapshot handler.
   66     const presented_frame: ?OpenGL.ExportedFrame = if (health == .healthy) frame: {
   67         const frame = self.renderer.api.present(
   68             self.target.*,
   69             self.renderer.presentationHealth(),
   70         ) catch |err| {
   71             log.warn("failed to present render target: err={}", .{err});
   72             break :frame null;
   73         };
   74         break :frame frame;
   75     } else null;
   76 
   77     // Sync after frame draw AND frame present GL calls.
   78     gl.finish();
   79 
   80     // At this point the ExportedFrame is finished and can be shared
   81     if (presented_frame) |frame| {
   82         self.renderer.pushFrame(frame);
   83 
   84         // Notify the surface that it should redraw
   85         _ = self.renderer.surface_mailbox.push(.redraw, .{ .forever = {} });
   86     }
   87 
   88     // Report the health to the renderer.
   89     self.renderer.frameCompleted(health);

 1501                     // A parameter digit:
 1502                     '0'...'9' => if (self.parser.params_idx < Parser.MAX_PARAMS) {
 1503                         self.parser.param_acc *|= 10;
 1504                         self.parser.param_acc +|= c - '0';
 1505                         // The parser's CSI param action uses param_acc_idx
 1506                         // to decide if there's a final param that needs to
 1507                         // be consumed or not, but it doesn't matter really
 1508                         // what it is as long as it's not 0.
 1509                         self.parser.param_acc_idx |= 1;
 1510                     },
 1511                     // A parameter separator:
 1512                     ':', ';' => if (self.parser.params_idx < Parser.MAX_PARAMS) {
 1513                         self.parser.params[self.parser.params_idx] = self.parser.param_acc;
 1514                         if (c == ':') self.parser.params_sep.set(self.parser.params_idx);
 1515                         self.parser.params_idx += 1;
 1516 
 1517                         self.parser.param_acc = 0;
 1518                         self.parser.param_acc_idx = 0;
 1519                     },
 1520                     // A final byte: dispatch the CSI directly.
 1521                     0x40...0x7E => if (comptime !has_vt_raw) {
 1522                         self.csiDispatchFinal(c);
 1523                     } else break :csi_param,
 1524                     // Explicitly ignored:
 1525                     0x7F => {},
 1526                     // Defer to the state machine to
 1527                     // handle any other characters:
 1528                     else => break :csi_param,
 1529                 }
 1530                 return;
 1531             }
 1532 
 1533             // Fast path for CSI entry, the state right after "ESC [".
 1534             if (comptime !has_vt_raw) {
 1535                 if (self.parser.state == .csi_entry) {
 1536                     if (self.csiEntryByte(c)) return;
 1537                 }
 1538             }
 1539 
 1540             // String boundaries are parser state transitions, not
 1541             // handler actions: an OSC or DCS can start and end without
 1542             // producing one, and a vtRaw handler can skip any action.
 1543             // So in until mode they are found by comparing the parser
 1544             // state before and after the byte.
 1545             const prev_state = if (comptime until) self.parser.state else {};
 1546 
 1547             // We explicitly inline this call here for performance reasons.
 1548             //
 1549             // We do this rather than mark Parser.next as inline because doing
 1550             // that causes weird behavior in some tests- I'm not sure if they
 1551             // miscompile or it's just very counter-intuitive comptime stuff,
 1552             // but regardless, this is the easy solution.
 1553             const actions = @call(.always_inline, Parser.next, .{ &self.parser, c });
 1554             if (comptime until) u.transition(prev_state, self.parser.state, c);
 1555 
 1556             for (actions) |action_opt| {
 1557                 const action = action_opt orelse continue;
 1558                 if (comptime debug) log.info("action: {f}", .{action});
 1559 
 1560                 // A handler can expose this to get the raw action before
 1561                 // it is further parsed. If this returns `true` then we skip
 1562                 // processing ourselves.
 1563                 if (@hasDecl(T, "vtRaw")) {
 1564                     const skip = self.handler.vtRaw(action) catch |err| err: {
 1565                         log.warn("error handling action manually err={} action={f}", .{
 1566                             err,
 1567                             action,
 1568                         });
 1569                         // Always skip erroneous actions because we can't
 1570                         // be sure...
 1571                         break :err true;
 1572                     };
 1573 
 1574                     if (skip) continue;
 1575                 }
 1576 
 1577                 switch (action) {
 1578                     .print => |p| self.print(p),
 1579                     .execute => |code| self.execute(code),
 1580                     .csi_dispatch => |csi_action| self.csiDispatch(csi_action),
 1581                     .esc_dispatch => |esc| self.escDispatch(esc),
 1582                     .osc_dispatch => |cmd| self.oscDispatch(cmd),
 1583                     .dcs_hook => |dcs| self.handler.vt(.dcs_hook, dcs),
 1584                     .dcs_put => |code| self.handler.vt(.dcs_put, code),
 1585                     .dcs_unhook => self.handler.vt(.dcs_unhook, {}),
 1586                     .apc_start => self.handler.vt(.apc_start, {}),
 1587                     .apc_put => |code| self.handler.vt(.apc_put, code),
 1588                     .apc_end => self.handler.vt(.apc_end, .{
 1589                         .terminated = c == std.ascii.control_code.esc or c == 0x9C,
 1590                     }),
 1591                 }
 1592             }
 1593         }

   92     // Start our subprocess
   93     const pty_fds = self.subprocess.start(alloc) catch |err| {
   94         // If we specifically got this error then we are in the forked
   95         // process and our child failed to execute. If we DIDN'T
   96         // get this specific error then we're in the parent and
   97         // we need to bubble it up.
   98         if (err != error.ExecFailedInChild) return err;
   99 
  100         // We're in the child. Nothing more we can do but abnormal exit.
  101         // The Command will output some additional information.
  102         std.process.exit(1);
  103     };
  104     errdefer self.subprocess.stop();

  106     // Watcher to detect subprocess exit
  107     var process: ?xev.Process = if (self.subprocess.process) |v| switch (v) {
  108         .fork_exec => |cmd| try xev.Process.init(
  109             cmd.pid orelse return error.ProcessNoPid,
  110         ),

  112         // If we're executing via Flatpak then we can't do
  113         // traditional process watching (its implemented
  114         // as a special case in os/flatpak.zig) since the
  115         // command is on the host.
  116         .flatpak => null,

  121     const process_start: std.Io.Timestamp = .now(global.io(), .awake);

  124     // pipe[0] is the read end, pipe[1] is the write end.

  136     var termios_timer = try xev.Timer.init();
  137     errdefer termios_timer.deinit();

  151         .process = process,
  152         .read_thread = read_thread,
  153         .read_thread_pipe = pipe[1],

  155         .termios_timer = termios_timer,
  156     } };
  157 
  158     // Start our process watcher. If we have an xev.Process use it.
  159     if (process) |*p| p.wait(
  160         td.loop,
  161         &td.backend.exec.process_wait_c,
  162         termio.Termio.ThreadData,
  163         td,
  164         processExit,
  165     ) else if (comptime build_config.flatpak) flatpak: {
  166         switch (self.subprocess.process orelse break :flatpak) {

  168             // then we can run the special flatpak logic for watching.
  169             .flatpak => |*c| c.waitXev(

  171                 &td.backend.exec.flatpak_wait_c,
  172                 termio.Termio.ThreadData,
  173                 td,
  174                 flatpakExit,
  175             ),
  176 
  177             .fork_exec => {},
  178         }
  179     }
  180

  182     // Fundamentally, we could support this on Windows so we're just
  183     // waiting for someone to implement it.
  184     if (comptime builtin.target.os.tag != .windows) {
  185         termios_timer.run(
  186             td.loop,
  187             &td.backend.exec.termios_timer_c,
  188             TERMIOS_POLL_MS,
  189             termio.Termio.ThreadData,
  190             td,
  191             termiosTimer,

  196 pub fn threadExit(self: *Exec, td: *termio.Termio.ThreadData) void {
  197     assert(td.backend == .exec);
  198     const exec = &td.backend.exec;
  199 
  200     if (exec.exited) self.subprocess.externalExit();
  201     self.subprocess.stop();
  202 
  203     // Quit our read thread after exiting the subprocess so that
  204     // we don't get stuck waiting for data to stop flowing if it is
  205     // a particularly noisy process.
  206     switch (posix.errno(posix.system.write(exec.read_thread_pipe, "x", 1))) {
  207         .SUCCESS => {},
  208 
  209         // EPIPE means that our read thread is closed already, which is
  210         // completely fine since that is what we were trying to achieve.
  211         .PIPE => {},
  212 
  213         else => |e| log.warn(
  214             "error writing to read thread quit pipe err=E{s}",
  215             .{@tagName(e)},
  216         ),
  217     }
  218 
  219     if (comptime builtin.target.os.tag == .windows) {
  220         // Interrupt the blocking read so the thread can see the quit message
  221         if (windows.exp.kernel32.CancelIoEx(exec.read_thread_fd, null) == windows.FALSE) {
  222             switch (windows.GetLastError()) {
  223                 .NOT_FOUND => {},
  224                 else => |err| log.warn("error interrupting read thread err={}", .{err}),
  225             }
  226         }
  227     }
  228 
  229     exec.read_thread.join();
  230 }

  403 pub fn queueWrite(
  404     self: *Exec,
  405     alloc: Allocator,
  406     td: *termio.Termio.ThreadData,
  407     data: []const u8,
  408     linefeed: bool,
  409 ) !void {
  410     _ = self;
  411     const exec = &td.backend.exec;
  412 
  413     // If our process is exited then we don't send any more writes.
  414     if (exec.exited) return;
  415 
  416     // We go through and chunk the data if necessary to fit into
  417     // our cached buffers that we can queue to the stream.
  418     var i: usize = 0;
  419     while (i < data.len) {
  420         const w = try exec.write_pool.create(alloc);
  421         w.td = exec;
  422         const buf = &w.buf;
  423         const slice = slice: {
  424             // The maximum end index is either the end of our data or
  425             // the end of our buffer, whichever is smaller.
  426             const max = @min(data.len, i + buf.len);
  427 
  428             // Fast
  429             if (!linefeed) {
  430                 fastmem.copy(u8, buf, data[i..max]);
  431                 const len = max - i;
  432                 i = max;
  433                 break :slice buf[0..len];
  434             }
  435 
  436             // Slow, have to replace \r with \r\n
  437             var buf_i: usize = 0;
  438             while (i < data.len and buf_i < buf.len - 1) {
  439                 const ch = data[i];
  440                 i += 1;
  441 
  442                 if (ch != '\r') {
  443                     buf[buf_i] = ch;
  444                     buf_i += 1;
  445                     continue;
  446                 }
  447 
  448                 // CRLF
  449                 buf[buf_i] = '\r';
  450                 buf[buf_i + 1] = '\n';
  451                 buf_i += 2;
  452             }
  453 
  454             break :slice buf[0..buf_i];
  455         };
  456 
  457         //for (slice) |b| log.warn("write: {x}", .{b});
  458 
  459         exec.write_stream.queueWrite(
  460             td.loop,
  461             &exec.write_queue,
  462             &w.req,
  463             .{ .slice = slice },
  464             ThreadData.Write,
  465             w,
  466             ttyWrite,
  467         );
  468     }
  469 }

  895     /// Start the subprocess. If the subprocess is already started this
  896     /// will crash.
  897     pub fn start(self: *Subprocess, alloc: Allocator) !struct {
  898         read: Pty.Fd,
  899         write: Pty.Fd,
  900     } {
  901         assert(self.pty == null and self.process == null);
  902 
  903         // This function is funny because on POSIX systems it can
  904         // fail in the forked process. This is flipped to true if
  905         // we're in an error state in the forked process (child
  906         // process).
  907         var in_child: bool = false;
  908 
  909         // Create our pty
  910         var pty = try Pty.open(.{
  911             .ws_row = @intCast(self.grid_size.rows),
  912             .ws_col = @intCast(self.grid_size.columns),
  913             .ws_xpixel = @intCast(self.screen_size.width),
  914             .ws_ypixel = @intCast(self.screen_size.height),
  915         });

  918             if (comptime builtin.target.os.tag != .windows) {
  919                 _ = posix.system.close(pty.slave);
  920             }
  921 
  922             pty.deinit();
  923             self.pty = null;
  924         };
  925 
  926         // Cleanup we only run in our parent when we successfully start
  927         // the process.
  928         defer if (!in_child and self.process != null) {
  929             if (comptime builtin.target.os.tag != .windows) {

  931                 // side. This prevents the slave fd from being leaked to
  932                 // future children.

 1359     /// The state shared between the gather and parse stages. This is
 1360     /// a fixed ring of buffers plus the metadata to rotate ownership
 1361     /// between the two threads. A buffer is owned by exactly one
 1362     /// stage at a time, so buffer contents need no locking. Only the
 1363     /// ring metadata is guarded by the mutex.
 1364     const Pipeline = struct {
 1365         mutex: std.Io.Mutex = .init,
 1366 
 1367         /// Signaled when a batch is published or the gather stage is
 1368         /// done. Waited on by the parse stage.
 1369         batch_ready: std.Io.Condition = .init,
 1370 
 1371         /// Signaled when a batch has been consumed. Waited on by the
 1372         /// gather stage when all buffers are in flight (backpressure).
 1373         slot_free: std.Io.Condition = .init,
 1374 
 1375         /// The number of valid bytes in each buffer. Set at publish
 1376         /// time by the gather stage, read by the parse stage.
 1377         lens: [buffer_count]usize = @splat(0),
 1378 
 1379         /// Ring state: head is the next slot the gather stage fills,
 1380         /// tail is the next slot the parse stage consumes, count is
 1381         /// the number of published, unconsumed batches.
 1382         head: usize = 0,
 1383         tail: usize = 0,
 1384         count: usize = 0,
 1385 
 1386         /// Set by the gather stage (under the mutex) while it sleeps
 1387         /// in a bridge poll. The parse stage only writes to the idle
 1388         /// pipe when this is set, so an interactive terminal never
 1389         /// pays the pipe syscalls.
 1390         bridging: bool = false,
 1391 
 1392         /// A self-pipe the parse stage uses to interrupt the gather
 1393         /// stage's bridge poll the moment it runs out of batches.
 1394         /// Bridging a refill gap is only free while the parse stage
 1395         /// is busy; once it goes idle, every additional microsecond
 1396         /// spent bridging is added straight to output latency. The
 1397         /// write end is used by the parse stage, the read end is
 1398         /// polled by the gather stage. -1 when unavailable, in which
 1399         /// case bridge polls are bounded by their timeout only.
 1400         idle_read_fd: posix.fd_t = -1,
 1401         idle_write_fd: posix.fd_t = -1,
 1402 
 1403         /// Set by the gather stage when the stream is over (quit
 1404         /// signal, EOF, or pty error). The parse stage drains any
 1405         /// remaining batches and then exits.
 1406         done: bool = false,
 1407 
 1408         /// The buffer storage itself.
 1409         bufs: [buffer_count][buffer_capacity]u8 = undefined,
 1410     };

 1412     fn threadMainPosix(fd: posix.fd_t, io: *termio.Termio, quit: posix.fd_t) void {
 1413         // Always close our end of the pipe when we exit.
 1414         defer _ = posix.system.close(quit);
 1415 
 1416         // Right now, on Darwin, `std.Thread.setName` can only name the current
 1417         // thread, and we have no way to get the current thread from within it,
 1418         // so instead we use this code to name the thread instead.
 1419         if (builtin.target.os.tag.isDarwin()) {
 1420             internal_os.macos.pthread_setname_np(&"io-reader".*);
 1421             setQosClass();
 1422         }
 1423 
 1424         // Setup our crash metadata
 1425         crash.sentry.thread_state = .{
 1426             .type = .io,
 1427             .surface = io.surface_mailbox.surface,
 1428         };
 1429         defer crash.sentry.thread_state = null;
 1430 
 1431         // Set the fd to non-blocking so the gather stage can drain it
 1432         // in a tight loop and fall back to poll for readiness. The
 1433         // pipeline can't run with a blocking fd (a blocking read
 1434         // would hang the gather stage on a quiet pty), but this also
 1435         // can't realistically fail on a valid pty master.
 1436         if (!setNonblock(fd)) {
 1437             log.err("read thread exiting, pty fd must be non-blocking", .{});
 1438             return;
 1439         }
 1440 
 1441         // Shared pipeline
 1442         var pipeline: Pipeline = .{};
 1443 
 1444         // The idle self-pipe (see the Pipeline field docs). If we
 1445         // can't create it we still run correctly, bridge polls are
 1446         // just bounded by their timeout instead of being interrupted
 1447         // when the parse stage goes idle.
 1448         if (compat_fd.pipe2(.{
 1449             .CLOEXEC = true,
 1450             .NONBLOCK = true,
 1451         })) |fds| {
 1452             pipeline.idle_read_fd = fds[0];
 1453             pipeline.idle_write_fd = fds[1];
 1454         } else |err| {
 1455             log.warn("read thread failed to create idle pipe err={}", .{err});
 1456         }
 1457         defer if (pipeline.idle_read_fd >= 0) {
 1458             compat_fd.close(pipeline.idle_read_fd);
 1459             compat_fd.close(pipeline.idle_write_fd);
 1460         };
 1461 
 1462         const gather_thread = std.Thread.spawn(
 1463             .{},
 1464             gatherMainPosix,
 1465             .{ fd, quit, &pipeline },
 1466         ) catch |err| {
 1467             // If we can't spawn a thread the process is already
 1468             // doomed (every surface spawns several), so don't try
 1469             // to limp along.
 1470             log.err("read thread exiting, failed to spawn gather thread err={}", .{err});
 1471             return;
 1472         };
 1473         defer gather_thread.join();
 1474         if (comptime !builtin.target.os.tag.isDarwin()) {
 1475             gather_thread.setName(global.io(), "io-gather") catch {};
 1476         }
 1477 
 1478         // This thread is the parse stage. We consume batches in ring
 1479         // order until the gather stage reports the stream is over and
 1480         // the ring is drained.
 1481         while (true) {
 1482             const batch: []const u8 = batch: {
 1483                 pipeline.mutex.lockUncancelable(global.io());
 1484                 defer pipeline.mutex.unlock(global.io());
 1485                 while (pipeline.count == 0) {
 1486                     if (pipeline.done) return;
 1487                     pipeline.batch_ready.waitUncancelable(global.io(), &pipeline.mutex);
 1488                 }
 1489                 const slot = pipeline.tail;
 1490                 break :batch pipeline.bufs[slot][0..pipeline.lens[slot]];
 1491             };
 1492 
 1493             // The batch buffer is owned by this stage until we advance
 1494             // the tail below, so it is safe to read outside the lock.
 1495             io.processOutput(batch);
 1496 
 1497             {
 1498                 pipeline.mutex.lockUncancelable(global.io());
 1499                 pipeline.tail = (pipeline.tail + 1) % buffer_count;
 1500                 pipeline.count -= 1;
 1501                 const wake = pipeline.count == 0 and
 1502                     pipeline.bridging and
 1503                     pipeline.idle_write_fd >= 0;
 1504                 pipeline.mutex.unlock(global.io());
 1505                 pipeline.slot_free.signal(global.io());
 1506 
 1507                 // We ran out of batches while the gather stage is
 1508                 // bridging a refill gap: interrupt its poll so it
 1509                 // delivers what it has instead of sleeping out the
 1510                 // timeout while we sit idle.
 1511                 if (wake) {
 1512                     _ = posix.system.write(pipeline.idle_write_fd, "i", 1);
 1513                 }
 1514             }
 1515 
 1516             // Batch boundary: hand the renderer state mutex off if
 1517             // the renderer is waiting. See renderer.State.lockDemand.
 1518             io.renderer_state.yieldToDemand(global.io());
 1519         }
 1520     }
 1521 
 1522     /// The gather stage. This drains the pty into rotating buffers,
 1523     /// bridging the kernel queue's refill gaps for saturated streams,
 1524     /// and publishes each batch to the parse stage. This thread owns
 1525     /// all fd monitoring, including the quit fd.
 1526     fn gatherMainPosix(fd: posix.fd_t, quit: posix.fd_t, pipeline: *Pipeline) void {
 1527         if (builtin.target.os.tag.isDarwin()) {
 1528             internal_os.macos.pthread_setname_np(&"io-gather".*);
 1529             setQosClass();
 1530         }
 1531 
 1532         // However we exit, tell the parse stage the stream is over so
 1533         // it drains the ring and joins us.
 1534         defer {
 1535             pipeline.mutex.lockUncancelable(global.io());
 1536             pipeline.done = true;
 1537             pipeline.mutex.unlock(global.io());
 1538             pipeline.batch_ready.signal(global.io());
 1539         }
 1540 
 1541         // The fds we poll: data on the pty, our quit notification,
 1542         // and the parse stage's idle wake. The idle fd only
 1543         // participates in bridge polls (the parse stage only writes
 1544         // while we're bridging), so the outer poll slices it off.
 1545         var pollfds: [3]posix.pollfd = .{
 1546             .{ .fd = fd, .events = posix.POLL.IN, .revents = undefined },
 1547             .{ .fd = quit, .events = posix.POLL.IN, .revents = undefined },
 1548             .{ .fd = pipeline.idle_read_fd, .events = posix.POLL.IN, .revents = undefined },
 1549         };
 1550 
 1551         while (true) {
 1552             // Claim the next free buffer. This blocks only when the
 1553             // parse stage is a full ring behind, which is exactly when
 1554             // we should stop reading and let the kernel queue exert
 1555             // backpressure on the child.
 1556             const buf: *[buffer_capacity]u8 = buf: {
 1557                 pipeline.mutex.lockUncancelable(global.io());
 1558                 defer pipeline.mutex.unlock(global.io());
 1559                 while (pipeline.count == buffer_count) {
 1560                     pipeline.slot_free.waitUncancelable(global.io(), &pipeline.mutex);
 1561                 }
 1562                 break :buf &pipeline.bufs[pipeline.head];
 1563             };
 1564 
 1565             var total: usize = 0;
 1566             var bridge_start: ?std.Io.Timestamp = null;
 1567             var spins: usize = 0;
 1568             var fatal = false;
 1569 
 1570             // Fill the buffer from the pty. For a saturated stream the
 1571             // kernel queue momentarily runs dry while the writer
 1572             // refills it in parallel, so we bridge those gaps with
 1573             // spin retries and a short poll instead of delivering a
 1574             // tiny batch.
 1575             gather: while (total < buffer_capacity) {
 1576                 const n = posix.read(
 1577                     fd,
 1578                     buf[total..],
 1579                 ) catch |err| switch (err) {
 1580                     error.WouldBlock => {
 1581                         // Anything below the threshold is interactive.
 1582                         if (total < bridge_threshold) break :gather;
 1583 
 1584                         // The stream is saturated, so we bridge the
 1585                         // gap. First retry the read directly a bounded
 1586                         // number of times, since the refill usually
 1587                         // lands within microseconds.
 1588                         if (spins < bridge_spin_max) {
 1589                             spins += 1;
 1590                             continue :gather;
 1591                         }
 1592 
 1593                         // Still dry, so we want to sleep in poll for
 1594                         // the next refill, within our latency budget.
 1595                         const now = std.Io.Timestamp.now(global.io(), .awake);
 1596                         if (bridge_start) |start| {
 1597                             if (start.durationTo(now).toNanoseconds() >= gather_budget_ns)
 1598                                 break :gather;
 1599                         } else bridge_start = now;
 1600 
 1601                         // Bridging a refill gap is only free while the
 1602                         // parse stage is busy, since the wait hides
 1603                         // behind parse time. Once the parser is idle,
 1604                         // every microsecond we hold this batch is
 1605                         // added straight to output latency, and for a
 1606                         // request/response producer (a burst ending
 1607                         // in a query, e.g. frame + cursor position
 1608                         // report) the writer is blocked on a reply to
 1609                         // data sitting in this buffer, so a poll here
 1610                         // would always sleep its full timeout. Deliver
 1611                         // now if the parser is idle, and otherwise arm
 1612                         // the idle wake so it can interrupt our poll
 1613                         // the moment that changes.
 1614                         {
 1615                             pipeline.mutex.lockUncancelable(global.io());
 1616                             defer pipeline.mutex.unlock(global.io());
 1617                             if (pipeline.count == 0) break :gather;
 1618                             pipeline.bridging = true;
 1619                         }
 1620 
 1621                         const r = posix.poll(
 1622                             &pollfds,
 1623                             bridge_poll_timeout_ms,
 1624                         ) catch |poll_err| {
 1625                             clearBridging(pipeline);
 1626                             log.warn("bridge poll failed err={}", .{poll_err});
 1627                             break :gather;
 1628                         };
 1629                         clearBridging(pipeline);
 1630 
 1631                         // Quiet for a full timeout means the burst
 1632                         // ended.
 1633                         if (r == 0) break :gather;
 1634 
 1635                         // On a quit signal we deliver what we have
 1636                         // and stop.
 1637                         if (pollfds[1].revents & posix.POLL.IN != 0) {
 1638                             log.info("read thread got quit signal", .{});
 1639                             fatal = true;
 1640                             break :gather;
 1641                         }
 1642 
 1643                         // The parse stage went idle: drain the wake
 1644                         // and deliver what we have. The pty may have
 1645                         // data as well, but the next batch can pick
 1646                         // that up; an idle parser means delivery must
 1647                         // not wait.
 1648                         if (pollfds[2].revents & posix.POLL.IN != 0) {
 1649                             var trash: [16]u8 = undefined;
 1650                             while (true) {
 1651                                 const drained = posix.read(
 1652                                     pipeline.idle_read_fd,
 1653                                     &trash,
 1654                                 ) catch break;
 1655                                 if (drained < trash.len) break;
 1656                             }
 1657                             break :gather;
 1658                         }
 1659 
 1660                         // HUP without IN means no more data is
 1661                         // coming. Deliver and let the outer poll
 1662                         // decide what to do.
 1663                         if (pollfds[0].revents & posix.POLL.IN == 0)
 1664                             break :gather;
 1665 
 1666                         continue :gather;
 1667                     },
 1668 
 1669                     // The pty is closed. We're probably gracefully
 1670                     // shutting down.
 1671                     error.NotOpenForReading,
 1672                     error.InputOutput,
 1673                     => {
 1674                         log.info("io gather exiting", .{});
 1675                         fatal = true;
 1676                         break :gather;
 1677                     },
 1678 
 1679                     else => {
 1680                         log.err("io gather error err={}", .{err});
 1681                         unreachable;
 1682                     },
 1683                 };
 1684 
 1685                 // This happens on macOS instead of WouldBlock when the
 1686                 // child process dies. Deliver what we have and let the
 1687                 // outer poll detect HUP.
 1688                 if (n == 0) break :gather;
 1689 
 1690                 total += n;
 1691 
 1692                 // Each refill gap gets a fresh spin budget.
 1693                 spins = 0;
 1694             }
 1695 
 1696             // Publish the batch (if any) to the parse stage and rotate
 1697             // to the next buffer.
 1698             if (total > 0) {
 1699                 pipeline.mutex.lockUncancelable(global.io());
 1700                 pipeline.lens[pipeline.head] = total;
 1701                 pipeline.head = (pipeline.head + 1) % buffer_count;
 1702                 pipeline.count += 1;
 1703                 pipeline.mutex.unlock(global.io());
 1704                 pipeline.batch_ready.signal(global.io());
 1705             }
 1706 
 1707             if (fatal) return;
 1708 
 1709             // A full buffer means the stream is still hot, so go
 1710             // claim the next buffer without an intervening poll.
 1711             if (total == buffer_capacity) continue;
 1712 
 1713             // Wait for data. The idle fd is sliced off: the parse
 1714             // stage only writes to it while we're bridging.
 1715             _ = posix.poll(pollfds[0..2], -1) catch |err| {
 1716                 log.warn("poll failed on read thread, exiting early err={}", .{err});
 1717                 return;
 1718             };
 1719 
 1720             // If our quit fd is set, we're done.
 1721             if (pollfds[1].revents & posix.POLL.IN != 0) {
 1722                 log.info("read thread got quit signal", .{});
 1723                 return;
 1724             }
 1725 
 1726             // If our pty fd is closed, then we're also done with our
 1727             // read thread.
 1728             if (pollfds[0].revents & posix.POLL.HUP != 0) {
 1729                 log.info("pty fd closed, read thread exiting", .{});
 1730                 return;
 1731             }
 1732         }
 1733     }

  713 /// Process output from readdata but the lock is already held.
  714 fn processOutputLocked(self: *Termio, buf: []const u8) void {

  716     self.terminal_stream.handler.queueRender() catch unreachable;

  718     // Whenever a character is typed, we ensure the cursor is in the
  719     // non-blink state so it is rendered if visible. If we're under
  720     // HEAVY read load, we don't want to send a ton of these so we
  721     // use a timer under the covers
  722     const now = std.Io.Timestamp.now(global.io(), .awake);
  723     cursor_reset: {
  724         if (self.last_cursor_reset) |last| {
  725             if (last.durationTo(now).toMilliseconds() <= 500) {
  726                 break :cursor_reset;
  727             }
  728         }
  729 
  730         self.last_cursor_reset = now;
  731         _ = self.renderer_mailbox.push(global.io(), .{
  732             .reset_cursor_blink = {},
  733         }, .{ .instant = {} });
  734     }
  735 
  736     // If we have an inspector, we enter SLOW MODE because we need to
  737     // process a byte at a time alternating between the inspector handler
  738     // and the termio handler. This is very slow compared to our optimizations
  739     // below but at least users only pay for it if they're using the inspector.
  740     if (self.renderer_state.inspector) |insp| {
  741         for (buf, 0..) |byte, i| {
  742             insp.recordPtyRead(
  743                 self.alloc,
  744                 &self.terminal,
  745                 buf[i .. i + 1],
  746             ) catch |err| {
  747                 log.err("error recording pty read in inspector err={}", .{err});
  748             };
  749 
  750             self.terminal_stream.next(byte);
  751         }
  752     } else {
  753         self.terminal_stream.nextSlice(buf);
  754     }
  755 
  756     // If our stream handling caused messages to be sent to the mailbox
  757     // thread, then we need to wake it up so that it processes them.
  758     if (self.terminal_stream.handler.termio_messaged) {
  759         self.terminal_stream.handler.termio_messaged = false;
  760         self.mailbox.notify();
  761     }

  209                     \\
  210                     \\This terminal is non-functional. Please close it and try again.
  211                 ,
  212                     .{err},
  213                 ) catch
  214                     \\Out of memory. This terminal is non-functional. Please close it and try again.
  215                 ;
  216 
  217                 t.eraseDisplay(.complete, false);
  218                 t.printString(str) catch {};
  219             },
  220         }

  224     // messages are drained and we can wait for the surface to send a stop
  225     // message.
  226     if (!self.loop.stopped()) {

  236 fn threadMain_(self: *Thread, io: *termio.Termio) !void {
  237     defer log.debug("IO thread exited", .{});

  239     // Right now, on Darwin, `std.Thread.setName` can only name the current
  240     // thread, and we have no way to get the current thread from within it,

  246     // Setup our crash metadata

  248         .type = .io,
  249         .surface = io.surface_mailbox.surface,
  250     };
  251     defer crash.sentry.thread_state = null;

  253     // Get the mailbox. This must be an SPSC mailbox for threading.
  254     const mailbox = switch (io.mailbox) {
  255         .spsc => |*v| v,
  256         // else => return error.TermioUnsupportedMailbox,
  257     };

  259     // This is the data sent to xev callbacks. We want a pointer to both
  260     // ourselves and the thread data so we can thread that through (pun intended).
  261     var cb: CallbackData = .{ .self = self, .io = io };
  262 
  263     // Run our thread start/end callbacks. This allows the implementation
  264     // to hook into the event loop as needed. The thread data is created
  265     // on the stack here so that it has a stable pointer throughout the
  266     // lifetime of the thread.
  267     try io.threadEnter(self, &cb.data);
  268     defer cb.data.deinit();
  269     defer io.threadExit(&cb.data);
  270 
  271     // Start the async handlers.
  272     mailbox.wakeup.wait(&self.loop, &self.wakeup_c, CallbackData, &cb, wakeupCallback);
  273     self.stop.wait(&self.loop, &self.stop_c, CallbackData, &cb, stopCallback);
  274 
  275     // Run
  276     log.debug("starting IO thread", .{});
  277     defer log.debug("starting IO thread shutdown", .{});
  278     try self.loop.run(.until_done);
  279 }

  288 /// Drain the mailbox, handling all the messages in our terminal implementation.
  289 fn drainMailbox(
  290     self: *Thread,
  291     cb: *CallbackData,
  292 ) !void {
  293     // We assert when starting the thread that this is the state
  294     const mailbox = cb.io.mailbox.spsc.queue;
  295     const io = cb.io;
  296     const data = &cb.data;
  297 
  298     // If we're draining, we just drain the mailbox and return.
  299     if (self.flags.drain) {
  300         while (mailbox.pop(global.io())) |msg| msg.deinit();
  301         return;
  302     }
  303 
  304     // This holds the mailbox lock for the duration of the drain. The
  305     // expectation is that all our message handlers will be non-blocking
  306     // ENOUGH to not mess up throughput on producers.
  307     var redraw: bool = false;
  308     while (mailbox.pop(global.io())) |message| {
  309         // If we have a message we always redraw
  310         redraw = true;
  311 
  312         log.debug("mailbox message={s}", .{@tagName(message)});
  313         switch (message) {
  314             .color_scheme_report => |v| try io.colorSchemeReport(data, v.force),
  315             .visibility_report => |v| try io.visibilityReport(
  316                 data,
  317                 v.visible,
  318                 v.force,
  319             ),
  320             .crash => @panic("crash request, crashing intentionally"),
  321             .change_config => |config| {
  322                 defer config.alloc.destroy(config.ptr);
  323                 try io.changeConfig(data, config.ptr);
  324             },
  325             .inspector => |v| self.flags.has_inspector = v,
  326             .resize => |v| self.handleResize(cb, v),
  327             .size_report => |v| try io.sizeReport(data, v),
  328             .clear_screen => |v| try io.clearScreen(data, v.history),
  329             .scroll_viewport => |v| io.scrollViewport(v),
  330             .selection_scroll => |v| {
  331                 if (v) {
  332                     self.startScrollTimer(cb);
  333                 } else {
  334                     self.stopScrollTimer();
  335                 }
  336             },
  337             .jump_to_prompt => |v| try io.jumpToPrompt(v),
  338             .kitty_clipboard_grant_read => |v| {
  339                 defer v.alloc.free(v.pw);
  340                 try io.kittyClipboardGrant(v.pw, .read);
  341             },
  342             .kitty_clipboard_grant_write => |v| {
  343                 defer v.alloc.free(v.pw);
  344                 try io.kittyClipboardGrant(v.pw, .write);
  345             },
  346             .start_synchronized_output => self.startSynchronizedOutput(cb),
  347             .linefeed_mode => |v| self.flags.linefeed_mode = v,
  348             .focused => |v| try io.focusGained(data, v),
  349             .write_small => |v| try io.queueWrite(
  350                 data,
  351                 v.data[0..v.len],
  352                 self.flags.linefeed_mode,
  353             ),
  354             .write_stable => |v| try io.queueWrite(
  355                 data,
  356                 v,
  357                 self.flags.linefeed_mode,
  358             ),
  359             .write_alloc => |v| {
  360                 defer v.alloc.free(v.data);
  361                 try io.queueWrite(
  362                     data,
  363                     v.data,
  364                     self.flags.linefeed_mode,
  365                 );
  366             },
  367         }
  368     }
  369 
  370     // Trigger a redraw after we've drained so we don't waste cyces
  371     // messaging a redraw.
  372     if (redraw) {
  373         try io.renderer_wakeup.notify();
  374     }
  375 }

  427 fn coalesceCallback(
  428     cb_: ?*CallbackData,
  429     _: *xev.Loop,
  430     _: *xev.Completion,
  431     r: xev.Timer.RunError!void,
  432 ) xev.CallbackAction {
  433     _ = r catch |err| switch (err) {
  434         error.Canceled => {},
  435         else => {
  436             log.warn("error during coalesce callback err={}", .{err});
  437             return .disarm;
  438         },
  439     };
  440 
  441     const cb = cb_ orelse return .disarm;
  442 
  443     if (cb.self.coalesce_data.resize) |v| {
  444         cb.self.coalesce_data.resize = null;
  445         cb.io.resize(&cb.data, v) catch |err| {
  446             log.warn("error during resize err={}", .{err});
  447         };
  448     }
  449 
  450     return .disarm;
  451 }

  453 fn wakeupCallback(
  454     cb_: ?*CallbackData,
  455     _: *xev.Loop,
  456     _: *xev.Completion,
  457     r: xev.Async.WaitError!void,
  458 ) xev.CallbackAction {
  459     _ = r catch |err| {
  460         log.err("error in wakeup err={}", .{err});
  461         return .rearm;
  462     };
  463 
  464     // When we wake up, we check the mailbox. Mailbox producers should
  465     // wake up our thread after publishing.
  466     const cb = cb_ orelse return .rearm;
  467     cb.self.drainMailbox(cb) catch |err|
  468         log.err("error draining mailbox err={}", .{err});
  469 
  470     return .rearm;
  471 }

   11 /// A queue used for storing messages that is periodically drained.
   12 /// Typically used by a multi-threaded application. The capacity is
   13 /// hardcoded to a value that empirically has made sense for Ghostty usage
   14 /// but I'm open to changing it with good arguments.
   15 const Queue = BlockingQueue(termio.Message, 64);
   16 
   17 /// The location to where write-related messages are sent.
   18 pub const Mailbox = union(enum) {
   19     // /// Write messages to an unbounded list backed by an allocator.
   20     // /// This is useful for single-threaded applications where you're not
   21     // /// afraid of running out of memory. You should be careful that you're
   22     // /// processing this in a timely manner though since some heavy workloads
   23     // /// will produce a LOT of messages.
   24     // ///
   25     // /// At the time of authoring this, the primary use case for this is
   26     // /// testing more than anything, but it probably will have a use case
   27     // /// in libghostty eventually.
   28     // unbounded: std.ArrayList(termio.Message),
   29 
   30     /// Write messages to a SPSC queue for multi-threaded applications.
   31     spsc: struct {
   32         queue: *Queue,
   33         wakeup: xev.Async,
   34     },
   35 
   36     /// Init the SPSC writer.
   37     pub fn initSPSC(alloc: Allocator) !Mailbox {
   38         var queue = try Queue.create(alloc);
   39         errdefer queue.destroy(alloc);
   40 
   41         var wakeup = try xev.Async.init();
   42         errdefer wakeup.deinit();
   43 
   44         return .{ .spsc = .{ .queue = queue, .wakeup = wakeup } };
   45     }
   46 
   47     pub fn deinit(self: *Mailbox, alloc: Allocator) void {
   48         switch (self.*) {
   49             .spsc => |*v| {
   50                 while (v.queue.pop(global.io())) |msg| msg.deinit();
   51                 v.queue.destroy(alloc);
   52                 v.wakeup.deinit();
   53             },
   54         }
   55     }
   56 
   57     /// Sends the given message without notifying there are messages.
   58     ///
   59     /// If the optional mutex is given, it must already be LOCKED. If the
   60     /// send would block, we'll unlock this mutex, resend the message, and
   61     /// lock it again. This handles an edge case where queues are full.
   62     /// This may not apply to all writer types.
   63     pub fn send(
   64         self: *Mailbox,
   65         msg: termio.Message,
   66         mutex: ?*std.Io.Mutex,
   67     ) void {
   68         switch (self.*) {
   69             .spsc => |*mb| send: {
   70                 // Try to write to the queue with an instant timeout. This is the
   71                 // fast path because we can queue without a lock.
   72                 if (mb.queue.push(global.io(), msg, .{ .instant = {} }) > 0) break :send;
   73 
   74                 // If we enter this conditional, the queue is full. We wake up
   75                 // the writer thread so that it can process messages to clear up
   76                 // space. However, the writer thread may require the renderer
   77                 // lock so we need to unlock.
   78                 mb.wakeup.notify() catch |err| {
   79                     log.warn("failed to wake up writer, data will be dropped err={}", .{err});
   80                     msg.deinit();
   81                     return;
   82                 };
   83 
   84                 // Unlock the renderer state so the writer thread can acquire it.
   85                 // Then try to queue our message before continuing. This is a very
   86                 // slow path because we are having a lot of contention for data.
   87                 // But this only gets triggered in certain pathological cases.
   88                 //
   89                 // Note that writes themselves don't require a lock, but there
   90                 // are other messages in the writer queue (resize, focus) that
   91                 // could acquire the lock. This is why we have to release our lock
   92                 // here.
   93                 if (mutex) |m| m.unlock(global.io());
   94                 defer if (mutex) |m| m.lockUncancelable(global.io());
   95                 if (mb.queue.push(global.io(), msg, .{ .forever = {} }) == 0) msg.deinit();
   96             },
   97         }
   98     }
   99 
  100     /// Notify that there are new messages. This may be a noop depending
  101     /// on the writer type.
  102     pub fn notify(self: *Mailbox) void {
  103         switch (self.*) {
  104             .spsc => |*v| v.wakeup.notify() catch |err| {
  105                 log.warn("failed to notify writer, data will be dropped err={}", .{err});
  106             },
  107         }
  108     }

  208     inline fn vtFallible(
  209         self: *StreamHandler,
  210         comptime action: Stream.Action.Tag,
  211         value: Stream.Action.Value(action),
  212     ) !void {
  213         // The branch hints here are based on real world data
  214         // which indicates that the most common actions are:
  215         //
  216         // 1. print
  217         // 2. set_attributes
  218         // 3. carriage_return
  219         // 4. line_feed
  220         // 5. cursor_pos
  221         //
  222         // Together, these 5 actions make up nearly 98% of
  223         // all actions encountered in real world scenarios.
  224         //
  225         // ref: https://github.com/qwerasd205/asciinema-stats
  226         switch (action) {
  227             .print => {
  228                 @branchHint(.likely);
  229                 try self.terminal.print(value.cp);
  230             },
  231             .print_slice => {
  232                 @branchHint(.likely);
  233                 try self.terminal.printSlice(value.cps);
  234             },
