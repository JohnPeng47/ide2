  149     fn snapshot(self: *Self, snap: *gtk.Snapshot) callconv(.c) void {
  150         const priv = self.private();
  151 
  152         // Take the latest present from the renderer's queue, if any, and
  153         // build a texture from it. `take` closes any present that was
  154         // still unconsumed in the queue, but not the texture we're
  155         // currently displaying.
  156         if (priv.core_surface) |core| {
  157             if (core.renderer.takeFrame()) |frame| {
  158                 self.rebuildTexture(frame) catch |err| switch (err) {
  159                     // GTK can't import our DMABUFs, e.g. due to an
  160                     // incompatible format modifier on multi-GPU setups.
  161                     // Report unhealthy presentation health so the
  162                     // renderer presents via CPU readback instead.
  163                     error.DmabufBuildFailed => {
  164                         log.warn(
  165                             "failed to import dmabuf, asking renderer for CPU presentation",
  166                             .{},
  167                         );
  168                         core.reportPresentationHealth(.unhealthy);
  169                     },
  170 
  171                     else => log.warn("error building texture from frame err={}", .{err}),
  172                 };
  173             }
  174         }
  175 
  176         // If we have a texture, draw it.
  177         const texture = priv.texture orelse return;
  178 
  179         const widget = self.as(gtk.Widget);
  180         if (widget.getWidth() == 0 or widget.getHeight() == 0) return;
  181 
  182         // Map one texture texel to one device pixel. Widget CSS size would
  183         // stretch a fractionally scaled buffer. Snap the surface-relative
  184         // origin so parent chrome cannot place the texture between pixels.
  185         const layout = scale_util.widgetLayout(widget);
  186         const origin = layout.snapOrigin();
  187         const scale: f32 = @floatCast(layout.scale);
  188         const css_w: f32 = @as(f32, @floatFromInt(texture.getWidth())) / scale;
  189         const css_h: f32 = @as(f32, @floatFromInt(texture.getHeight())) / scale;
  190 
  191         // GTK, Metal, Vulkan, etc. all have their origin points in the
  192         // top left, but not OpenGL!
  193         const flip = !rendererpkg.Renderer.custom_shader_y_is_down;
  194         if (flip) snap.save();
  195         defer if (flip) snap.restore();
  196         if (flip) {
  197             snap.translate(&.{ .f_x = 0, .f_y = css_h });
  198             snap.scale(1, -1);
  199         }
  200 
  201         snap.appendTexture(texture, &.{
  202             .f_origin = .{ .f_x = @floatCast(origin.x), .f_y = @floatCast(origin.y) },
  203             .f_size = .{ .f_width = css_w, .f_height = css_h },
  204         });
  205     }

  314     /// Build a `GdkDmabufTexture` from the present and set it as our
  315     /// current texture, unrefing any previous texture.
  316     fn rebuildDmabufTexture(self: *Self, frame: Dmabuf) !void {
  317         const priv = self.private();
  318         const widget = self.as(gtk.Widget);
  319         const display = widget.getDisplay();
  320 
  321         const builder = gdk.DmabufTextureBuilder.new();
  322         defer builder.unref();
  323 
  324         builder.setDisplay(display);
  325         builder.setWidth(frame.width);
  326         builder.setHeight(frame.height);
  327         builder.setFourcc(frame.fourcc);
  328         builder.setModifier(frame.modifier);
  329         builder.setPremultiplied(@intFromBool(frame.premultiplied));
  330         builder.setNPlanes(@intCast(frame.planes.count));
  331 
  332         for (0..frame.planes.count) |i| {
  333             builder.setFd(@intCast(i), frame.planes.fds[i]);
  334             builder.setOffset(@intCast(i), @intCast(frame.planes.offsets[i]));
  335             builder.setStride(@intCast(i), @intCast(frame.planes.strides[i]));
  336         }
  337 
  338         // Build the texture. We retain ownership over the DMABUF FDs
  339         // until the asynchronous texture building process is complete,
  340         // so we have to make a heap copy of them, send them to GTK,
  341         // and then destroy them once successful.
  342         const app = Application.default();
  343         const alloc = app.allocator();
  344 
  345         const planes = try alloc.create(Planes);
  346         errdefer alloc.destroy(planes);
  347 
  348         planes.* = frame.planes;
  349         errdefer planes.deinit();
  350 
  351         var err_: ?*glib.Error = null;
  352         const texture = builder.build(
  353             dmabufDestroy,
  354             planes,
  355             &err_,
  356         ) orelse {
  357             if (err_) |err|
  358                 log.warn("failed to build dmabuf err={s}", .{err.f_message orelse "(unknown)"});
  359             return error.DmabufBuildFailed;
  360         };
  361 
  362         // Swap out the old texture.
  363         if (priv.texture) |old| old.as(gobject.Object).unref();
  364         priv.texture = texture;
  365     }
  390     fn rebuildMemoryTexture(self: *Self, frame: ExportedFrame.Memory) !void {
  391         const priv = self.private();
  392 
  393         const app = Application.default();
  394         const alloc = app.allocator();
  395 
  396         const holder = alloc.create(MemoryHolder) catch |err| {
  397             // We own the pixels; make sure they don't leak.
  398             frame.deinit();
  399             return err;
  400         };
  401         errdefer alloc.destroy(holder);
  402         holder.* = .{ .alloc = alloc, .pixels = frame.pixels };
  403 
  404         // `GdkMemoryTexture` keeps a reference to the `GBytes` for as
  405         // long as it needs the pixel data, so we hand it our holder and
  406         // get the pixels freed via the destroy notify below.
  407         const bytes = glib.Bytes.newWithFreeFunc(
  408             holder.pixels.ptr,
  409             holder.pixels.len,
  410             &memoryDestroy,
  411             holder,
  412         );
  413         errdefer bytes.unref();
  414 
  415         const texture = gdk.MemoryTexture.new(
  416             @intCast(frame.width),
  417             @intCast(frame.height),
  418             .r8g8b8a8_premultiplied,
  419             bytes,
  420             frame.width * 4, // stride
  421         );
  422 
  423         // The texture holds its own reference to the bytes now.
  424         bytes.unref();
  425 
  426         // Swap out the old texture.
  427         if (priv.texture) |old| old.as(gobject.Object).unref();
  428         priv.texture = texture.as(gdk.Texture);
  429     }

  136 pub fn reserve(
  137     self: *Atlas,
  138     alloc: Allocator,
  139     width: u32,
  140     height: u32,
  141 ) (Allocator.Error || Error)!Region {
  142     const tw = reserve_tw;
  143 
  144     // x, y are populated within :best_idx below
  145     var region: Region = .{ .x = 0, .y = 0, .width = width, .height = height };
  146 
  147     // If our width/height are 0, then we return the region as-is. This
  148     // may seem like an error case but it simplifies downstream callers who
  149     // might be trying to write empty data.
  150     if (width == 0 and height == 0) return region;
  151 
  152     // Find the location in our nodes list to insert the new node for this region.
  153     const best_idx: usize = best_idx: {
  154         var best_height: u32 = std.math.maxInt(u32);
  155         var best_width: u32 = best_height;
  156         var chosen: ?usize = null;
  157 
  158         var i: usize = 0;
  159         while (i < self.nodes.items.len) : (i += 1) {
  160             // Check if our region fits within this node.
  161             const y = self.fit(i, width, height) orelse continue;
  162 
  163             const node = self.nodes.items[i];
  164             if ((y + height) < best_height or
  165                 ((y + height) == best_height and
  166                     (node.width > 0 and node.width < best_width)))
  167             {
  168                 chosen = i;
  169                 best_width = node.width;
  170                 best_height = y + height;
  171                 region.x = node.x;
  172                 region.y = y;
  173             }
  174         }
  175 
  176         // If we never found a chosen index, the atlas cannot fit our region.
  177         break :best_idx chosen orelse return Error.AtlasFull;
  178     };
  179 
  180     // Insert our new node for this rectangle at the exact best index
  181     try tw.check(.insert_node);
  182     try self.nodes.insert(alloc, best_idx, .{
  183         .x = region.x,
  184         .y = region.y + height,
  185         .width = width,
  186     });
  187     errdefer comptime unreachable;
  188 
  189     // Optimize our rectangles
  190     var i: usize = best_idx + 1;
  191     while (i < self.nodes.items.len) : (i += 1) {
  192         const node = &self.nodes.items[i];
  193         const prev = self.nodes.items[i - 1];
  194         if (node.x < (prev.x + prev.width)) {
  195             const shrink = prev.x + prev.width - node.x;
  196             node.x += shrink;
  197             node.width -|= shrink;
  198             if (node.width <= 0) {
  199                 _ = self.nodes.orderedRemove(i);
  200                 i -= 1;
  201                 continue;
  202             }
  203         }
  204 
  205         break;
  206     }
  207     self.merge();
  208 
  209     return region;
  210 }
  256 pub fn set(self: *Atlas, reg: Region, data: []const u8) void {
  257     assert(reg.x < (self.size - 1));
  258     assert((reg.x + reg.width) <= (self.size - 1));
  259     assert(reg.y < (self.size - 1));
  260     assert((reg.y + reg.height) <= (self.size - 1));
  261 
  262     const depth = self.format.depth();
  263     var i: u32 = 0;
  264     while (i < reg.height) : (i += 1) {
  265         const tex_offset = (((reg.y + i) * self.size) + reg.x) * depth;
  266         const data_offset = i * reg.width * depth;
  267         fastmem.copy(
  268             u8,
  269             self.data[tex_offset..],
  270             data[data_offset .. data_offset + (reg.width * depth)],
  271         );
  272     }
  273 
  274     _ = self.modified.fetchAdd(1, .monotonic);
  275 }

  389

  394         defer self.lock.unlockShared(global.io());

  400     defer self.lock.unlock(global.io());

  424             .align_horizontal = .center,
  425             .align_vertical = .center,
  426 
  427             // Add a small bit of padding so the emoji
  428             // doesn't quite touch the edges of the cells.
  429             .pad_left = 0.025,

  431         };
  432     }

   47     pub fn next(self: *RunIterator, alloc: Allocator) !?TextRun {
   48         const slice = &self.opts.cells;
   49         const cells: []const terminal.page.Cell = slice.items(.raw);
   50         const graphemes: []const []const u21 = slice.items(.grapheme);
   51         const styles: []const terminal.Style = slice.items(.style);
   52 
   53         // Trim the right side of a row that might be empty
   54         const max: usize = max: {
   55             for (0..cells.len) |i| {
   56                 const rev_i = cells.len - i - 1;
   57                 if (!cells[rev_i].isEmpty()) break :max rev_i + 1;
   58             }
   59 
   60             break :max 0;
   61         };
   62 
   63         // Invisible cells don't have any glyphs rendered,
   64         // so we explicitly skip them in the shaping process.
   65         while (self.i < max and
   66             (cells[self.i].hasStyling() and
   67                 styles[self.i].flags.invisible)) self.i += 1;
   68 
   69         // We're over at the max
   70         if (self.i >= max) return null;
   71 
   72         // Track the font for our current run
   73         var current_font: font.Collection.Index = .{};
   74 
   75         // Allow the hook to prepare
  211             // We need to find a font that supports this character. If
  212             // there are additional zero-width codepoints (to form a single
  213             // grapheme, i.e. combining characters), we need to find a font
  214             // that supports all of them.
  215             const font_info: struct {
  216                 idx: font.Collection.Index,
  217                 fallback: ?u32 = null,
  218             } = font_info: {
  219                 // If we find a font that supports this entire grapheme
  220                 // then we use that.
  221                 if (try self.indexForCell(
  222                     alloc,
  223                     cell,
  224                     graphemes[j],
  225                     font_style,
  226                     presentation,
  227                 )) |idx| break :font_info .{ .idx = idx };
  228 
  229                 // Otherwise we need a fallback character. Prefer the
  230                 // official replacement character.
  231                 if (try self.opts.grid.getIndex(
  232                     alloc,
  233                     0xFFFD, // replacement char
  234                     font_style,
  235                     presentation,
  236                 )) |idx| break :font_info .{ .idx = idx, .fallback = 0xFFFD };
  237 
  238                 // Fallback to space
  239                 if (try self.opts.grid.getIndex(
  240                     alloc,
  241                     ' ',
  242                     font_style,
  243                     presentation,
  244                 )) |idx| break :font_info .{ .idx = idx, .fallback = ' ' };
  245 
  246                 // We can't render at all. This is a bug, we should always
  247                 // have a font that can render a space.
  248                 unreachable;
  249             };
  250 
  251             //log.warn("char={x} info={}", .{ cell.char, font_info });
  252             if (j == self.i) current_font = font_info.idx;
  253 
  254             // If our fonts are not equal, then we're done with our run.
  255             if (font_info.idx != current_font) break;
  256 
  257             // If we're a fallback character, add that and continue; we
  258             // don't want to add the entire grapheme.
  259             if (font_info.fallback) |cp| {
  260                 try self.addCodepoint(&hasher, cp, @intCast(cluster));
  261                 continue;
  262             }
  263 
  264             // If we're a Kitty unicode placeholder then we add a blank.
  265             if (cell.codepoint() == terminal.kitty.graphics.unicode.placeholder) {
  266                 try self.addCodepoint(&hasher, ' ', @intCast(cluster));
  267                 continue;
  268             }
  269 
  270             // Add all the codepoints for our grapheme
  271             try self.addCodepoint(
  272                 &hasher,
  273                 if (cell.codepoint() == 0) ' ' else cell.codepoint(),
  274                 @intCast(cluster),
  275             );
  276             if (cell.hasGrapheme()) {
  277                 for (graphemes[j]) |cp| {
  278                     // Do not send presentation modifiers
  279                     if (cp == 0xFE0E or cp == 0xFE0F) continue;
  280                     try self.addCodepoint(&hasher, cp, @intCast(cluster));
  281                 }
  282             }
  283         }
  284 
  285         // Finalize our buffer
  286         self.hooks.finalize();
  287 
  288         // Add our length to the hash as an additional mechanism to avoid collisions
  289         autoHash(&hasher, j - self.i);
  290 
  291         // Add our font index
  292         autoHash(&hasher, current_font);
  293 
  294         // Move our cursor. Must defer since we use self.i below.
  295         defer self.i = j;
  296 
  297         return .{
  298             .hash = hasher.final(),
  299             .offset = @intCast(self.i),
  300             .cells = @intCast(j - self.i),
  301             .grid = self.opts.grid,
  302             .font_index = current_font,
  303         };
  304     }

  120 pub const Preedit = struct {
  121     /// The codepoints to render as preedit text.
  122     codepoints: []const Codepoint = &.{},
  123 
  124     /// A single codepoint to render as preedit text.
  125     pub const Codepoint = struct {
  126         codepoint: u21,
  127         wide: bool = false,
  128     };
  135     /// Allocate a copy of this preedit in the given allocator..
  136     pub fn clone(self: *const Preedit, alloc: Allocator) !Preedit {
  137         return .{
  138             .codepoints = try alloc.dupe(Codepoint, self.codepoints),
  139         };
  140     }
  152     /// Range returns the start and end x position of the preedit text
  153     /// along with any codepoint offset necessary to fit the preedit
  154     /// into the available space.
  155     pub fn range(
  156         self: *const Preedit,
  157         start: terminalpkg.size.CellCountInt,
  158         max: terminalpkg.size.CellCountInt,
  159     ) struct {
  160         start: terminalpkg.size.CellCountInt,
  161         end: terminalpkg.size.CellCountInt,
  162         cp_offset: usize,
  163     } {
  164         // If our width is greater than the number of cells we have
  165         // then we need to adjust our codepoint start to a point where
  166         // our width would be less than the number of cells we have.
  167         const w, const cp_offset = width: {
  168             // max is inclusive, so we need to add 1 to it.
  169             const max_width = max - start + 1;
  170 
  171             // Rebuild our width in reverse order. This is because we want
  172             // to offset by the end cells, not the start cells (if we have to).
  173             var w: terminalpkg.size.CellCountInt = 0;
  174             for (0..self.codepoints.len) |i| {
  175                 const reverse_i = self.codepoints.len - i - 1;
  176                 const cp = self.codepoints[reverse_i];
  177                 w += if (cp.wide) 2 else 1;
  178                 if (w > max_width) {
  179                     break :width .{ w, reverse_i };
  180                 }
  181             }
  182 
  183             // Width fit in the max width so no offset necessary.
  184             break :width .{ w, 0 };
  185         };
  186 
  187         // If our preedit goes off the end of the screen, we adjust it so
  188         // that it shifts left.
  189         const end = if (w > 0) start + (w - 1) else start;
  190         const start_offset = if (end > max) end - max else 0;
  191         return .{
  192             .start = start -| start_offset,
  193             .end = end -| start_offset,
  194             .cp_offset = cp_offset,
  195         };
  196     }

  200 fn threadMain_(self: *Thread) !void {
  220     // Run our loop start/end callbacks if the renderer cares.
  221     const has_loop = @hasDecl(rendererpkg.Renderer, "loopEnter");
  222     if (has_loop) try self.renderer.loopEnter(self);
  223     defer if (has_loop) self.renderer.loopExit();
  224 
  225     // Run our thread start/end callbacks. This is important because some
  226     // renderers have to do per-thread setup. For example, OpenGL has to set
  227     // some thread-local state since that is how it works.
  228     try self.renderer.threadEnter(self.surface);
  229     defer self.renderer.threadExit();
  230 
  231     // Start the async handlers
  232     self.wakeup.wait(&self.loop, &self.wakeup_c, Thread, self, wakeupCallback);
  233     self.stop.wait(&self.loop, &self.stop_c, Thread, self, stopCallback);
  234     self.draw_now.wait(&self.loop, &self.draw_now_c, Thread, self, drawNowCallback);
  235 
  236     // Send an initial wakeup message so that we render right away.
  237     try self.wakeup.notify();
  238 
  239     // Start blinking the cursor.
  240     self.cursor_h.run(
  241         &self.loop,
  242         &self.cursor_c,
  243         cursorBlinkInterval(),
  244         Thread,
  245         self,
  246         cursorTimerCallback,
  247     );
  248 
  249     // Arm the animation timer in case the renderer already needs
  250     // animation wakes (e.g. custom shaders loaded at startup).
  251     self.armAnimationTimer();
  252 
  253     // Run
  254     log.debug("starting renderer thread", .{});
  255     defer log.debug("starting renderer thread shutdown", .{});
  256     _ = try self.loop.run(.until_done);
  257 }
  290 fn drainMailbox(self: *Thread) !void {
  301     while (self.mailbox.pop(global.io())) |message| {
  302         log.debug("mailbox message={}", .{message});
  303         switch (message) {
  304             .crash => @panic("crash request, crashing intentionally"),
  305 
  306             .visible => |v| visible: {
  307                 // If our state didn't change we do nothing.
  308                 if (self.flags.visible == v) break :visible;
  309 
  310                 // Set our visible state
  311                 self.flags.visible = v;
  312 
  313                 // Visibility affects our QoS class
  314                 self.setQosClass();
  315 
  316                 // If we became visible then we immediately rebuild cells
  317                 // (renderCallback skips updateFrame while invisible) and
  318                 // draw. Going through renderCallback also reschedules
  319                 // any Kitty graphics animation wakeup that lapsed
  320                 // while we were invisible.
  321                 if (v) _ = renderCallback(self, undefined, undefined, {});
  322 
  323                 // Notify the renderer so it can update any state.
  324                 self.renderer.setVisible(v);
  462 fn drawFrame(self: *Thread, now: bool) void {
  463     // If we're invisible, we do not draw.
  464     if (!self.flags.visible) return;
  465 
  466     // If the renderer is managing a vsync on its own, we only draw
  467     // when we're forced to via `now`.
  468     if (!now and self.renderer.hasVsync()) return;
  469 
  470     self.renderer.drawFrame(false) catch |err|
  471         log.warn("error drawing err={}", .{err});
  472 }

  537 fn renderCallback(
  538     self_: ?*Thread,
  539     _: *xev.Loop,
  540     _: *xev.Completion,
  541     r: xev.Timer.RunError!void,
  542 ) xev.CallbackAction {
  543     _ = r catch |err| switch (err) {
  544         // Sent when a scheduled animation wakeup is superseded by a
  545         // newer one (Timer.reset cancels the pending run). Nothing to
  546         // do; the replacement timer carries on.
  547         error.Canceled => return .disarm,
  548         else => unreachable,
  549     };
  550     const t: *Thread = self_ orelse {
  551         // This shouldn't happen so we log it.
  552         log.warn("render callback fired without data set", .{});
  553         return .disarm;
  554     };
  555 
  556     // If the display is now unrealized, release GPU resources now
  557     // we're on the render thread, and do not try to update and draw
  558     // this frame.
  559     if (!t.renderer.display_realized) {
  560         t.renderer.draw_mutex.lockUncancelable(global.io());
  561         defer t.renderer.draw_mutex.unlock(global.io());
  562 
  563         t.renderer.releaseGpuResources();
  564         return .disarm;
  565     }
  566 
  567     // If we're not visible there's no point spending CPU rebuilding cells —
  568     // we'll catch up when the .visible mailbox message flips us back on.
  569     // Kitty graphics animations pause with us and resume on visibility.
  570     if (!t.flags.visible) return .disarm;
  571 
  572     // Update our frame data
  573     t.renderer.updateFrame(
  574         t.state,
  575         t.flags.cursor_blink_visible,
  576     ) catch |err|
  577         log.warn("error rendering err={}", .{err});
  578 
  579     // Draw
  580     t.drawFrame(false);
  581 
  582     // Schedule the next animation wake, if the renderer needs one.
  583     t.armAnimationTimer();
  584 
  585     return .disarm;
  586 }
  596 fn armAnimationTimer(self: *Thread) void {
  597     const wake = self.renderer.animationWake() orelse return;
  598     self.animation_wake = wake.kind;
  599     self.render_h.reset(
  600         &self.loop,
  601         &self.render_c,
  602         &self.render_c_cancel,
  603         wake.delay_ms,
  604         Thread,
  605         self,
  606         animationTimerCallback,
  607     );
  608 }

  610 fn animationTimerCallback(
  611     self_: ?*Thread,
  612     _: *xev.Loop,
  613     _: *xev.Completion,
  614     r: xev.Timer.RunError!void,
  615 ) xev.CallbackAction {
  616     _ = r catch |err| switch (err) {
  617         // Sent when a scheduled animation wake is superseded by a
  618         // newer one (Timer.reset cancels the pending run). Nothing to
  619         // do; the replacement timer carries on.
  620         error.Canceled => return .disarm,
  621         else => unreachable,
  622     };

  627     };
  628 
  629     // Animations pause entirely while we're invisible; the .visible
  630     // mailbox message re-arms us when we can be seen again.
  631     if (!t.flags.visible) return .disarm;
  632 
  633     switch (t.animation_wake) {
  634         // Frame data must be updated (a Kitty animation frame is

  638             undefined,
  639             undefined,
  640             {},

  645         // are much cheaper than a frame update.
  646         .draw => {
  647             t.drawFrame(false);
  648             t.armAnimationTimer();
  649             return .disarm;
  650         },
  651     }
  652 }

   35 /// if a cursor should not be rendered at all.
   36 pub fn style(
   37     state: *const terminal.RenderState,
   38     opts: StyleOptions,
   39 ) ?Style {
   40     // Note the order of conditionals below is important. It represents
   41     // a priority system of how we determine what state overrides cursor
   42     // visibility and style.
   43 
   44     // The cursor must be visible in the viewport to be rendered.
   45     if (state.cursor.viewport == null) return null;
   46 
   47     // If we are in preedit, then we always show the block cursor. We do
   48     // this even if the cursor is explicitly not visible because it shows
   49     // an important editing state to the user.
   50     if (opts.preedit) return .block;
   51 
   52     // If we're at a password input its always a lock.
   53     if (state.cursor.password_input) return .lock;
   54 
   55     // If the cursor is explicitly not visible by terminal mode, we don't render.
   56     if (!state.cursor.visible) return null;
   57 
   58     // If we're not focused, our cursor is always visible so that
   59     // we can show the hollow box.
   60     if (!opts.focused) return .block_hollow;
   61 
   62     // If the cursor is blinking and our blink state is not visible,
   63     // then we don't show the cursor.
   64     if (state.cursor.blinking and !opts.blink_visible) return null;
   65 
   66     // Otherwise, we use whatever style the terminal wants.
   67     return .fromTerminal(state.cursor.visual_style);

 1020         /// want to queue up frames behind a slow compositor.
 1021         const LatestFrame = struct {
 1022             const Self = @This();
 1023             mutex: std.Io.Mutex = .init,
 1024             latest: ?ExportedFrame = null,
 1025 
 1026             pub fn push(self: *LatestFrame, io: std.Io, value: ExportedFrame) void {
 1027                 if (comptime ExportedFrame == void) return;
 1028 
 1029                 self.mutex.lockUncancelable(io);
 1030                 defer self.mutex.unlock(io);
 1031                 if (self.latest) |*old| old.deinit();
 1032                 self.latest = value;
 1033             }
 1034 
 1035             pub fn take(self: *LatestFrame, io: std.Io) ?ExportedFrame {
 1036                 if (comptime ExportedFrame == void) return null;
 1037 
 1038                 self.mutex.lockUncancelable(io);
 1039                 defer self.mutex.unlock(io);
 1040                 const result = self.latest orelse return null;
 1041                 self.latest = null;
 1042                 return result;
 1043             }
 1055         /// Push the latest completed frame, replacing if one previously
 1056         /// existed. Called on the render thread.
 1057         ///
 1058         /// Has no effect for renderers that do not export frames
 1059         /// (i.e. `ExportedFrame == void`).
 1060         pub fn pushFrame(self: *Self, frame: ExportedFrame) void {
 1061             return self.latest_frame.push(global.io(), frame);
 1062         }
 1063 
 1064         /// Take the latest completed frame for the apprt to composite.
 1065         /// Returns null if no frame is available. The caller takes
 1066         /// ownership of the returned frame. Called on the main thread.
 1067         ///
 1068         /// Has no effect for renderers that do not export frames
 1069         /// (i.e. `ExportedFrame == void`).
 1070         pub fn takeFrame(self: *Self) ?ExportedFrame {
 1071             return self.latest_frame.take(global.io());
 1072         }
 1074         /// The display link output callback. CoreVideo invokes this on
 1075         /// its own thread once per display refresh while the link is
 1076         /// running. All it does is wake the render thread, which then
 1077         /// draws a frame.
 1078         fn displayLinkCallback(
 1079             _: *macos.video.DisplayLink,
 1080             ud: ?*xev.Async,
 1081         ) void {
 1082             const draw_now = ud orelse return;
 1083             draw_now.notify() catch |err| {
 1084                 log.err("error notifying draw_now err={}", .{err});
 1085             };
 1086         }
 1136         pub fn animationWake(self: *const Self) ?AnimationWake {
 1137             // Custom shaders animate by redrawing on a fixed cadence,
 1138             // gated by configuration and focus.
 1139             const shader_delay: ?u64 = shader: {
 1140                 if (!self.has_custom_shaders) break :shader null;
 1141                 break :shader switch (self.config.custom_shader_animation) {
 1142                     .false => null,
 1143                     .always => draw_interval_ms,
 1144                     .true => if (self.focused) draw_interval_ms else null,
 1145                 };
 1146             };
 1147 
 1148             // Kitty animations tick during updateFrame; between
 1149             // updates the deadline is absolute on the animation
 1150             // clock, so a stream of draw wakes recomputing this
 1151             // cannot starve it into the future.
 1152             const kitty_delay: ?u64 = kitty: {
 1153                 const next = self.kitty_animation_next_ms orelse break :kitty null;
 1154                 const base = self.kitty_animation_clock orelse break :kitty null;
 1155                 const now: std.Io.Timestamp = .now(global.io(), .awake);
 1156                 const now_ms: u64 = @intCast(@divTrunc(
 1157                     base.durationTo(now).nanoseconds,
 1158                     std.time.ns_per_ms,
 1159                 ));
 1160                 // Never wake faster than the draw interval; an
 1161                 // overdue frame is picked up on the next wake.
 1162                 break :kitty @max(next -| now_ms, draw_interval_ms);
 1163             };
 1164 
 1165             // An update wake includes a draw, so it wins ties.
 1166             if (kitty_delay) |k| {
 1167                 if (shader_delay == null or k <= shader_delay.?) {
 1168                     return .{ .delay_ms = k, .kind = .update };
 1169                 }
 1170             }
 1171 
 1172             if (shader_delay) |s| return .{ .delay_ms = s, .kind = .draw };
 1173 
 1174             return null;
 1175         }
 1273         fn syncDisplayLink(
 1274             self: *Self,
 1275             display_id: ?u32,
 1276             draw_now: ?*xev.Async,
 1277         ) void {
 1278             if (comptime DisplayLink == void) return;
 1279 
 1280             const display_link = self.display_link orelse display_link: {
 1281                 if (!self.config.vsync) return;
 1282                 const callback = draw_now orelse return;
 1283                 const result = DisplayLink.create(
 1284                     self.alloc,
 1285                     xev.Async,
 1286                     &displayLinkCallback,
 1287                     callback,
 1288                 ) catch |err| {

 1290                     // displays. Rendering continues without vsync and the
 1291                     // next call to this function tries again.
 1292                     log.warn("error creating display link; using fallback rendering err={}", .{err});
 1293                     return;
 1294                 };
 1295 
 1296                 self.display_link = result;
 1297                 log.info("created display link", .{});
 1298                 break :display_link result;
 1299             };
 1300 
 1301             if (display_id) |id| display_link.setDisplay(id);
 1302 
 1303             const should_run =
 1304                 // Non-visible windows never vsync
 1305                 self.visible and
 1306                 // Only vsync if we have cell or image changes or animation
 1307                 (self.cells_rebuilt or
 1308                     self.images_rebuilt or
 1309                     self.animationWake() != null);

 1370         /// Update the frame data.
 1371         pub fn updateFrame(
 1372             self: *Self,
 1373             state: *renderer.State,
 1374             cursor_blink_visible: bool,
 1375         ) Allocator.Error!void {
 1376             // CoreText shaping accumulates objects for deferred release over
 1377             // the course of a frame. Always flush those objects, including
 1378             // when rebuilding the frame fails due to memory pressure.
 1379             defer self.font_shaper.endFrame();
 1380 
 1381             // We fully deinit and reset the terminal state every so often
 1382             // so that a particularly large terminal state doesn't cause
 1383             // the renderer to hold on to retained memory.
 1384             //
 1385             // Frame count is ~12 minutes at 120Hz.
 1386             const max_terminal_state_frame_count = 100_000;
 1387             if (self.terminal_state_frame_count >= max_terminal_state_frame_count) {
 1388                 self.terminal_state.deinit(self.alloc);
 1389                 self.terminal_state = .empty;
 1390                 self.terminal_state_frame_count = 0;
 1391             }
 1392             self.terminal_state_frame_count += 1;
 1393 
 1394             // Create an arena for all our temporary allocations while rebuilding
 1395             var arena = ArenaAllocator.init(self.alloc);
 1396             defer arena.deinit();
 1397             const arena_alloc = arena.allocator();
 1398 
 1399             // Data we extract out of the critical area.
 1400             const Critical = struct {
 1401                 links: terminal.RenderState.CellSet,
 1402                 mouse: renderer.State.Mouse,
 1403                 preedit: ?renderer.State.Preedit,
 1404                 scrollbar: terminal.Scrollbar,
 1405                 overlay_features: []const Overlay.Feature,
 1406             };
 1407 
 1408             // Update all our data as tightly as possible within the mutex.
 1409             var critical: Critical = critical: {
 1410                 // NOTE: This code needs be updated to 0.16.0 before you
 1411                 // un-comment it ;)
 1412                 //
 1413                 // const start = try std.time.Instant.now();
 1414                 // const start_micro = std.time.microTimestamp();
 1415                 // defer {
 1416                 //     const end = std.time.Instant.now() catch unreachable;
 1417                 //     std.log.err("[updateFrame critical time] start={}\tduration={} us", .{ start_micro, end.since(start) / std.time.ns_per_us });
 1418                 // }
 1419 
 1420                 state.lockDemand(global.io());
 1421                 defer state.unlockDemand(global.io());
 1422 
 1423                 // If we're in a synchronized output state, we pause all rendering.
 1424                 if (state.terminal.modes.get(.synchronized_output)) {
 1425                     log.debug("synchronized output started, skipping render", .{});
 1426                     return;
 1427                 }
 1428 
 1429                 // If scroll-to-bottom on output is enabled, check if the final line
 1430                 // changed by comparing the bottom-right pin. If the node pointer or
 1431                 // y offset changed, new content was added to the screen.
 1432                 // Update this BEFORE we update our render state so we can
 1433                 // draw the new scrolled data immediately.
 1434                 if (self.config.scroll_to_bottom_on_output) scroll: {
 1435                     const br = state.terminal.screens.active.pages.getBottomRight(.screen) orelse break :scroll;
 1436 
 1437                     // If the pin hasn't changed, then don't scroll.
 1438                     if (self.last_bottom_node == @intFromPtr(br.node) and
 1439                         self.last_bottom_y == br.y) break :scroll;
 1440 
 1441                     // Update tracked pin state for next frame
 1442                     self.last_bottom_node = @intFromPtr(br.node);
 1443                     self.last_bottom_y = br.y;
 1444 
 1445                     // Scroll
 1446                     state.terminal.scrollViewport(.bottom);
 1447                 }
 1448 
 1449                 // Advance any running Kitty graphics animations to the
 1450                 // frame due now, and remember when the next frame is
 1451                 // due (as an absolute deadline, see animationWake) so
 1452                 // the renderer thread can schedule a wakeup for it.
 1453                 //
 1454                 // This runs before the render state update. Advancing
 1455                 // a frame gives the image a new generation, so the
 1456                 // update then reports placements with the new frame,
 1457                 // and the frame is uploaded and drawn right away.
 1458                 self.kitty_animation_next_ms = next: {
 1459                     // Likely case: we have no kitty images, so do nothing.
 1460                     const storage = &state.terminal.screens.active.kitty_images;
 1461                     if (storage.images.count() == 0) break :next null;
 1462 
 1463                     const now: std.Io.Timestamp = .now(global.io(), .awake);
 1464                     const base = self.kitty_animation_clock orelse base: {
 1465                         self.kitty_animation_clock = now;
 1466                         break :base now;
 1467                     };
 1468                     const now_ms: u64 = @intCast(@divTrunc(
 1469                         base.durationTo(now).nanoseconds,
 1470                         std.time.ns_per_ms,
 1471                     ));
 1472                     const delay = storage.animationTick(
 1473                         global.io(),
 1474                         now_ms,
 1475                     ) orelse break :next null;
 1476                     break :next now_ms + delay;
 1477                 };
 1478 
 1479                 // Begin the update of our terminal state. Work that
 1480                 // doesn't require terminal access (e.g. style
 1481                 // denormalization) is deferred to the endUpdate call
 1482                 // outside of this critical section, keeping our lock
 1483                 // hold time as short as possible.
 1484                 try self.terminal_state.beginUpdate(
 1485                     self.alloc,
 1486                     state.terminal,
 1487                 );
 1488 
 1489                 // If our terminal state is dirty at all we need to redo
 1490                 // the viewport search.
 1491                 if (self.terminal_state.dirty != .false) {
 1492                     state.terminal.flags.search_viewport_dirty = true;
 1493                 }
 1494 
 1495                 // Get our scrollbar out of the terminal. We synchronize
 1496                 // the scrollbar read with frame data updates because this
 1497                 // naturally limits the number of calls to this method (it
 1498                 // can be expensive) and also makes it so we don't need another
 1499                 // cross-thread mailbox message within the IO path.
 1500                 const scrollbar = state.terminal.screens.active.pages.scrollbar();
 1501 
 1502                 // Get our preedit state
 1503                 const preedit: ?renderer.State.Preedit = preedit: {
 1504                     const p = state.preedit orelse break :preedit null;
 1505                     break :preedit try p.clone(arena_alloc);
 1506                 };
 1507 
 1508                 // Rebuild our Kitty image placements when the placements
 1509                 // or the images changed. The render state update above
 1510                 // has already positioned every placement, so this is
 1511                 // cheap. Most frames that only change text skip it.
 1512                 if (self.images.kittyRequiresUpdate(
 1513                     state.terminal,
 1514                     &self.terminal_state,
 1515                 )) {
 1516                     // We need to grab the draw mutex since this updates
 1517                     // our image state that drawFrame uses.
 1518                     self.draw_mutex.lockUncancelable(global.io());
 1519                     defer self.draw_mutex.unlock(global.io());
 1520                     self.images.kittyUpdate(
 1521                         self.alloc,
 1522                         state.terminal,
 1523                         &self.terminal_state,
 1524                     );
 1525                     self.images_rebuilt = true;
 1526                 }
 1527 
 1528                 // Get our OSC8 links we're hovering if we have a mouse.
 1529                 // This requires terminal state because of URLs.
 1530                 const links: terminal.RenderState.CellSet = osc8: {
 1531                     // If our mouse isn't hovering, we have no links.
 1532                     const vp = state.mouse.point orelse break :osc8 .empty;
 1533 
 1534                     // If the right mods aren't pressed, then we can't match.
 1535                     if (!state.mouse.mods.equal(inputpkg.ctrlOrSuper(.{})))
 1536                         break :osc8 .empty;
 1537 
 1538                     break :osc8 self.terminal_state.linkCells(
 1539                         arena_alloc,
 1540                         vp,
 1541                     ) catch |err| {
 1542                         log.warn("error searching for OSC8 links err={}", .{err});
 1543                         break :osc8 .empty;
 1544                     };
 1545                 };
 1546 
 1547                 const overlay_features: []const Overlay.Feature = overlay: {
 1548                     const insp = state.inspector orelse break :overlay &.{};
 1549                     const renderer_info = insp.rendererInfo();
 1550                     break :overlay renderer_info.overlayFeatures(
 1551                         arena_alloc,
 1552                     ) catch &.{};
 1553                 };
 1554 
 1555                 break :critical .{
 1556                     .links = links,
 1557                     .mouse = state.mouse,
 1558                     .preedit = preedit,
 1559                     .scrollbar = scrollbar,
 1560                     .overlay_features = overlay_features,
 1561                 };
 1562             };
 1563 
 1564             // Outside the critical area, complete the update we began
 1565             // within it. This must be done before anything reads the
 1566             // render state (e.g. rebuildCells).
 1567             self.terminal_state.endUpdate();
 1568 
 1569             // Outside the critical area we can update our links to contain
 1570             // our regex results.
 1571             self.config.links.renderCellMap(
 1572                 arena_alloc,
 1573                 &critical.links,
 1574                 &self.terminal_state,
 1575                 state.mouse.point,
 1576                 state.mouse.mods,
 1577             ) catch |err| {
 1578                 log.warn("error searching for regex links err={}", .{err});
 1579             };
 1580 
 1581             // Clear our highlight state and update.
 1582             if (self.search_matches_dirty or self.terminal_state.dirty != .false) {
 1583                 self.search_matches_dirty = false;
 1584 
 1585                 // Clear the prior highlights
 1586                 const row_data = self.terminal_state.row_data.slice();
 1587                 var any_dirty: bool = false;
 1588                 for (
 1589                     row_data.items(.highlights),
 1590                     row_data.items(.dirty),
 1591                 ) |*highlights, *dirty| {
 1592                     if (highlights.items.len > 0) {
 1593                         highlights.clearRetainingCapacity();
 1594                         dirty.* = true;
 1595                         any_dirty = true;
 1596                     }
 1597                 }
 1598                 if (any_dirty and self.terminal_state.dirty == .false) {
 1599                     self.terminal_state.dirty = .partial;
 1600                 }
 1601 
 1602                 // NOTE: The order below matters. Highlights added earlier
 1603                 // will take priority.
 1604 
 1605                 if (self.search_selected_match) |m| {
 1606                     self.terminal_state.updateHighlightsFlattened(
 1607                         self.alloc,
 1608                         @backingInt(HighlightTag.search_match_selected),
 1609                         &.{m.match},
 1610                     ) catch |err| {
 1611                         // Not a critical error, we just won't show highlights.
 1612                         log.warn("error updating search selected highlight err={}", .{err});
 1613                     };
 1614                 }
 1615 
 1616                 if (self.search_matches) |m| {
 1617                     self.terminal_state.updateHighlightsFlattened(
 1618                         self.alloc,
 1619                         @backingInt(HighlightTag.search_match),
 1620                         m.matches,
 1621                     ) catch |err| {
 1622                         // Not a critical error, we just won't show highlights.
 1623                         log.warn("error updating search highlights err={}", .{err});
 1624                     };
 1625                 }
 1626             }
 1627 
 1628             // From this point forward no more errors.
 1629             errdefer comptime unreachable;
 1630 
 1631             // Reset our dirty state after updating.
 1632             defer self.terminal_state.dirty = .false;
 1633 
 1634             // Rebuild the overlay image if we have one. We can do this
 1635             // outside of any critical areas.
 1636             self.rebuildOverlay(
 1637                 critical.overlay_features,
 1638             ) catch |err| {
 1639                 log.warn(
 1640                     "error rebuilding overlay surface err={}",
 1641                     .{err},
 1642                 );
 1643             };
 1644 
 1645             // Acquire the draw mutex for all remaining state updates.
 1646             {
 1647                 self.draw_mutex.lockUncancelable(global.io());
 1648                 defer self.draw_mutex.unlock(global.io());
 1649 
 1650                 // Build our GPU cells
 1651                 self.rebuildCells(
 1652                     critical.preedit,
 1653                     renderer.cursorStyle(&self.terminal_state, .{
 1654                         .preedit = critical.preedit != null,
 1655                         .focused = self.focused,
 1656                         .blink_visible = cursor_blink_visible,
 1657                     }),
 1658                     &critical.links,
 1659                 ) catch |err| {
 1660                     // This means we weren't able to allocate our buffer
 1661                     // to update the cells. In this case, we continue with
 1662                     // our old buffer (frozen contents) and log it.
 1663                     comptime assert(@TypeOf(err) == error{OutOfMemory});
 1664                     log.warn("error rebuilding GPU cells err={}", .{err});
 1665                 };
 1666 
 1667                 // The scrollbar is only emitted during draws so we also
 1668                 // check the scrollbar cache here and update if needed.
 1669                 // This is pretty fast.
 1670                 if (!self.scrollbar.eql(critical.scrollbar)) {
 1671                     self.scrollbar = critical.scrollbar;
 1672                     self.scrollbar_dirty = true;
 1673                 }
 1674 
 1675                 // Update our background color
 1676                 self.uniforms.bg_color = .{
 1677                     self.terminal_state.colors.background.r,
 1678                     self.terminal_state.colors.background.g,
 1679                     self.terminal_state.colors.background.b,
 1680                     @intFromFloat(@round(self.config.background_opacity * 255.0)),
 1681                 };
 1682 
 1683                 // If we're on macOS and have glass styles, we remove
 1684                 // the background opacity because the glass effect handles
 1685                 // it.
 1686                 if (comptime builtin.target.os.tag == .macos) switch (self.config.background_blur) {
 1687                     .@"macos-glass-regular",
 1688                     .@"macos-glass-clear",
 1689                     => self.uniforms.bg_color[3] = 0,
 1690 
 1691                     else => {},
 1692                 };
 1693 
 1694                 // Prepare our overlay image for upload (or unload). This
 1695                 // has to use our general allocator since it modifies
 1696                 // state that survives frames.
 1697                 self.images.overlayUpdate(
 1698                     self.alloc,
 1699                     self.overlay,
 1700                 ) catch |err| {
 1701                     log.warn("error updating overlay images err={}", .{err});
 1702                 };
 1703 
 1704                 // Update custom shader uniforms that depend on terminal state.
 1705                 self.updateCustomShaderUniformsFromState();
 1706             }
 1707 
 1708             // Start the display link now that the rebuilt frame is ready.
 1709             self.syncDisplayLink(null, null);
 1710         }
 1712         /// Draw the frame to the screen.
 1713         ///
 1714         /// If `sync` is true, this will synchronously block until
 1715         /// the frame is finished drawing and has been presented.
 1716         pub fn drawFrame(
 1717             self: *Self,
 1718             sync: bool,
 1719         ) !void {
 1720             // Everything that touches draw state happens under the draw
 1721             // mutex. The display link is synced after the mutex is
 1722             // released. It doesn't need the lock, and this keeps the
 1723             // critical section to just drawing.
 1724             const sync_display_link = locked: {
 1725                 self.draw_mutex.lockUncancelable(global.io());
 1726                 defer self.draw_mutex.unlock(global.io());
 1727                 break :locked try self.drawFrameLocked(sync);
 1728             };
 1729 
 1730             if (sync_display_link) self.syncDisplayLink(null, null);
 1731         }
 1732 
 1733         /// The body of `drawFrame`. Must be called with `draw_mutex` held.
 1734         ///
 1735         /// Returns true if the display link should be resynced once the
 1736         /// draw mutex is released. This is only ever true on the no-redraw
 1737         /// path, which a sync draw never takes. Sync draws come from the
 1738         /// main thread, so this keeps `syncDisplayLink` on the render
 1739         /// thread as its contract requires.
 1740         fn drawFrameLocked(
 1741             self: *Self,
 1742             sync: bool,
 1743         ) !bool {
 1744             // After the graphics API is complete (so we defer) we want to
 1745             // update our scrollbar state.
 1746             defer if (self.scrollbar_dirty) {
 1747                 // Fail instantly if the surface mailbox if full, we'll just
 1748                 // get it on the next frame.
 1749                 if (self.surface_mailbox.push(.{
 1750                     .scrollbar = self.scrollbar,
 1751                 }, .instant) > 0) self.scrollbar_dirty = false;
 1752             };
 1753 
 1754             // Let our graphics API do any bookkeeping, etc.
 1755             // that it needs to do before / after `drawFrame`.
 1756             self.api.drawFrameStart();
 1757             defer self.api.drawFrameEnd();
 1758 
 1759             // Retrieve the most up-to-date surface size from the Graphics API
 1760             const surface_size = try self.api.surfaceSize();
 1761 
 1762             // If either of our surface dimensions is zero
 1763             // then drawing is absurd, so we just return.
 1764             if (surface_size.width == 0 or surface_size.height == 0) return false;
 1765 
 1766             // If we have no graphics context we can't draw. This is
 1767             // only the case while unrealized (GTK) or after the render
 1768             // thread exits; displayRealized rebuilds the swap chain.
 1769             if (!self.display_realized) return false;
 1770 
 1771             // Get our swap chain, rebuilding it if it was released
 1772             // while we were hidden. Rebuilding is deferred to draw
 1773             // time because resource creation must happen somewhere
 1774             // our graphics API allows it (OpenGL requires a current
 1775             // context, which drawFrame guarantees).
 1776             const swap_chain: *SwapChain, const swap_chain_rebuilt: bool =
 1777                 if (self.swap_chain) |*sc| .{ sc, false } else rebuild: {
 1778                     self.swap_chain = try SwapChain.init(
 1779                         self.api,
 1780                         self.has_custom_shaders,
 1781                     );
 1782                     break :rebuild .{ &self.swap_chain.?, true };
 1783                 };
 1784 
 1785             const size_changed =
 1786                 self.size.screen.width != surface_size.width or
 1787                 self.size.screen.height != surface_size.height;
 1788 
 1789             // Conditions under which we need to draw the frame, otherwise we
 1790             // don't need to since the previous frame should be identical.
 1791             //
 1792             // While any animation is in progress (a pending animation wake)
 1793             // every draw must actually render.
 1794             const needs_redraw =
 1795                 size_changed or
 1796                 swap_chain_rebuilt or
 1797                 self.cells_rebuilt or
 1798                 self.images_rebuilt or
 1799                 self.animationWake() != null or
 1800                 sync;
 1801 
 1802             if (!needs_redraw) {
 1803                 // Ask our caller to resync the display link once the draw
 1804                 // mutex is released, because we can probably pause the
 1805                 // display link at this point.
 1806                 return true;
 1807             }
 1808             self.cells_rebuilt = false;
 1809             self.images_rebuilt = false;
 1810 
 1811             // Wait for a frame to be available.
 1812             const frame = swap_chain.nextFrame();
 1813             errdefer swap_chain.releaseFrame();
 1814             // log.debug("drawing frame index={}", .{swap_chain.frame_index});
 1815 
 1816             // If we need to reinitialize our shaders, do so.
 1817             if (self.reinitialize_shaders) {
 1818                 self.reinitialize_shaders = false;
 1819                 self.shaders.deinit(self.alloc);
 1820                 try self.initShaders();
 1821             }
 1822 
 1823             // Our shaders should not be defunct at this point.
 1824             assert(!self.shaders.defunct);
 1825 
 1826             // If we have custom shaders, make sure we have the
 1827             // custom shader state in our frame state, otherwise
 1828             // if we have a state but don't need it we remove it.
 1829             if (self.has_custom_shaders) {
 1830                 if (frame.custom_shader_state == null) {
 1831                     frame.custom_shader_state = try .init(self.api);
 1832                     try frame.custom_shader_state.?.resize(
 1833                         self.api,
 1834                         surface_size.width,
 1835                         surface_size.height,
 1836                     );
 1837                 }
 1838             } else if (frame.custom_shader_state) |*state| {
 1839                 state.deinit();
 1840                 frame.custom_shader_state = null;
 1841             }
 1842 
 1843             // If our stored size doesn't match the
 1844             // surface size we need to update it.
 1845             if (size_changed) {
 1846                 self.size.screen = .{

 2033                         .samplers = &.{state.sampler},
 2034                         .draw = .{
 2035                             .type = .triangle,
 2036                             .vertex_count = 3,
 2037                         },
 2038                     });
 2039                 }
 2040             }
 2041 
 2042             return false;

 2045         // Callback from the graphics API when a frame is completed.
 2046         pub fn frameCompleted(
 2047             self: *Self,
 2048             health: Health,
 2049         ) void {
 2050             // If our health value hasn't changed, then we do nothing. We don't
 2051             // do a cmpxchg here because strict atomicity isn't important.
 2052             if (self.health.load(.seq_cst) != health) {
 2053                 self.health.store(health, .seq_cst);
 2054 
 2055                 // Our health value changed, so we notify the surface so that it
 2056                 // can do something about it.
 2057                 _ = self.surface_mailbox.push(.{
 2058                     .renderer_health = health,
 2059                 }, .{ .forever = {} });
 2060             }
 2061 
 2062             // Always release our semaphore. The swap chain is
 2063             // guaranteed to exist here: it is only torn down after
 2064             // waiting for all in-flight frames to complete, and this
 2065             // callback is what signals that completion.
 2066             self.swap_chain.?.releaseFrame();
 2067         }

  268     /// Rebuild the Kitty placements from the render state, and prepare
  269     /// or unload the textures they need. `render_state` must have just
  270     /// been updated from `t`. The terminal is only read to look up
  271     /// images.
  272     ///
  273     /// This reads and writes state used by drawing.
  274     pub fn kittyUpdate(
  275         self: *State,
  276         alloc: Allocator,
  277         t: *const terminal.Terminal,
  278         render_state: *const terminal.RenderState,
  279     ) void {
  280         const storage = &t.screens.active.kitty_images;
  281 
  282         // Remember what we are building from, so that kittyRequiresUpdate
  283         // can tell when it changes.
  284         self.kitty_placements_generation = render_state.imagePlacementsGeneration();
  285         self.kitty_storage_generation = storage.generation;
  286 
  287         // We always clear our previous placements no matter what because
  288         // we rebuild them from scratch.
  289         self.kitty_placements.clearRetainingCapacity();
  290 
  291         // Go through our known images and if there are any that are no longer
  292         // in use then mark them to be freed.
  293         //
  294         // This never conflicts with the below because a placement can't
  295         // reference an image that doesn't exist.
  296         {
  297             var it = self.images.iterator();
  298             while (it.next()) |kv| {
  299                 switch (kv.key_ptr.*) {
  300                     // We're only looking at Kitty images
  301                     .kitty => |id| if (storage.imageById(id)) |image| {
  302                         // An image whose displayed pixels are pending must
  303                         // not keep an older texture for the same ID drawable
  304                         // while its placements wait. For an animation, the
  305                         // displayed pixels are the current frame's.
  306                         if (image.renderData().isPending()) {
  307                             kv.value_ptr.image.markForUnload();
  308                         }
  309                     } else {
  310                         kv.value_ptr.image.markForUnload();
  311                     },
  312 
  313                     .overlay => {},
  314                 }
  315             }
  316         }
  317 
  318         // The render state has already positioned every placement. It
  319         // was updated under the same lock and only returns placements it
  320         // can draw, so their images exist and are fully loaded.
  321         var it = render_state.imagePlacementIterator();
  322         while (it.next()) |p| {
  323             const image = switch (p.protocol) {
  324                 .kitty => storage.imageById(p.image_id) orelse continue,
  325             };
  326 
  327             // For errors we log and continue. We try to place other
  328             // placements even if one fails.
  329             self.prepKittyImage(alloc, &image) catch |err| {
  330                 log.warn("error preparing kitty placement err={}", .{err});
  331                 continue;
  332             };
  333             self.kitty_placements.append(alloc, .{
  334                 .image_id = .{ .kitty = image.id },
  335                 .layer = p.layer,
  336                 .x = p.x,
  337                 .y = p.y,
  338                 .z = p.z,
  339                 .width = p.width,
  340                 .height = p.height,
  341                 .cell_offset_x = p.offset_x,
  342                 .cell_offset_y = p.offset_y,
  343                 .source_x = p.source_x,
  344                 .source_y = p.source_y,
  345                 .source_width = p.source_width,
  346                 .source_height = p.source_height,
  347             }) catch |err| {
  348                 log.warn("error preparing kitty placement err={}", .{err});
  349             };
  350         }
  351 
  352         // Sort the placements by layer, then by z-index. Placements with
  353         // the same z-index are ordered by image, so that the order is
  354         // stable from frame to frame.
  355         std.mem.sortUnstable(
  356             Placement,
  357             self.kitty_placements.items,
  358             {},
  359             struct {
  360                 fn lessThan(
  361                     ctx: void,
  362                     lhs: Placement,
  363                     rhs: Placement,
  364                 ) bool {
  365                     _ = ctx;
  366                     const lhs_layer = @backingInt(lhs.layer);
  367                     const rhs_layer = @backingInt(rhs.layer);
  368                     if (lhs_layer != rhs_layer) return lhs_layer < rhs_layer;
  369                     return lhs.z < rhs.z or
  370                         (lhs.z == rhs.z and lhs.image_id.zLessThan(rhs.image_id));
  371                 }
  372             }.lessThan,
  373         );
  374 
  375         self.kitty_bg_end = layerEnd(self.kitty_placements.items, .below_background);
  376         self.kitty_text_end = layerEnd(self.kitty_placements.items, .below_text);
  377     }

  125     // the buffer is completed and invoke the block directly.
  126     if (sync) {

   50 /// Sets the layer's `contents` to the provided IOSurface.
   51 ///
   52 /// Makes sure to do so on the main thread to avoid visual artifacts.
   53 pub inline fn setSurface(self: *IOSurfaceLayer, surface: *IOSurface) !void {
   54     // We retain the surface to make sure it's not GC'd
   55     // before we can set it as the contents of the layer.
   56     //
   57     // We release in the callback after setting the contents.
   58     surface.retain();
   59     // NOTE: Since `self.layer` is passed as an `objc.c.id`, it's
   60     //       automatically retained when the block is copied, so we
   61     //       don't need to retain it ourselves like with the surface.
   62 
   63     var block = SetSurfaceBlock.init(.{
   64         .layer = self.layer.value,
   65         .surface = surface,
   66     }, &setSurfaceCallback);
   67 
   68     // We check if we're on the main thread and run the block directly if so.
   69     const NSThread = objc.getClass("NSThread").?;
   70     if (NSThread.msgSend(bool, "isMainThread", .{})) {
   71         setSurfaceCallback(&block);
   72     } else {
   73         // NOTE: The block will be copied when we pass it to dispatch_async,
   74         //       and then automatically be deallocated by the objc runtime
   75         //       once it's executed.
   76 
   77         macos.dispatch.dispatch_async(
   78             @ptrCast(macos.dispatch.queue.getMain()),
   79             @ptrCast(&block),
   80         );
   81     }
   82 }

   84 /// Sets the layer's `contents` to the provided IOSurface.
   85 ///
   86 /// Does not ensure this happens on the main thread.
   87 pub inline fn setSurfaceSync(self: *IOSurfaceLayer, surface: *IOSurface) void {
   88     self.layer.setProperty("contents", surface);
   89 }
   90 
   91 const SetSurfaceBlock = objc.Block(struct {
   92     layer: objc.c.id,
   93     surface: *IOSurface,
   94 }, .{}, void);

   96 fn setSurfaceCallback(
   97     block: *const SetSurfaceBlock.Context,
   98 ) callconv(.c) void {
   99     const layer = objc.Object.fromId(block.layer);
  100     const surface: *IOSurface = block.surface;
  101 
  102     // See explanation of why we retain and release in `setSurface`.
  103     defer surface.release();
  104 
  105     // We check to see if the surface is the appropriate size for
  106     // the layer, if it's not then we discard it. This is because
  107     // asynchronously drawn frames can sometimes finish just after
  108     // a synchronously drawn frame during a resize, and if we don't
  109     // discard the improperly sized surface it creates jank.
  110     const bounds = layer.getProperty(macos.graphics.Rect, "bounds");
  111     const scale = layer.getProperty(f64, "contentsScale");
  112     const width: usize = @intFromFloat(bounds.size.width * scale);
  113     const height: usize = @intFromFloat(bounds.size.height * scale);
  114     if (width != surface.getWidth() or height != surface.getHeight()) {
  115         log.debug(
  116             "setSurfaceCallback(): surface is wrong size for layer, discarding. surface = {d}x{d}, layer = {d}x{d}",
  117             .{ surface.getWidth(), surface.getHeight(), width, height },
  118         );
  119         return;
  120     }
  121 
  122     layer.setProperty("contents", surface);
  123 }

  124 pub fn step(self: *const Self, s: Step) void {
  125     if (s.draw.instance_count == 0) return;
  126 
  127     // Set pipeline state
  128     self.encoder.msgSend(
  129         void,
  130         objc.sel("setRenderPipelineState:"),
  131         .{s.pipeline.state.value},
  132     );
  133 
  134     if (s.buffers.len > 0) {
  135         // We reserve index 0 for the vertex buffer, this isn't very
  136         // flexible but it lines up with the API we have for OpenGL.
  137         if (s.buffers[0]) |buf| {
  138             self.encoder.msgSend(
  139                 void,
  140                 objc.sel("setVertexBuffer:offset:atIndex:"),
  141                 .{ buf.value, @as(c_ulong, 0), @as(c_ulong, 0) },
  142             );
  143             self.encoder.msgSend(
  144                 void,
  145                 objc.sel("setFragmentBuffer:offset:atIndex:"),
  146                 .{ buf.value, @as(c_ulong, 0), @as(c_ulong, 0) },
  147             );
  148         }
  149 
  150         // Set the rest of the buffers starting at index 2, this is
  151         // so that we can use index 1 for the uniforms if present.
  152         //
  153         // Also, we set buffers (and textures) for both stages.
  154         //
  155         // Again, not very flexible, but it's consistent and predictable,
  156         // and we need to treat the uniforms as special because of OpenGL.
  157         //
  158         // TODO: Maybe in the future add info to the pipeline struct which
  159         //       allows it to define a mapping between provided buffers and
  160         //       what index they get set at for the vertex / fragment stage.
  161         for (s.buffers[1..], 2..) |b, i| if (b) |buf| {
  162             self.encoder.msgSend(
  163                 void,
  164                 objc.sel("setVertexBuffer:offset:atIndex:"),
  165                 .{ buf.value, @as(c_ulong, 0), @as(c_ulong, i) },
  166             );
  167             self.encoder.msgSend(
  168                 void,
  169                 objc.sel("setFragmentBuffer:offset:atIndex:"),
  170                 .{ buf.value, @as(c_ulong, 0), @as(c_ulong, i) },
  171             );
  172         };
  173     }
  174 
  175     // Set the uniforms as buffer index 1 if present.
  176     if (s.uniforms) |buf| {
  177         self.encoder.msgSend(
  178             void,
  179             objc.sel("setVertexBuffer:offset:atIndex:"),
  180             .{ buf.value, @as(c_ulong, 0), @as(c_ulong, 1) },
  181         );
  182         self.encoder.msgSend(
  183             void,
  184             objc.sel("setFragmentBuffer:offset:atIndex:"),
  185             .{ buf.value, @as(c_ulong, 0), @as(c_ulong, 1) },
  186         );
  187     }
  188 
  189     // Set textures.
  190     for (s.textures, 0..) |t, i| if (t) |tex| {
  191         self.encoder.msgSend(
  192             void,
  193             objc.sel("setVertexTexture:atIndex:"),
  194             .{ tex.texture.value, @as(c_ulong, i) },
  195         );
  196         self.encoder.msgSend(
  197             void,
  198             objc.sel("setFragmentTexture:atIndex:"),
  199             .{ tex.texture.value, @as(c_ulong, i) },
  200         );
  201     };
  202 
  203     // Set samplers.
  204     for (s.samplers, 0..) |samp, i| if (samp) |sampler| {
  205         self.encoder.msgSend(
  206             void,
  207             objc.sel("setFragmentSamplerState:atIndex:"),
  208             .{ sampler.sampler.value, @as(c_ulong, i) },
  209         );
  210     };

  212     // Draw!
  213     self.encoder.msgSend(
  214         void,
  215         objc.sel("drawPrimitives:vertexStart:vertexCount:instanceCount:"),
  216         .{
  217             @backingInt(s.draw.type),
  218             @as(c_ulong, 0),
  219             @as(c_ulong, s.draw.vertex_count),
  220             @as(c_ulong, s.draw.instance_count),
  221         },
  222     );
  223 }

   51 /// Complete this frame and present the target.
   52 ///
   53 /// If `sync` is true, this will block until the frame is presented.
   54 ///
   55 /// NOTE: For OpenGL, `sync` is ignored and we never block, instead
   56 /// pushing the newly presented and exported frame to the frame queue.
   57 pub fn complete(self: *const Self, sync: bool) void {
   58     _ = sync;
   59 
   60     // If there are any GL errors, consider the frame unhealthy.
   61     const health: Health = if (gl.errors.getError()) .healthy else |_| .unhealthy;
   62 
   63     // If the frame is healthy, draw it to an ExportedFrame
   64     // Then sync before exporting and pushing to the present queue.
   65     // The apprt pulls from this queue in its snapshot handler.
   66     const presented_frame: ?OpenGL.ExportedFrame = if (health == .healthy) frame: {
   67         const frame = self.renderer.api.present(
   68             self.target.*,
   69             self.renderer.presentationHealth(),
   70         ) catch |err| {
   71             log.warn("failed to present render target: err={}", .{err});
   72             break :frame null;
   73         };
   74         break :frame frame;
   75     } else null;
   76 
   77     // Sync after frame draw AND frame present GL calls.
   78     gl.finish();
   79 
   80     // At this point the ExportedFrame is finished and can be shared
   81     if (presented_frame) |frame| {
   82         self.renderer.pushFrame(frame);
   83 
   84         // Notify the surface that it should redraw
   85         _ = self.renderer.surface_mailbox.push(.redraw, .{ .forever = {} });
   86     }
   87 
   88     // Report the health to the renderer.
   89     self.renderer.frameCompleted(health);
   90 }

   62 pub fn step(self: *Self, s: Step) void {
   63     if (s.draw.instance_count == 0) return;
   64 
   65     const pbind = s.pipeline.program.use() catch return;
   66     defer pbind.unbind();
   67 
   68     const vaobind = s.pipeline.vao.bind() catch return;

   71     const fbobind = switch (self.attachments[0].target) {
   72         .target => |t| t.framebuffer.bind(.framebuffer) catch return,
   73         .texture => |t| bind: {

   75             fbobind.texture2D(.color0, t.target, t.texture, 0) catch {

   78             };

   80         },
   81     };
   82     defer fbobind.unbind();
   83 
   84     defer self.step_number += 1;
   85 
   86     // If we have a clear color and this is the
   87     // first step in the pass, go ahead and clear.
   88     if (self.step_number == 0) if (self.attachments[0].clear_color) |c| {
   89         gl.clearColor(c[0], c[1], c[2], c[3]);
   90         gl.clear(gl.c.GL_COLOR_BUFFER_BIT);
   91     };
   92 
   93     // Bind the uniform buffer we bind at index 1 to align with Metal.
   94     if (s.uniforms) |ubo| {
   95         _ = ubo.bindBase(.uniform, 1) catch return;
   96     }
   97 
   98     // Bind relevant texture units.
   99     for (s.textures, 0..) |t, i| if (t) |tex| {
  100         gl.Texture.active(@intCast(i)) catch return;
  101         _ = tex.texture.bind(tex.target) catch return;
  102     };
  103 
  104     // Bind relevant samplers.
  105     for (s.samplers, 0..) |s_, i| if (s_) |sampler| {
  106         _ = sampler.sampler.bind(@intCast(i)) catch return;
  107     };
  108 
  109     // Bind 0th buffer as the vertex buffer,
  110     // and bind the rest as storage buffers.
  111     if (s.buffers.len > 0) {
  112         if (s.buffers[0]) |vbo| vaobind.bindVertexBuffer(
  113             0,
  114             vbo.id,
  115             0,
  116             @intCast(s.pipeline.stride),
  117         ) catch return;
  118 
  119         for (s.buffers[1..], 1..) |b, i| if (b) |buf| {
  120             _ = buf.bindBase(.storage, @intCast(i)) catch return;
  121         };
  122     }
  123 
  124     if (s.pipeline.blending_enabled) {
  125         gl.enable(gl.c.GL_BLEND) catch return;
  126         gl.blendFunc(gl.c.GL_ONE, gl.c.GL_ONE_MINUS_SRC_ALPHA) catch return;
  127     } else {
  128         gl.disable(gl.c.GL_BLEND) catch return;
  129     }
  130 
  131     gl.drawArraysInstanced(
  132         s.draw.type,
  133         0,
  134         @intCast(s.draw.vertex_count),
  135         @intCast(s.draw.instance_count),
  136     ) catch return;
  137 }

  937     /// until `endUpdate` is called.
  938     ///
  939     /// This will reset the terminal dirty state since it is consumed
  940     /// by this render state update.
  941     pub fn beginUpdate(
  942         self: *RenderState,
  943         alloc: Allocator,
  944         t: *Terminal,
  945     ) Allocator.Error!void {
  946         const s: *Screen = t.screens.active;
  947         const viewport_pin = s.pages.getTopLeft(.viewport);
  948 
  949         // The overscan rows to capture beyond the viewport. The request
  950         // decides the layout of row_data, and the actual counts are
  951         // limited to the rows that exist. With no request, everything
  952         // below behaves exactly as it does without overscan.
  953         const above_req: usize = self.overscan_request.above;
  954         const below_req: usize = self.overscan_request.below;
  955         const row_data_len: usize = above_req + s.pages.rows + below_req;
  956 
  957         // The first captured row and how many rows above the viewport
  958         // we actually got.
  959         const top: struct { pin: PageList.Pin, above: usize } = if (above_req == 0)
  960             .{ .pin = viewport_pin, .above = 0 }
  961         else switch (viewport_pin.upOverflow(above_req)) {
  962             .offset => |p| .{ .pin = p, .above = above_req },
  963             .overflow => |o| .{ .pin = o.end, .above = above_req - o.remaining },
  964         };
  965 
  966         // How many rows below the viewport we actually get. The page list
  967         // ends at the last active row, so this is zero whenever the
  968         // viewport follows the active area. This is computed before the
  969         // loop below so that a change can force a redraw. When new output
  970         // appears below a scrolled viewport, it can land in an entry that
  971         // was never built or that held a different row.
  972         const below: usize = if (below_req == 0) 0 else below: {
  973             const bottom = viewport_pin.down(s.pages.rows - 1).?;
  974             break :below switch (bottom.downOverflow(below_req)) {
  975                 .offset => below_req,
  976                 .overflow => |o| below_req - o.remaining,
  977             };
  978         };
  979 
  980         // The index of top.pin in row_data.
  981         const first: usize = above_req - top.above;
  982 
  983         const redraw = redraw: {
  984             // If our screen key changed, we need to do a full rebuild
  985             // because our render state is viewport-specific.
  986             if (t.screens.active_key != self.screen) break :redraw true;
  987 
  988             // If our terminal is dirty at all, we do a full rebuild. These
  989             // dirty values are full-terminal dirty values.
  990             {
  991                 const Int = @typeInfo(Terminal.Dirty).@"struct".backing_integer.?;
  992                 const v: Int = @bitCast(t.flags.dirty);
  993                 if (v > 0) break :redraw true;
  994             }
  995 
  996             // If our screen is dirty at all, we do a full rebuild. This is
  997             // a full screen dirty tracker.
  998             {
  999                 const Int = @typeInfo(Screen.Dirty).@"struct".backing_integer.?;
 1000                 const v: Int = @bitCast(t.screens.active.dirty);
 1001                 if (v > 0) break :redraw true;
 1002             }
 1003 
 1004             // If our dimensions changed, we do a full rebuild.
 1005             if (self.rows != s.pages.rows or
 1006                 self.cols != s.pages.cols)
 1007             {
 1008                 break :redraw true;
 1009             }
 1010 
 1011             // If our row_data layout changed (overscan request changed),
 1012             // we do a full rebuild.
 1013             if (self.row_data.len != row_data_len) break :redraw true;
 1014 
 1015             // If the captured overscan changed, rows may have entered
 1016             // row_data that were never built. This only happens when the
 1017             // viewport is scrolled, so it's cheap to be safe.
 1018             if (self.overscan.above != top.above or
 1019                 self.overscan.below != below) break :redraw true;
 1020 
 1021             // If our viewport pin changed, we do a full rebuild.
 1022             if (self.viewport_pin) |old| {
 1023                 if (!old.eql(viewport_pin)) break :redraw true;
 1024             }
 1025 
 1026             break :redraw false;
 1027         };
 1028 
 1029         // Always set our cheap fields, its more expensive to compare
 1030         self.rows = s.pages.rows;
 1031         self.cols = s.pages.cols;
 1032         self.viewport_pin = viewport_pin;
 1033         self.cursor.active = .{ .x = s.cursor.x, .y = s.cursor.y };
 1034         self.cursor.cell = s.cursor.page_cell.*;
 1035         self.cursor.style = s.cursor.style;
 1036         self.cursor.visual_style = s.cursor.cursor_style;
 1037         self.cursor.password_input = t.flags.password_input;
 1038         self.cursor.visible = t.modes.get(.cursor_visible);
 1039         self.cursor.blinking = t.modes.get(.cursor_blinking);
 1040 
 1041         // Always reset the cursor viewport position. In the future we can
 1042         // probably cache this by comparing the cursor pin and viewport pin
 1043         // but may not be worth it.
 1044         self.cursor.viewport = null;
 1045 
 1046         // Colors.
 1047         self.colors.cursor = t.colors.cursor.get();
 1048 
 1049         // The palette is a relatively large copy (768 bytes at the time
 1050         // of writing) so we only copy it when it could have changed. All
 1051         // palette modifications set a terminal-level dirty flag (see
 1052         // Terminal.Dirty.palette), and any terminal-level dirty flag
 1053         // forces a redraw, so checking redraw is sufficient.
 1054         if (redraw) self.colors.palette = t.colors.palette.current;
 1055 
 1056         bg_fg: {
 1057             // Background/foreground can be unset initially which would
 1058             // depend on "default" background/foreground. The expected use
 1059             // case of Terminal is that the caller set their own configured
 1060             // defaults on load so this doesn't happen.
 1061             const bg = t.colors.background.get() orelse break :bg_fg;
 1062             const fg = t.colors.foreground.get() orelse break :bg_fg;
 1063             if (t.modes.get(.reverse_colors)) {
 1064                 self.colors.background = fg;
 1065                 self.colors.foreground = bg;
 1066             } else {
 1067                 self.colors.background = bg;
 1068                 self.colors.foreground = fg;
 1069             }
 1070         }
 1071 
 1072         // Ensure our row length is exactly our height plus requested
 1073         // overscan, freeing or allocating data as necessary. In
 1074         // most cases we'll have a perfectly matching size.
 1075         if (self.row_data.len != row_data_len) {
 1076             @branchHint(.unlikely);
 1077 
 1078             if (self.row_data.len < row_data_len) {
 1079                 // Resize our rows to the desired length, marking any added
 1080                 // values undefined.
 1081                 const old_len = self.row_data.len;
 1082                 try self.row_data.resize(alloc, row_data_len);
 1083 
 1084                 // Initialize all our values. Its faster to use slice() + set()
 1085                 // because appendAssumeCapacity does this multiple times.
 1086                 var row_data = self.row_data.slice();
 1087                 for (old_len..row_data_len) |y| {
 1088                     row_data.set(y, .{
 1089                         .arena = .{},
 1090                         .pin = undefined,
 1091                         .serial = undefined,
 1092                         .raw = undefined,
 1093                         .cells = .empty,
 1094                         .dirty = true,
 1095                         .selection = null,
 1096                         .highlights = .empty,
 1097                         .placeholders = .empty,
 1098                         .applied_styles = .empty,
 1099                     });
 1100                 }
 1101             } else {
 1102                 const row_data = self.row_data.slice();
 1103                 for (
 1104                     row_data.items(.arena)[row_data_len..],
 1105                     row_data.items(.cells)[row_data_len..],
 1106                     row_data.items(.applied_styles)[row_data_len..],
 1107                 ) |state, *cell, *applied| {
 1108                     var arena: ArenaAllocator = state.promote(alloc);
 1109                     arena.deinit();
 1110                     cell.deinit(alloc);
 1111                     applied.deinit(alloc);
 1112                 }
 1113                 self.row_data.shrinkRetainingCapacity(row_data_len);
 1114             }
 1115         }
 1116 
 1117         // Break down our row data
 1118         const row_data = self.row_data.slice();
 1119         const row_arenas = row_data.items(.arena);
 1120         const row_pins = row_data.items(.pin);
 1121         const row_serials = row_data.items(.serial);
 1122         const row_rows = row_data.items(.raw);
 1123         const row_cells = row_data.items(.cells);
 1124         const row_sels = row_data.items(.selection);
 1125         const row_highlights = row_data.items(.highlights);
 1126         const row_placeholders = row_data.items(.placeholders);
 1127         const row_dirties = row_data.items(.dirty);
 1128         const row_applied = row_data.items(.applied_styles);
 1129 
 1130         // If we're redrawing then every row will be rebuilt, superseding
 1131         // any pending style runs from prior updates. Clearing also
 1132         // guarantees pending runs always match the current dimensions
 1133         // (dimension changes force a redraw).
 1134         if (redraw) self.pending_styles.clearRetainingCapacity();
 1135 
 1136         // Go through and setup our rows. We iterate page chunks rather
 1137         // than individual rows so that per-page work (dirty flags, cursor
 1138         // detection, memory pointers) is hoisted out of the row loop. This
 1139         // makes the common case of a clean (or mostly clean) frame very
 1140         // cheap: a contiguous scan of row dirty flags.
 1141         const builder: RowBuilder = .{
 1142             .alloc = alloc,
 1143             .cols = self.cols,
 1144             .arenas = row_arenas,
 1145             .raws = row_rows,
 1146             .cells = row_cells,
 1147             .sels = row_sels,
 1148             .highlights = row_highlights,
 1149             .placeholders = row_placeholders,
 1150             .kitty_stale = &self.kitty.stale,
 1151             .dirties = row_dirties,
 1152             .pending_styles = &self.pending_styles,
 1153             .applied_styles = row_applied,
 1154         };
 1155         const row_data_end: usize = first + top.above + self.rows + below;
 1156         var y: usize = first;
 1157         var any_dirty: bool = false;
 1158         var page_it = top.pin.pageIterator(.right_down, null);
 1159         while (y < row_data_end) {
 1160             const chunk = page_it.next() orelse break;
 1161             const node = chunk.node;
 1162             const node_serial = node.serial;
 1163             const p: *page.Page = node.page();
 1164 
 1165             // The number of rows we consume from this chunk. The chunk
 1166             // may extend beyond what we capture (the viewport is always
 1167             // exactly `rows` tall, plus any overscan) so we clamp.
 1168             const take: usize = @min(
 1169                 @as(usize, chunk.end - chunk.start),
 1170                 row_data_end - y,
 1171             );
 1172 
 1173             // Find our cursor if we haven't found it yet. We do this even
 1174             // if rows are not dirty because the cursor is unrelated. We
 1175             // can check the chunk bounds once rather than every row.
 1176             if (self.cursor.viewport == null and
 1177                 node == s.cursor.page_pin.node)
 1178             cursor: {
 1179                 const cy = s.cursor.page_pin.y;
 1180                 if (cy < chunk.start or cy >= chunk.start + take) break :cursor;
 1181 
 1182                 // The cursor may be in an overscan row, in which case it
 1183                 // is not visible in the viewport.
 1184                 const idx = y + (cy - chunk.start);
 1185                 const vp_start = self.viewportStart();
 1186                 if (idx < vp_start or idx >= vp_start + self.rows) break :cursor;
 1187                 self.cursor.viewport = .{
 1188                     .y = @intCast(idx - vp_start),
 1189                     .x = s.cursor.x,
 1190 
 1191                     // Future: we should use our own state here to look this
 1192                     // up rather than calling this.
 1193                     .wide_tail = if (s.cursor.x > 0)
 1194                         s.cursorCellLeft(1).wide == .wide
 1195                     else
 1196                         false,
 1197                 };
 1198             }
 1199 
 1200             // The page-level dirty flag applies to every row in the chunk.
 1201             // We consume (clear) it now; each node appears at most once in
 1202             // this iteration and we're the only consumer of dirty state.
 1203             const page_dirty = p.dirty;
 1204             if (page_dirty) p.dirty = false;
 1205 
 1206             // Get our contiguous rows for this chunk.
 1207             const page_rows: []page.Row = p.rows.ptr(p.memory)[chunk.start..][0..take];
 1208             assert(p.size.cols == self.cols);
 1209 
 1210             // Store our pins and their node generations. We have to store
 1211             // these even for rows that aren't dirty because dirty is only a
 1212             // renderer optimization; it doesn't apply to memory movement.
 1213             // This lets us remap any cell pins back to an exact entry in our
 1214             // RenderState and validate them later without dereferencing a
 1215             // potentially stale node.
 1216             //
 1217             // We can skip the writes when the pins and serials are unchanged:
 1218             // if we're not redrawing, every value was stored by a prior update
 1219             // (row count changes force a redraw). Within a single update a
 1220             // node appears at most once and its stored pins have consecutive
 1221             // y values, so if the first and last entries of this chunk's range
 1222             // already match then every entry in between matches too.
 1223             if (redraw or
 1224                 row_pins[y].node != node or
 1225                 row_pins[y].y != chunk.start or
 1226                 row_serials[y] != node_serial or
 1227                 row_pins[y + take - 1].node != node or
 1228                 row_pins[y + take - 1].y != chunk.start + take - 1 or
 1229                 row_serials[y + take - 1] != node_serial)
 1230             {
 1231                 for (
 1232                     row_pins[y..][0..take],
 1233                     row_serials[y..][0..take],
 1234                     chunk.start..,
 1235                 ) |*pin, *serial, py| {
 1236                     pin.* = .{ .node = node, .y = @intCast(py) };
 1237                     serial.* = node_serial;
 1238                 }
 1239             }
 1240 
 1241             if (!redraw and !page_dirty) {
 1242                 // Only dirty rows (usually none) need a rebuild. Scan the
 1243                 // dirty flags a group at a time; the dirty bit is directly
 1244                 // testable on the packed row representation.
 1245                 var i: usize = 0;
 1246                 while (take - i >= RowDirtyMask.group_len) : (i += RowDirtyMask.group_len) {
 1247                     if (RowDirtyMask.match(page_rows, i)) {
 1248                         @branchHint(.likely);
 1249                         continue;
 1250                     }
 1251 
 1252                     for (page_rows[i..][0..RowDirtyMask.group_len], i..) |*page_row, j| {
 1253                         if (!page_row.dirty) continue;
 1254                         any_dirty = true;
 1255                         try builder.row(p, page_row, y + j);
 1256                         page_row.dirty = false;
 1257                     }
 1258                 }
 1259                 while (i < take) : (i += 1) {
 1260                     const page_row = &page_rows[i];
 1261                     if (!page_row.dirty) continue;
 1262                     any_dirty = true;
 1263                     try builder.row(p, page_row, y + i);
 1264                     page_row.dirty = false;
 1265                 }
 1266             } else {
 1267                 // Rebuild every row in the chunk.
 1268                 any_dirty = true;
 1269                 for (page_rows, 0..) |*page_row, i| {
 1270                     builder.row(p, page_row, y + i) catch |err| {
 1271                         // A failed row and every row not yet rebuilt must stay
 1272                         // dirty so the next update cannot skip them and expose
 1273                         // partially initialized managed cell data.
 1274                         for (page_rows[i..]) |*remaining| remaining.dirty = true;
 1275                         return err;
 1276                     };
 1277                     page_row.dirty = false;
 1278                 }
 1279             }
 1280 
 1281             y += take;
 1282         }
 1283         assert(y == row_data_end);
 1284 
 1285         self.overscan = .{
 1286             .above = @intCast(top.above),
 1287             .below = @intCast(below),
 1288         };
 1289 
 1290         // Work out what the Kitty placeholder runs draw. This must come
 1291         // after setting the overscan above, because it reads the
 1292         // captured rows.
 1293         if (comptime build_options.kitty_graphics) try Kitty.update(
 1294             self,
 1295             alloc,
 1296             t,
 1297             redraw,
 1298             any_dirty,
 1299         );
 1300 
 1301         // If our screen has a selection, then mark the rows with the
 1302         // selection. We do this outside of the loop above because its unlikely
 1303         // a selection exists and because the way our selections are structured
 1304         // today is very inefficient.
 1305         //
 1306         // NOTE: To improve the performance of the block below, we'll need
 1307         // to rethink how we model selections in general.
 1308         //
 1309         // There are performance improvements that can be made here, though.
 1310         // For example, `containedRow` recalculates a bunch of information
 1311         // we can cache.
 1312         if (s.selection) |*sel| selection: {
 1313             @branchHint(.unlikely);
 1314 
 1315             // Populate our selection cache to avoid some expensive
 1316             // recalculation.
 1317             const cache: *const SelectionCache = cache: {
 1318                 if (self.selection_cache) |*c| cache_check: {
 1319                     // If we're redrawing, we recalculate the cache just to
 1320                     // be safe.
 1321                     if (redraw) break :cache_check;
 1322 
 1323                     // If our selection isn't equal, we aren't cached!
 1324                     if (!c.selection.eql(sel.*)) break :cache_check;
 1325 
 1326                     // If we have no dirty rows, we can not recalculate.
 1327                     if (!any_dirty) break :selection;
 1328 
 1329                     // We have dirty rows, we can utilize the cache.
 1330                     break :cache c;
 1331                 }
 1332 
 1333                 // Create a new cache
 1334                 const tl_pin = sel.topLeft(s);
 1335                 const br_pin = sel.bottomRight(s);
 1336                 self.selection_cache = .{
 1337                     .selection = .init(tl_pin, br_pin, sel.rectangle),
 1338                     .tl_pin = tl_pin,
 1339                     .br_pin = br_pin,
 1340                 };
 1341                 break :cache &self.selection_cache.?;
 1342             };
 1343 
 1344             // Grab the inefficient data we need from the selection. At
 1345             // least we can cache it.
 1346             const tl = s.pages.pointFromPin(.screen, cache.tl_pin).?.screen;
 1347 
 1348             // Not cached since the cell contents may change.
 1349             const br_pin: PageList.Pin = if (!sel.rectangle and
 1350                 cache.br_pin.rowAndCell().cell.wide == .spacer_head)
 1351                 cache.br_pin.rightWrap(1) orelse cache.br_pin
 1352             else
 1353                 cache.br_pin;
 1354             const br = s.pages.pointFromPin(.screen, br_pin).?.screen;
 1355 
 1356             // We need to determine if our selection is within the viewport.
 1357             // The viewport is generally very small so the efficient way to
 1358             // do this is to traverse the viewport pages and check for the
 1359             // matching selection pages. Unused entries are skipped.
 1360             const range = self.rowDataRange();
 1361             for (
 1362                 row_pins[range.start..range.end],
 1363                 row_sels[range.start..range.end],
 1364                 row_dirties[range.start..range.end],
 1365             ) |pin, *sel_bounds, *dirty| {
 1366                 const p = s.pages.pointFromPin(.screen, pin).?.screen;
 1367                 const row_sel = sel.containedRowCached(
 1368                     s,
 1369                     cache.tl_pin,
 1370                     br_pin,
 1371                     pin,
 1372                     tl,
 1373                     br,
 1374                     p,
 1375                 ) orelse {
 1376                     if (sel_bounds.* != null) {
 1377                         sel_bounds.* = null;
 1378                         dirty.* = true;
 1379                         any_dirty = true;
 1380                     }
 1381                     continue;
 1382                 };
 1383                 const start = row_sel.start();
 1384                 const end = row_sel.end();
 1385                 assert(start.node == end.node);
 1386                 assert(start.x <= end.x);
 1387                 assert(start.y == end.y);
 1388 
 1389                 const start_x = if (start.x > 0 and
 1390                     start.rowAndCell().cell.wide == .spacer_tail)
 1391                     start.x - 1
 1392                 else
 1393                     start.x;
 1394 
 1395                 sel_bounds.* = .{ start_x, end.x };
 1396             }
 1397         }
 1398 
 1399         // Handle dirty state.
 1400         if (redraw) {
 1401             // Fully redraw resets some other state.
 1402             self.screen = t.screens.active_key;
 1403             self.dirty = .full;
 1404 
 1405             // Note: we don't clear any row_data here because our rebuild
 1406             // above did this.
 1407         } else if (any_dirty and self.dirty == .false) {
 1408             self.dirty = .partial;
 1409         }
 1410 
 1411         // Clear our dirty flags
 1412         t.flags.dirty = .{};
 1413         s.dirty = .{};
 1414     }
 1416     /// Complete a prior `beginUpdate` call by performing any deferred
 1417     /// work. At the time of writing, this denormalizes the pending
 1418     /// style runs into the per-cell style data.
 1419     ///
 1420     /// This only reads and writes memory owned by the render state, so
 1421     /// it is safe to call while the terminal is being modified (no
 1422     /// terminal lock is required).
 1423     pub fn endUpdate(self: *RenderState) void {
 1424         // Common case: no styled rows were rebuilt.
 1425         if (self.pending_styles.items.len == 0) return;
 1426 
 1427         const row_data = self.row_data.slice();
 1428         const row_cells = row_data.items(.cells);
 1429         const row_applied = row_data.items(.applied_styles);
 1430 
 1431         // Process the pending runs one row segment at a time. All the
 1432         // runs for a row are appended contiguously by a single
 1433         // beginUpdate, so a segment boundary is simply a change in y.
 1434         //
 1435         // We consume the runs as a shrinking slice to reduce per-segment setup.
 1436         var rest = self.pending_styles.items;
 1437         while (rest.len > 0) {
 1438             const y = rest[0].y;
 1439             var n: usize = 1;
 1440             while (n < rest.len and rest[n].y == y) n += 1;
 1441             const segment = rest[0..n];
 1442             rest = rest[n..];
 1443 
 1444             // Defensive: the row data may have changed shape if the
 1445             // caller violated ordering (e.g. an error path skipped an
 1446             // endUpdate between updates). Any update that changes
 1447             // dimensions clears the pending list (redraw), so this
 1448             // should never actually trigger, but the cost is trivial.
 1449             if (y >= row_cells.len) continue;
 1450 
 1451             // If the segment matches the runs already denormalized
 1452             // into this row's cell data then the per-cell styles are
 1453             // already correct and the (comparatively large) fill can
 1454             // be skipped entirely. This is the common case: rebuilt
 1455             // rows usually keep their styling (only the text
 1456             // changed), and full redraws of an unchanged screen keep
 1457             // both.
 1458             const applied = &row_applied[y];
 1459             if (runsEql(applied.items, segment)) continue;
 1460 
 1461             const cells = row_cells[y].slice();
 1462             const styles = cells.items(.style);
 1463             for (segment) |run| {
 1464                 // Clamp to cells.len rather than styles.len to avoid repeating
 1465                 // the capacity check from items(.style) for every run.
 1466                 const end = @min(run.end, cells.len);
 1467                 const start = @min(run.start, end);
 1468 
 1469                 fillStyles(styles[start..end], @bitCast(run.style));
 1470             }
 1471 
 1472             // Record what we applied so the next rebuild of this row
 1473             // can skip the fill. beginUpdate reserved the capacity
 1474             // (we cannot allocate here); if it doesn't fit (e.g.
 1475             // segments merged across multiple begins without an end)
 1476             // leave the cache empty, which never matches and simply
 1477             // means the next rebuild applies its runs.
 1478             applied.clearRetainingCapacity();
 1479             if (applied.capacity >= segment.len) {
 1480                 applied.appendSliceAssumeCapacity(segment);
 1481             }
 1482         }
 1483         self.pending_styles.clearRetainingCapacity();
 1484     }
 2114     fn update(
 2115         state: *RenderState,
 2116         alloc: Allocator,
 2117         t: *const Terminal,
 2118         redraw: bool,
 2119         any_dirty: bool,
 2120     ) Allocator.Error!void {
 2121         const storage = &t.screens.active.kitty_images;
 2122 
 2123         // `roots` and what a run draws depend only on the image storage
 2124         // and the cell size. Switching to the other screen is caught
 2125         // too, because no two storages share a generation.
 2126         //
 2127         // This runs on every update, so the common case (no images and
 2128         // nothing changed) must stay a few comparisons. A redraw alone,
 2129         // such as a viewport scroll, must not build `roots` again. A
 2130         // change in rows or columns always causes a redraw, so those are
 2131         // only compared on redraws.
 2132         const roots_stale = state.kitty.generation != storage.generation or
 2133             state.kitty.width_px != t.width_px or
 2134             state.kitty.height_px != t.height_px or
 2135             (redraw and (state.kitty.rows != state.rows or
 2136                 state.kitty.cols != state.cols));
 2137         if (!roots_stale and !state.kitty.stale) {
 2138             @branchHint(.likely);
 2139 
 2140             // No runs to look up. Placements only move when the viewport
 2141             // moves, when the storage says that they moved, or (for
 2142             // placements positioned from placeholder cells) when rows
 2143             // are rebuilt.
 2144             if (state.kitty.roots.items.len == 0) return;
 2145             if (!redraw and state.kitty.dirty == storage.dirty and
 2146                 !(any_dirty and state.kitty.virtual_anchors > 0)) return;
 2147         }
 2148

 2150             .never_inline,
 2151             updateSlow,
 2152             .{ state, alloc, t, roots_stale },
 2153         );
 2154     }

 2156     /// The part of `update` that does the work. It is kept out of line
 2157     /// so that the common case in `update` stays small. `roots_stale` is
 2158     /// true when `roots` must be built again.
 2159     fn updateSlow(
 2160         state: *RenderState,
 2161         alloc: Allocator,
 2162         t: *const Terminal,
 2163         roots_stale: bool,
 2164     ) Allocator.Error!void {
 2165         const storage = &t.screens.active.kitty_images;
 2166         if (roots_stale) {
 2167             state.kitty.generation = storage.generation;
 2168             state.kitty.width_px = t.width_px;
 2169             state.kitty.height_px = t.height_px;
 2170             state.kitty.rows = state.rows;
 2171             state.kitty.cols = state.cols;
 2172             try rebuildRoots(state, alloc, t);
 2173         }
 2174 
 2175         // Whether the runs that the iterator returns changed. They did if
 2176         // the row builder added or dropped runs, or if the loop below
 2177         // looks up any run again, since its result may differ.
 2178         var runs_changed = state.kitty.stale;
 2179         if (roots_stale or state.kitty.stale) {
 2180             state.kitty.stale = false;
 2181             const cell_width: u32 = t.width_px / state.cols;
 2182             const cell_height: u32 = t.height_px / state.rows;
 2183             const row_data = state.row_data.slice();
 2184             const row_pins = row_data.items(.pin);
 2185             const row_placeholders = row_data.items(.placeholders);
 2186             const range = state.rowDataRange();
 2187             for (
 2188                 row_pins[range.start..range.end],
 2189                 row_placeholders[range.start..range.end],
 2190             ) |pin, runs| {
 2191                 for (runs.items) |*run| {
 2192                     if (!roots_stale and run.state != .stale) continue;
 2193                     runs_changed = true;
 2194                     resolvePlaceholder(
 2195                         storage,
 2196                         pin,
 2197                         run,
 2198                         cell_width,
 2199                         cell_height,
 2200                     );
 2201                 }
 2202             }
 2203         }
 2204 
 2205         try placeRoots(state, alloc);
 2206         state.kitty.dirty = storage.dirty;
 2207 
 2208         // Tell consumers to read the iterator again if anything it
 2209         // returns may be different. Rebuilding `roots` alone doesn't
 2210         // count: if no run was looked up and `placed` is the same, the
 2211         // output is the same.
 2212         if (runs_changed or !placedEql(
 2213             state.kitty.placed.items,
 2214             state.kitty.placed_prev.items,
 2215         )) state.kitty.placements_generation +%= 1;
 2216     }
 2666     fn row(
 2667         b: *const RowBuilder,
 2668         p: *page.Page,
 2669         page_row: *const page.Row,
 2670         vy: usize,
 2671     ) Allocator.Error!void {
 2672         // Promote our arena. State is copied by value so we need to
 2673         // restore it on all exit paths so we don't leak memory.
 2674         var arena = b.arenas[vy].promote(b.alloc);
 2675         defer b.arenas[vy] = arena.state;
 2676 
 2677         // Reset our per-row state if we're rebuilding this row. A
 2678         // non-zero cell length means the row was populated by a prior
 2679         // update.
 2680         if (b.cells[vy].len > 0) {
 2681             _ = arena.reset(.retain_capacity);
 2682             b.sels[vy] = null;
 2683             b.highlights[vy] = .empty;
 2684             // Dropping this row's runs changes what the image placement
 2685             // iterator returns, so tell the Kitty update. Only a row
 2686             // that had placeholder cells can have runs. Checking that
 2687             // flag first means other rows never read their run list.
 2688             if (comptime build_options.kitty_graphics) {
 2689                 if (b.raws[vy].kitty_virtual_placeholder and
 2690                     b.placeholders[vy].items.len > 0) b.kitty_stale.* = true;
 2691             }
 2692             b.placeholders[vy] = .empty;
 2693         }
 2694         b.dirties[vy] = true;
 2695 
 2696         // Get all our cells in the page.
 2697         const page_cells: []const page.Cell = page_row.cells.ptr(p.memory)[0..b.cols];
 2698 
 2699         // Copy our raw row data. The terminal row stays dirty until the
 2700         // build succeeds, but the render-state snapshot is always rebuilt.
 2701         b.raws[vy] = page_row.*;
 2702         b.raws[vy].dirty = false;
 2703 
 2704         // Note: our cells MultiArrayList uses our general allocator.
 2705         // We do this on purpose because as rows become dirty, we do
 2706         // not want to reallocate space for cells (which are large). This
 2707         // was a source of huge slowdown.
 2708         //
 2709         // Our per-row arena is only used for temporary allocations
 2710         // pertaining to cells directly (e.g. graphemes, hyperlinks).
 2711         const cells: *std.MultiArrayList(RenderState.Cell) = &b.cells[vy];
 2712         if (cells.len != b.cols) {
 2713             // The cell storage (including the per-cell style data) is
 2714             // being reallocated, so the applied style cache no longer
 2715             // describes it. Clear it before the resize so an error
 2716             // can't leave it stale.
 2717             b.applied_styles[vy].clearRetainingCapacity();
 2718             try cells.resize(b.alloc, b.cols);
 2719         }
 2720 
 2721         // We always copy our raw cell data. In the case we have no
 2722         // managed memory, we can skip setting any other fields.
 2723         //
 2724         // This is an important optimization. For plain-text screens
 2725         // this ends up being something around 300% faster based on
 2726         // the `screen-clone` benchmark.
 2727         const cells_slice = cells.slice();
 2728         fastmem.copy(
 2729             page.Cell,
 2730             cells_slice.items(.raw),
 2731             page_cells,
 2732         );
 2733         // Rows with Kitty placeholders are rare. They get their own copy
 2734         // of the scan below, so other rows pay nothing for them. These
 2735         // rows are scanned even without managed memory, because a
 2736         // placeholder cell can be a plain character with no style.
 2737         const placeholder_row = build_options.kitty_graphics and
 2738             page_row.kitty_virtual_placeholder;
 2739         if (!page_row.managedMemory() and !placeholder_row) return;
 2740 
 2741         const arena_alloc = arena.allocator();
 2742         if (placeholder_row) {
 2743             @branchHint(.unlikely);
 2744             var decoder: Kitty.PlaceholderDecoder = .{
 2745                 .list = &b.placeholders[vy],
 2746                 .alloc = arena_alloc,
 2747                 .stale = b.kitty_stale,
 2748             };
 2749             try @call(
 2750                 .never_inline,
 2751                 scan,
 2752                 .{ b, true, true, p, page_cells, cells_slice, arena_alloc, vy, &decoder },
 2753             );
 2754             try decoder.flush();
 2755             return;
 2756         }
 2757 
 2758         // Rows with graphemes are rare, so they get their own copy of the
 2759         // scan. A row without the grapheme flag has no grapheme cells.
 2760         if (page_row.grapheme) {
 2761             try @call(
 2762                 .never_inline,
 2763                 scan,
 2764                 .{ b, false, true, p, page_cells, cells_slice, arena_alloc, vy, {} },
 2765             );
 2766         } else {
 2767             // The scan below never looks at the content tag, so a
 2768             // grapheme cell on this row would keep an undefined
 2769             // grapheme slice in the render state.
 2770             if (build_options.slow_runtime_safety) {
 2771                 for (page_cells) |cell| assert(!cell.hasGrapheme());
 2772             }
 2773 
 2774             try @call(
 2775                 .always_inline,
 2776                 scan,
 2777                 .{ b, false, false, p, page_cells, cells_slice, arena_alloc, vy, {} },
 2778             );
 2779         }
 2780     }
 2790     fn scan(
 2791         b: *const RowBuilder,
 2792         comptime placeholders: bool,
 2793         comptime graphemes: bool,
 2794         p: *page.Page,
 2795         page_cells: []const page.Cell,
 2796         cells_slice: std.MultiArrayList(RenderState.Cell).Slice,
 2797         arena_alloc: Allocator,
 2798         vy: usize,
 2799         decoder: if (placeholders) *Kitty.PlaceholderDecoder else void,
 2800     ) Allocator.Error!void {
 2801         const cells_grapheme = cells_slice.items(.grapheme);
 2802         const n = page_cells.len;
 2803 
 2804         // Each cell starts at most one style run, so reserve a row's worth
 2805         // up front to keep the appends below allocation-free.
 2806         try b.pending_styles.ensureUnusedCapacity(b.alloc, n);
 2807 
 2808         // We append through a local list to avoid aliasing that forces the
 2809         // compiler to reload the list's pointer and length after every run.
 2810         var runs: std.ArrayList(RenderState.StyleRun) = .initBuffer(
 2811             b.pending_styles.unusedCapacitySlice(),
 2812         );
 2813         defer b.pending_styles.items.len += runs.items.len;
 2814 
 2815         // Likewise, look styles up through locals so run stores
 2816         // can't alias them and force a reload before every lookup.
 2817         const styles = p.styles;
 2818         const memory = p.memory;
 2819 
 2820         const Mask = if (graphemes or placeholders) CellSpecialMask else CellStyleMask;
 2821 
 2822         var x: usize = 0;
 2823         while (x < n) {
 2824             const cell = page_cells[x];
 2825             const start = x;
 2826 
 2827             // Skip runs of plain cells a group at a time. Cells that
 2828             // need managed handling are often rare even within rows that
 2829             // have managed memory (e.g. a row is "styled" if a single
 2830             // cell has a style) so groups are skipped with a few vector
 2831             // operations. On placeholder rows a plain cell can be a
 2832             // placeholder, so those rows still check each cell. Without
 2833             // graphemes or placeholders, we skip background color cells too.
 2834             //
 2835             // The skip starts at this cell rather than the next one so
 2836             // that the first group starts here. A plain run of a whole
 2837             // number of groups then needs no scalar checks after this one.
 2838             if (!placeholders and Mask.matchScalar(cell)) {
 2839                 x = Mask.eqlEnd(page_cells, x, 0);
 2840                 continue;
 2841             }
 2842 
 2843             // Find the run of cells sharing this cell's style and record
 2844             // it: this does one style lookup per run and defers the
 2845             // (large) per-cell fill to endUpdate, outside of any terminal
 2846             // locks. Note grapheme cells may also be styled. The style
 2847             // must be recorded as a run (rather than written directly)
 2848             // so that it is ordered correctly relative to possibly-stale
 2849             // runs from prior updates.
 2850             x = Mask.eqlEnd(page_cells, x + 1, Mask.pattern(cell));
 2851 
 2852             // We use a pointer here so we can reuse the style without
 2853             // keeping a copy alive across the grapheme allocations below.
 2854             const style_val: *const Style = if (cell.hasStyling())
 2855                 styles.get(memory, cell.style_id)
 2856             else
 2857                 &.{};
 2858 
 2859             if (cell.hasStyling()) runs.appendAssumeCapacity(.{
 2860                 .y = @intCast(vy),
 2861                 .start = @intCast(start),
 2862                 .end = @intCast(x),
 2863                 .style = style_val.*,
 2864             });
 2865 
 2866             // If we have multi-codepoint graphemes, look them up. On
 2867             // placeholder rows, feed each placeholder to the decoder.
 2868             const grapheme = graphemes and cell.content_tag == .codepoint_grapheme;
 2869             if (grapheme or placeholders) {
 2870                 for (start..x) |cx| {
 2871                     const cps: []const u21 = if (grapheme) cps: {
 2872                         const cps = try arena_alloc.dupe(
 2873                             u21,
 2874                             p.lookupGrapheme(&page_cells[cx]) orelse &.{},
 2875                         );
 2876                         cells_grapheme[cx] = cps;
 2877                         break :cps cps;
 2878                     } else &.{};
 2879 
 2880                     // A placeholder can be a plain cell, or leave out its row
 2881                     // and column (it then continues the run to its left). Most
 2882                     // placeholders give their row and column in the image as
 2883                     // combining characters, so most are graphemes.
 2884                     if (placeholders and
 2885                         page_cells[cx].codepoint() == kitty_gfx.unicode.placeholder)
 2886                     {
 2887                         try decoder.feed(cx, style_val, cps);
 2888                     }
 2889                 }
 2890             }
 2891         }
 2892 
 2893         // Reserve the applied style cache capacity for the runs we
 2894         // appended so that endUpdate (which cannot allocate) is able
 2895         // to record what it applies. See Row.applied_styles.
 2896         if (runs.items.len > 0) try b.applied_styles[vy].ensureTotalCapacity(
 2897             b.alloc,
 2898             runs.items.len,
 2899         );
 2900     }
