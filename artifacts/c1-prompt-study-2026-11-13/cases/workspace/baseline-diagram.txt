|-- WorkspacePersistence [workspacePersistence.ts]: shared live-state-to-record adapter

|   |   |-- Write metadata + saved baseline
|   |   |-- Per-session cursor -> controller.persistenceChanges(cursor)

|-- Shared persistence pipeline [recordStorage.ts, workspaceStorage.ts, workspaceStorage.worker.ts]
|   |-- RecordStorage.batch: synchronous cache update
|   |   |-- Deep-equal values / absent deletions -> no queued write
|   |   |-- pending[key] retains the latest change for that key
|   |   `-- Default debounce: 150 ms; scheduling window capped at 1 s from first pending change
|   |-- pump(): one write in flight
|   |   |-- Freeze pending changes as { requestId, expectedRevision, changes }
|   |   |-- New edits accumulate separately while that batch is in flight
|   |   `-- WorkspaceStorage sends the batch to one reusable worker
|   |       |-- Normalize records and cache encoded body by requestId
|   |       `-- POST /api/mapping/workspace/records
|   |           |-- Success -> update known records and release cached body
|   |           |   `-- RecordStorage advances revision/persisted sequence and clears flight
|   |           |-- Transport/server uncertainty -> retain flight and encoded body
|   |           |   `-- Retry identical request ID, expected revision and changes
|   |           |       `-- Background exponential backoff capped at 30 s
|   |           `-- HTTP 4xx -> REJECTED; discard cached body
|   |               `-- Background scheduling stops on rejection or version/request-ID conflict
|   |-- flush(): wait until the sequence captured at entry is acknowledged
|   |-- Explicit commit(changes)
|   |   |-- Flush earlier writes -> save prior cache -> submit transaction -> await flush
|   |   |-- Ambiguous failure -> retry same flight after 500 ms while not disposed
|   |   `-- Definitive rejection -> restore prior cache, clear abandoned write, throw
|   `-- WorkspaceStorage operations share this queue
|       |-- load -> worker GET ?history=current -> reconstruct workspace + initialize records
|       |-- save(snapshot) -> flush -> worker diffs full replacement -> commit
|       |-- saveDocument(document, index) -> flush -> commit canvas records + index together
|       `-- Transfer/projection UI installs prepared state only after storage acknowledgement
`-- Lifetime boundaries
    |-- Visibility becomes hidden -> flush view state and attempt storage flush
    |-- Pending view/storage changes or dirty/busy drafts -> beforeunload warning
    |-- Page cleanup -> detach persistence subscriptions; dispose workspace sessions and view state
    |   `-- Locally owned storage: attempt flush, then dispose records and terminate worker
    |       `-- Injected storage remains under its external owner's lifetime
    `-- Abrupt tab termination has no flush guarantee
