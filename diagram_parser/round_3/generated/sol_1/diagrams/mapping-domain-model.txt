    |-- resolves relative position to absolute bounds
    |-- measures visible ASCII content
    |-- expands containers around visible children
    |-- applies minimumSize only while expanded
    `-- uses collapsedContentId while collapsed


PROJECTION MODEL
================

Selected SourceEvidence ids
`-- ProjectProjectionRecord
    |-- projectId -----------------------------------------> Project
    |-- targetRepresentationId ----------------------------> ProjectRepresentation
    |-- targetRevisionId ----------------------------------> pinned ProjectRepresentationRevision
    `-- links[]
        |-- sourceEvidenceIds[] ----------------------------> SourceEvidence
        `-- targetNodeIds[] --------------------------------> CallTreeNode
            |
            v
        projectSelection()
        `-- ProjectionSelectionResult
            |-- matched / unmatched SourceEvidence ids
            |-- target nodes
            |-- visible nodes = targets + ancestors + descendants
            |-- projected nodes = targets + descendants
            |-- rows { nodeId, sourceLine }
            `-- ascii call tree
                `-- displayed by MappingPoc companion pane

Selected CallTreeNode ids
`-- CompactionProjection
    |-- links[]
    |   |-- callTreeNodeId --------------------------------> CallTreeNode
    |   |-- file ------------------------------------------> CompactionSnapshot
    |   `-- treeLines[]
    `-- files: CompactionSnapshot[]
        |-- sourceFile
        |-- sourceSha256
        |-- compacted tree
        `-- mappings: treeLine -> sourceRanges
            |
            v
        projectCompactions()
        `-- projected files
            |-- visible rows = selected rows + ancestors
            |-- selectedLineCount
            |-- sourceRanges ------------------------------> original source file
            `-- ascii compacted source tree
                `-- companion pane --> ProjectedSourcePane
