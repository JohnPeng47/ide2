|   +-- Rendering contract and coordinator state
|   |   +-- ICoordinatedRendering supplies a window and four phase callbacks
|   |   \-- EditorRenderingCoordinator.INSTANCE

|   +-- scheduleCoordinatedRendering(rendering)
|   |   +-- append rendering to shared queue
|   |   +-- _scheduleRender(rendering.window)
|   |   \-- return cancellation handle
|   |       \-- WHEN dispose() is called
|   |           +-- rendering no longer queued? --> return
|   |           +-- remove rendering
|   |           \-- queue now empty?

|   |               \-- yes --> dispose every window's runner --> clear runner map
|   |
|   +-- EditorRenderingCoordinator._scheduleRender(window)
|   |   \-- runner already registered for window?
|   |       +-- yes --> return
|   |       \-- no --> create callback and register/store animation-frame handle
|   |           +-- dom.runAtThisOrScheduleAtNextAnimationFrame(window, runner, 100)
|   |           \-- WHEN runner executes
|   |               +-- delete this window's runner entry
|   |               \-- _onRenderScheduled()
|   |
|   +-- _onRenderScheduled()
|   |   +-- snapshot shared queue --> replace shared queue with []
|   |   \-- finish each full batch pass before starting the next
|   |       +-- 1. Every rendering: safeInvokeNoArg(prepareRenderText)
|   |       |      \-- retain result in viewportDatas[i]
|   |       +-- 2. Every rendering: viewportDatas[i] present?
|   |       |      +-- no  --> datas[i] = null; skip
|   |       |      \-- yes --> datas[i] = safeInvokeNoArg(renderText(viewport))
|   |       +-- 3. Every rendering: datas[i] present?
|   |       |      +-- no  --> skip
|   |       |      \-- yes --> safeInvokeNoArg(prepareRender(parts, ctx))
|   |       \-- 4. Every rendering: datas[i] present?
|   |              +-- no  --> skip
|   |              \-- yes --> safeInvokeNoArg(render(parts, ctx))
|   |                  \-- prepareRender's return value does not gate this call
