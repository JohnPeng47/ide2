+-- Render requests and shared frame coordination
|   |
|   +-- render(now, everything)
|   |   |
|   |   +-- if everything:
|   |   |   force _viewLines dirty
|   |   |   force every _viewPart dirty
|   |   |
|   |   now ? _flushAccumulatedAndRenderNow() : _scheduleRender()
|   |
|   +-- View._scheduleRender()
|   |   if disposed: throw BugIndicatingError
|   |   if _renderAnimationFrame already set: return
|   |   |
|   |   +-- if NativeEditContext:
|   |   |   setEditContextOnDomNode()
|   |   |
|   |   rendering = _createCoordinatedRendering()
|   |   |
|   |   \-- store coordinator.scheduleCoordinatedRendering(window, callbacks) handle
|   |       |
|   |       +-- prepareRenderText callback
|   |       |   if disposed: throw BugIndicatingError  // before try
|   |       |   try: return rendering.prepareRenderText()
|   |       |   finally: _renderAnimationFrame = null
|   |       |
|   |       +-- renderText callback
|   |       |   if disposed: throw BugIndicatingError
|   |       |   return rendering.renderText(viewport)
|   |       |
|   |       +-- prepareRender callback
|   |       |   if disposed: throw BugIndicatingError
|   |       |   return rendering.prepareRender(parts, ctx)
|   |       |
|   |       \-- render callback
|   |           if disposed: throw BugIndicatingError
|   |           return rendering.render(parts, ctx)
|   |
|   +-- _flushAccumulatedAndRenderNow()
|   |   rendering = _createCoordinatedRendering()
|   |   viewport = safeInvokeNoArg(() => rendering.prepareRenderText())
|   |   if !viewport: return
|   |   data = safeInvokeNoArg(() => rendering.renderText(viewport))
|   |   if !data: return
|   |   [parts, ctx] = data
|   |   safeInvokeNoArg(() => rendering.prepareRender(parts, ctx))
|   |   safeInvokeNoArg(() => rendering.render(parts, ctx))
|   |   // a prepare failure does not gate render
|   |
|   +-- EditorRenderingCoordinator: singleton batch scheduler
|   |   |
|   |   +-- state
|   |   |   shared rendering queue
|   |   |   animation-frame handle per CodeWindow
|   |   |
|   |   +-- scheduleCoordinatedRendering(rendering)
|   |   |   queue.push(rendering)
|   |   |   _scheduleRender(rendering.window)
|   |   |   |
|   |   |   \-- return cancellation handle
|   |   |       |
|   |   |       \-- on dispose:
|   |   |           if rendering no longer queued: return
|   |   |           remove rendering from queue
|   |   |           |
|   |   |           \-- if queue now empty:
|   |   |               dispose ALL window runners
|   |   |               clear runner map
|   |   |
|   |   +-- _scheduleRender(window)
|   |   |   if runner already registered: return
|   |   |   |
|   |   |   \-- register runner via dom.runAtThisOrScheduleAtNextAnimationFrame
|   |   |       priority = 100
|   |   |       store returned handle for window
|   |   |       |
|   |   |       \-- on runner firing:
|   |   |           delete this window's runner entry
|   |   |           _onRenderScheduled()
|   |   |
|   |   \-- _onRenderScheduled()
|   |       batch = snapshot shared queue
|   |       shared queue = []
|   |       |
|   |       +-- PASS 1: for every editor i
|   |       |   viewportDatas[i] = safeInvokeNoArg(() => prepareRenderText())
|   |       |
|   |       +-- PASS 2: for every editor i
|   |       |   if !viewportDatas[i]: datas[i] = null; continue
|   |       |   datas[i] = safeInvokeNoArg(() => renderText(viewportDatas[i]))
|   |       |
|   |       +-- PASS 3: for every editor i
|   |       |   if datas[i]: safeInvokeNoArg(() => prepareRender(parts, ctx))
|   |       |   // result ignored
|   |       |
|   |       \-- PASS 4: for every editor i
|   |           if datas[i]: safeInvokeNoArg(() => render(parts, ctx))
|   |
|   \-- safeInvokeNoArg(func)
|       try: return func()
|       |
|       \-- catch error:
|           onUnexpectedError(error)
|           return null

+-- Per-view rendering phases
|   |
|   +-- _getViewPartsToRender()
|   |   return _viewParts.filter(part => part.shouldRender())
|   |   // preserve order
|   |
|   \-- _createCoordinatedRendering(): return four callbacks
|       |
|       +-- prepareRenderText()
|       |   |
|       |   +-- if lane recomputation requested:
|       |   |   clear flag
|       |   |   model = _computeGlyphMarginLanes()

|       |   |
|       |   inputLatency.onRenderStart()
|       |   if editor DOM disconnected: return null
|       |   parts = _getViewPartsToRender()

|       |   lines = get viewport lines
|       |   viewModel.setViewport(start, end, center)
|       |   viewportData = ViewportData(selections, lines, whitespace, viewModel)
|       |   |
|       |   +-- for each currently dirty _viewPart:
|       |   |   part.onBeforeRender(viewportData)
|       |   |
|       |   return viewportData
|       |
|       +-- renderText(viewportData)
|       |   |
|       |   +-- if DOM _viewLines dirty:
|       |   |   _viewLines.renderText(viewportData)
|       |   |   _viewLines.onDidRender()
|       |   |
|       |   +-- if GPU _viewLinesGpu dirty:
|       |   |   _viewLinesGpu.renderText(viewportData)
|       |   |   _viewLinesGpu.onDidRender()
|       |   |
|       |   parts = _getViewPartsToRender()

|       |   return [parts, RenderingContext(layout, viewportData, DOM lines, GPU lines)]
