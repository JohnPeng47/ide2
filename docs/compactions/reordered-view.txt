+-- Render requests, shared scheduling and render phases
|   +-- render(now, everything)
|   |   +-- everything? --> force DOM _viewLines and every _viewPart to render
|   |   \-- now?
|   |       +-- true --> _flushAccumulatedAndRenderNow()
|   |       \-- false --> _scheduleRender()
|   |
|   +-- View._scheduleRender()
|   |   +-- Disposed? --> throw BugIndicatingError
|   |   \-- _renderAnimationFrame already set? --> leave existing request in place
|   |       +-- Otherwise: NativeEditContext? --> setEditContextOnDomNode()
|   |       \-- Create phase callbacks; enqueue with coordinator for this DOM window; save handle
|   |           +-- WHEN prepareRenderText callback runs
|   |           |   +-- Disposed? --> throw BugIndicatingError before entering try/finally
|   |           |   \-- Invoke rendering.prepareRenderText(); finally clear _renderAnimationFrame
|   |           +-- WHEN renderText callback runs: disposed? throw; otherwise delegate
|   |           +-- WHEN prepareRender callback runs: disposed? throw; otherwise delegate
|   |           \-- WHEN render callback runs: disposed? throw; otherwise delegate
|   |
|   +-- ICoordinatedRendering: phase data contract
|   |   +-- A window identifies the animation-frame scheduling target
|   |   +-- prepareRenderText() --> ViewportData or null
|   |   +-- renderText(ViewportData) --> [ViewPart[], RenderingContext]
|   |   +-- prepareRender(parts, context) --> void
|   |   \-- render(parts, restricted context) --> void
|   |
|   +-- EditorRenderingCoordinator: shared state
|   |   +-- INSTANCE coordinates a shared queue
|   |   \-- _animationFrameRunners stores at most one registered handle per window
|   |
|   +-- scheduleCoordinatedRendering(rendering)
|   |   +-- 1. Push rendering onto shared queue
|   |   +-- 2. _scheduleRender(rendering.window)
|   |   \-- 3. Return cancellation handle
|   |       \-- WHEN dispose() runs
|   |           +-- Rendering absent from queue? --> return
|   |           +-- Otherwise remove rendering
|   |           \-- Queue now empty? no --> keep runners; yes --> cancel them
|   |               \-- yes --> dispose every window's runner --> clear runner map
|   |
|   +-- EditorRenderingCoordinator._scheduleRender(window)
|   |   \-- Runner already registered? --> no new scheduling
|   |       +-- Otherwise define runner callback
|   |       |   \-- WHEN runner runs: delete this window's entry --> _onRenderScheduled()
|   |       \-- Register through dom.runAtThisOrScheduleAtNextAnimationFrame(window, runner, 100)
|   |
|   +-- _onRenderScheduled(): sequential whole-batch passes
|   |   +-- 1. Snapshot shared queue --> replace shared queue with []
|   |   +-- 2. For every queued rendering: safeInvokeNoArg(prepareRenderText) --> viewportDatas[i]
|   |   +-- 3. After pass 2, visit every rendering for text output
|   |   |   +-- No viewportDatas[i]? --> datas[i] = null; skip renderText
|   |   |   \-- Otherwise safeInvokeNoArg(renderText(viewportData)) --> datas[i]
|   |   +-- 4. After pass 3, visit every rendering for preparation
|   |   |   +-- No datas[i]? --> skip
|   |   |   \-- Otherwise safeInvokeNoArg(prepareRender(parts, ctx)); ignore its result
|   |   \-- 5. After pass 4, visit every rendering for final rendering
|   |       +-- No datas[i]? --> skip; previous prepareRender success is not a gate
|   |       \-- Otherwise safeInvokeNoArg(render(parts, ctx)); ignore its result
|   |
|   +-- _flushAccumulatedAndRenderNow(): immediate single-view path
|   |   +-- 1. _createCoordinatedRendering()
|   |   +-- 2. safeInvokeNoArg(prepareRenderText); no viewport data? --> return
|   |   +-- 3. safeInvokeNoArg(renderText); no returned data? --> return
|   |   \-- 4. safeInvokeNoArg(prepareRender) --> safeInvokeNoArg(render), even after preparation error
|   |
|   +-- safeInvokeNoArg(func)
|   |   +-- Invoke func(); successful return --> return that value
|   |   \-- Throws --> onUnexpectedError(error) --> return null
|   |
|   +-- _createCoordinatedRendering(): returns callbacks
|   |   +-- prepareRenderText()
|   |   |   +-- Glyph lanes dirty? yes --> recompute before continuing
|   |   |   |   +-- yes --> clear dirty flag --> _computeGlyphMarginLanes()
|   |   |   |   \-- Set configured glyph lane count to model.requiredLanes
|   |   |   +-- Record inputLatency.onRenderStart()
|   |   |   +-- Editor DOM disconnected? --> return null
|   |   |   +-- No DOM text work and no dirty _viewParts? --> return null
|   |   |   +-- Read viewport lines --> set model viewport
|   |   |   +-- Build ViewportData from selections, line viewport, whitespace and view model
|   |   |   +-- Every currently dirty _viewPart --> onBeforeRender(viewportData)
|   |   |   \-- Return viewportData
|   |   |
|   |   +-- renderText(viewportData)
|   |   |   +-- DOM _viewLines dirty? --> renderText --> onDidRender
|   |   |   +-- GPU _viewLinesGpu exists and is dirty? --> renderText --> onDidRender
|   |   |   +-- Recollect dirty parts via _getViewPartsToRender(); text rendering may trigger scroll events
|   |   |   \-- Return [dirty parts, new RenderingContext(layout, viewport, DOM lines, GPU lines)]
|   |   |
|   |   +-- prepareRender(parts, ctx): for each part --> prepareRender(ctx)
|   |   |
|   |   \-- render(parts, ctx): for each part --> render(ctx) --> onDidRender()
|   |
|   \-- _getViewPartsToRender()
|       \-- Keep _viewParts whose shouldRender() is true, preserving part order
