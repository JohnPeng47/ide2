|   |   +-- unchanged entry count: return state
|   |   +-- filter key from each selection.referenceKeys; discard selections with no remaining keys

|       +-- additions = uniqueRefs absent from existing; referenceKeys = all uniqueRefs keys
|       +-- diagramSelection = supplied selection with referenceKeys when nonempty; otherwise undefined
|       +-- selectionExists = selection key present in state, or true when no selection supplied
|       +-- no additions and selectionExists: return state
|       \-- append additions and any new selection; preserve scopeKey
